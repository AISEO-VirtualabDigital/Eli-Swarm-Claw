   132→// ─── Helper: Count-up animation hook (fixed count) ───────────
   133→function useMetricCounts(targets: number[], duration = 800) {
   134→  const [counts, setCounts] = useState(targets.map(() => 0));
   135→  const [started, setStarted] = useState(false);
   136→
   137→  useEffect(() => {
   138→    if (!started) return;
   139→    const startTime = performance.now();
   140→  let rafId: number;
   141→    const step = (now: number) => {
   142→      const elapsed = now - startTime;
   143→      const progress = Math.min(elapsed / duration, 1);
   144→      const eased = 1 - Math.pow(1 - progress, 3);
   145→      setCounts(targets.map((t) => Math.round(eased * t)));
   146→      if (progress < 1) rafId = requestAnimationFrame(step);
   147→    };
   148→    rafId = requestAnimationFrame(step);
   149→    return () => cancelAnimationFrame(rafId);
   150→  }, [started, duration, ...targets]);
   151→
   152→  return { counts, start: () => setStarted(true) };
   153→}

   156→function useLiveTime() {
   157→  const [time, setTime] = useState('');
   158→  useEffect(() => {
   159→    const update = () => {
   160→      const now = new Date();
   161→      setTime(
   162→        now.toLocaleTimeString('en-US', {
   163→          hour: '2-digit',
   164→          minute: '2-digit',
   165→          second: '2-digit',
   166→          hour12: false,
   167→        })
   168→      );
   169→    };
   170→    update();
   171→    const interval = setInterval(update, 1000);
   172→    return () => clearInterval(interval);
   173→  }, []);
   174→  return time;
   175→}

   576→        id: 'welcome',
   577→        role: 'eli',
   578→        content: ELI_WELCOME,
   579→        timestamp: new Date(),
   580→      },
   581→    ]);
   582→  }, []);
   583→
   584→  const handleSend = async () => {
   585→    const trimmed = input.trim();
   586→    if (!trimmed || isLoading) return;
   587→
   588→    const userMsg: ChatMessage = {
   589→      id: `user-${Date.now()}`,
   590→      role: 'user',
   591→      content: trimmed,
   592→      timestamp: new Date(),
   593→    };
   594→
   595→    setMessages((prev) => [...prev, userMsg]);
   596→    setInput('');
   597→    setIsLoading(true);
   598→
   599→    // Reset textarea height
   600→    if (textareaRef.current) {
   601→      textareaRef.current.style.height = 'auto';
   602→    }
   603→
   604→    try {
   605→      const history = messages
   606→        .filter((m) => m.id !== 'welcome')
   607→        .map((m) => ({ role: m.role, content: m.content }));
   608→
   609→      const res = await fetch('/api/eli-chat', {
   610→        method: 'POST',
   611→        headers: { 'Content-Type': 'application/json' },

   658→      let processed = line.replace(/\*\*(.+?)\*\*/g, '<strong class="text-[#f5f6ff] font-semibold">$1</strong>');

   662→      if (processed.startsWith('• ') || processed.startsWith('- ')) {

   664→          <div key={i} className="flex gap-2 pl-1">
   665→            <span className="text-[#8b5cf6] mt-0.5">•</span>
   666→            <span dangerouslySetInnerHTML={{ __html: processed.replace(/^[•-]\s*/, '') }} />
   667→          </div>
   668→        );
   669→      }
   670→      if (processed.trim() === '') return <div key={i} className="h-2" />;
   671→      return (
   672→        <p key={i} dangerouslySetInnerHTML={{ __html: processed }} className="leading-relaxed" />
   673→      );
   674→    });
   675→  };
   676→
   677→  return (
   678→    <div className="flex flex-col h-full animate-fadeIn">
   679→      {/* Chat Header */}
   680→      <div
   681→        className="px-5 py-3 flex items-center gap-3 flex-shrink-0"
   682→        style={{ borderBottom: '1px solid #252a46' }}

   684→        <div
   685→          className="w-8 h-8 rounded-full flex items-center justify-center text-xs font-bold"
   686→          style={{
   687→            background: 'linear-gradient(135deg, #8b5cf6, #6d4aff)',
   688→            boxShadow: '0 0 12px rgba(139,92,246,0.3)',
   689→          }}
   690→        >
   691→          E
   692→        </div>
   693→        <div>
   694→          <div className="text-sm font-semibold text-[#f5f6ff]">Eli</div>
   695→          <div className="text-[10px] text-[#36d399] flex items-center gap-1">
   696→            <div className="w-1.5 h-1.5 rounded-full bg-[#36d399]" />
   697→            Growth Intelligence Online
   698→          </div>
   699→        </div>
   700→      </div>
   701→
   702→      {/* Messages */}
   703→      <ScrollArea className="flex-1 px-4 md:px-6 py-4">
   704→        <div className="max-w-3xl mx-auto space-y-4">
   705→          {messages.map((msg) => (
   706→            <div
   707→              key={msg.id}
   708→              className={`animate-slideUp flex ${
   709→                msg.role === 'user' ? 'justify-end' : 'justify-start'

   711→            >
   712→              <div
   713→                className={`max-w-2xl px-4 py-3 text-sm leading-relaxed ${
   714→                  msg.role === 'user'
   715→                    ? 'rounded-2xl rounded-tr-sm text-white'
   716→                    : 'rounded-2xl rounded-tl-sm text-[#f5f6ff]'
   717→                }`}
   718→                style={
   719→                  msg.role === 'user'
   720→                    ? {
   721→                        background: 'linear-gradient(135deg, #8b5cf6, #6d4aff)',
   722→                        boxShadow: '0 4px 15px rgba(139,92,246,0.25)',
   723→                      }
   724→                    : {
   725→                        backgroundColor: '#0d1020',
   726→                        border: '1px solid #252a46',
   727→                      }
   728→                }
   729→              >
   730→                {msg.role === 'eli' && (
   731→                  <div className="flex items-center gap-1.5 mb-2">

   733→                    <span className="text-[10px] text-[#9ba4c5]">
   734→                      {msg.timestamp.toLocaleTimeString('en-US', {
   735→                        hour: '2-digit',
   736→                        minute: '2-digit',
   737→                        hour12: true,
   738→                      })}
   739→                    </span>
   740→                  </div>
   741→                )}
   742→                <div className="text-[#f5f6ff]/90">{renderContent(msg.content)}</div>
   743→                {msg.sources && msg.sources.length > 0 && (
   744→                  <div
   745→                    className="mt-3 pt-2 flex flex-wrap gap-1.5"

   761→            </div>

   765→          {isLoading && (
   766→            <div className="flex justify-start animate-fadeIn">
   767→              <div
   768→                className="rounded-2xl rounded-tl-sm px-4 py-3"
   769→                style={{
   770→                  backgroundColor: '#0d1020',
   771→                  border: '1px solid #252a46',
   772→                }}
   773→              >
   774→                <div className="flex items-center gap-1.5">
