   164→   164→// ─── Helper: Count-up animation hook (fixed count) ───────────
   165→   165→function useMetricCounts(targets: number[], duration = 800) {
   166→   166→  const [counts, setCounts] = useState(targets.map(() => 0));
   167→   167→  const [started, setStarted] = useState(false);
   168→   168→
   169→   169→  useEffect(() => {
   170→   170→    if (!started) return;
   171→   171→    const startTime = performance.now();
   172→   172→  let rafId: number;
   173→   173→    const step = (now: number) => {
   174→   174→      const elapsed = now - startTime;
   175→   175→      const progress = Math.min(elapsed / duration, 1);
   176→   176→      const eased = 1 - Math.pow(1 - progress, 3);
   177→   177→      setCounts(targets.map((t) => Math.round(eased * t)));
   178→   178→      if (progress < 1) rafId = requestAnimationFrame(step);
   179→   179→    };
   180→   180→    rafId = requestAnimationFrame(step);
   181→   181→    return () => cancelAnimationFrame(rafId);
   182→   182→  }, [started, duration, ...targets]);
   183→   183→
   184→   184→  return { counts, start: () => setStarted(true) };
   185→   185→}

   188→   188→function useLiveTime() {
   189→   189→  const [time, setTime] = useState('');
   190→   190→  useEffect(() => {
   191→   191→    const update = () => {
   192→   192→      const now = new Date();
   193→   193→      setTime(
   194→   194→        now.toLocaleTimeString('en-US', {
   195→   195→          hour: '2-digit',
   196→   196→          minute: '2-digit',
   197→   197→          second: '2-digit',
   198→   198→          hour12: false,
   199→   199→        })
   200→   200→      );
   201→   201→    };
   202→   202→    update();
   203→   203→    const interval = setInterval(update, 1000);
   204→   204→    return () => clearInterval(interval);
   205→   205→  }, []);
   206→   206→  return time;
   207→   207→}

   589→   589→function ChatView() {
   590→   590→  const [messages, setMessages] = useState<ChatMessage[]>([]);
   591→   591→  const [input, setInput] = useState('');
   592→   592→  const [isLoading, setIsLoading] = useState(false);
   593→   593→  const messagesEndRef = useRef<HTMLDivElement>(null);
   594→   594→  const textareaRef = useRef<HTMLTextAreaElement>(null);
   595→   595→
   596→   596→  const scrollToBottom = useCallback(() => {
   597→   597→    messagesEndRef.current?.scrollIntoView({ behavior: 'smooth' });
   598→   598→  }, []);
   599→   599→
   600→   600→  useEffect(() => {
   601→   601→    scrollToBottom();
   602→   602→  }, [messages, isLoading, scrollToBottom]);
   603→   603→
   604→   604→  // Welcome message on mount
   605→   605→  useEffect(() => {
   606→   606→    setMessages([
   607→   607→      {
   608→   608→        id: 'welcome',
   609→   609→        role: 'eli',
   610→   610→        content: ELI_WELCOME,
   611→   611→        timestamp: new Date(),
   612→   612→      },
   613→   613→    ]);
   614→   614→  }, []);
   615→   615→
   616→   616→  const handleSend = async () => {
   617→   617→    const trimmed = input.trim();
   618→   618→    if (!trimmed || isLoading) return;
   619→   619→
   620→   620→    const userMsg: ChatMessage = {
   621→   621→      id: `user-${Date.now()}`,
   622→   622→      role: 'user',
   623→   623→      content: trimmed,
   624→   624→      timestamp: new Date(),
   625→   625→    };
   626→   626→
   627→   627→    setMessages((prev) => [...prev, userMsg]);
   628→   628→    setInput('');
   629→   629→    setIsLoading(true);
   630→   630→
   631→   631→    // Reset textarea height
   632→   632→    if (textareaRef.current) {
   633→   633→      textareaRef.current.style.height = 'auto';
   634→   634→    }
   635→   635→
   636→   636→    try {
   637→   637→      const history = messages
   638→   638→        .filter((m) => m.id !== 'welcome')
   639→   639→        .map((m) => ({ role: m.role, content: m.content }));
   640→   640→
   641→   641→      const res = await fetch('/api/eli-chat', {
   642→   642→        method: 'POST',
   643→   643→        headers: { 'Content-Type': 'application/json' },
   644→   644→        body: JSON.stringify({ message: trimmed, history }),
   645→   645→      });
   646→   646→
   647→   647→      const data = await res.json();
   648→   648→
   649→   649→      const eliMsg: ChatMessage = {
   650→   650→        id: `eli-${Date.now()}`,
   651→   651→        role: 'eli',

   660→   660→        id: `error-${Date.now()}`,

   700→   700→  };
   701→   701→
