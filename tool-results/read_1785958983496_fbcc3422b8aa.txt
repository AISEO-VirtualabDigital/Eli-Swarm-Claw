   132→   132→// ─── Helper: Count-up animation hook (fixed count) ───────────
   133→   133→function useMetricCounts(targets: number[], duration = 800) {
   134→   134→  const [counts, setCounts] = useState(targets.map(() => 0));
   135→   135→  const [started, setStarted] = useState(false);
   136→   136→
   137→   137→  useEffect(() => {
   138→   138→    if (!started) return;
   139→   139→    const startTime = performance.now();
   140→   140→  let rafId: number;
   141→   141→    const step = (now: number) => {
   142→   142→      const elapsed = now - startTime;
   143→   143→      const progress = Math.min(elapsed / duration, 1);
   144→   144→      const eased = 1 - Math.pow(1 - progress, 3);
   145→   145→      setCounts(targets.map((t) => Math.round(eased * t)));
   146→   146→      if (progress < 1) rafId = requestAnimationFrame(step);
   147→   147→    };
   148→   148→    rafId = requestAnimationFrame(step);
   149→   149→    return () => cancelAnimationFrame(rafId);
   150→   150→  }, [started, duration, ...targets]);
   151→   151→
   152→   152→  return { counts, start: () => setStarted(true) };
   153→   153→}

   156→   156→function useLiveTime() {

   158→   158→  useEffect(() => {

   161→   161→      setTime(
   162→   162→        now.toLocaleTimeString('en-US', {
   163→   163→          hour: '2-digit',
   164→   164→          minute: '2-digit',
   165→   165→          second: '2-digit',
   166→   166→          hour12: false,
   167→   167→        })
   168→   168→      );
   169→   169→    };
   170→   170→    update();
   171→   171→    const interval = setInterval(update, 1000);
   172→   172→    return () => clearInterval(interval);
   173→   173→  }, []);
   174→   174→  return time;
   175→   175→}
