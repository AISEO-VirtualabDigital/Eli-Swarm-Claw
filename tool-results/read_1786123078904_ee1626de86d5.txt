     1→/**
     2→ * Open Claw Engine — Infinite Email Generator + Autonomous Reader
     3→ * 
     4→ * "Open Claw" = self-sufficient, zero-cost, multi-provider email system.
     5→ * The claw reaches out through multiple free services to generate and read
     6→ * temporary emails without any API keys or payment.
     7→ * 
     8→ * Providers:
     9→ *   1. Guerrilla Mail (primary) — session-based, instant, 1hr TTL, no registration
    10→ *   2. mail.tm (secondary) — account-based, JWT auth, @web-library.net
    11→ *   3. OpenInbox (tertiary) — creation-only (free), can't read without paid key
    12→ * 
    13→ * The claw auto-failovers between providers. If one fails, it slices to the next.
    14→ */
    15→
    16→// ─── Types ────────────────────────────────────────────────────────

    20→  email: string;

    23→  createdAt: number;
    24→  sessionData: any;         // provider-specific session (sid_token, jwt, etc.)
    25→  emailCount: number;
    26→  status: 'fresh' | 'polling' | 'expired' | 'error';
    27→}

   109→async function probeGuerrilla(): Promise<ProviderHealth> {
   110→  const start = Date.now();
   111→  try {
   112→    const res = await fetch(`${GM_BASE}/ajax.php?f=get_email_address&lang=en`, {
   113→      signal: AbortSignal.timeout(PROBE_TIMEOUT),
   114→    });
   115→    const latency = Date.now() - start;
   116→    return { status: res.ok ? 'ok' : 'error', latencyMs: latency, lastProbeAt: Date.now(), consecutiveFailures: 0, penalty: 0, tier: 0 };
   117→  } catch (err) {
   118→    return { status: (err as Error).name === 'TimeoutError' ? 'timeout' : 'broken', latencyMs: Date.now() - start, lastProbeAt: Date.now(), consecutiveFailures: 1, penalty: 3, tier: 0 };
   119→  }
   120→}

   162→async function gmCheck(inbox: ClawInbox): Promise<ClawEmail[]> {
   163→  const { sidToken } = inbox.sessionData;
   164→  const res = await fetch(
   165→    `${GM_BASE}/ajax.php?f=check_email&lang=en&sid_token=${sidToken}&seq=0`
   166→  );
   167→  if (!res.ok) throw new Error(`Guerrilla check failed: ${res.status}`);
   168→  const data = await res.json();
   169→  const list = data.list || [];
   170→  return list.map((m: any) => ({
   171→    id: `gm-${m.mail_id}`,
   172→    from: m.mail_from,
   173→    subject: m.mail_subject || '',
   174→    bodyText: m.mail_body || m.mail_excerpt || '',
   175→    bodyHtml: m.mail_body || '',
   176→    receivedAt: (m.mail_timestamp || Date.now() / 1000) * 1000,
   177→  }));

   200→let mtDomains: string[] = [];
   201→
   202→async function mtGetDomains(): Promise<string[]> {
   203→  if (mtDomains.length > 0) return mtDomains;
   204→  const res = await fetch(`${MT_BASE}/domains`);
   205→  if (!res.ok) return ['web-library.net']; // fallback
   206→  const data = await res.json();
   207→  mtDomains = (data['hydra:member'] || []).map((d: any) => d.domain);
   208→  return mtDomains;

   211→async function mtCreate(): Promise<ClawInbox> {
   212→  const domains = await mtGetDomains();
   213→  const domain = domains[0] || 'web-library.net';
   214→  const localPart = `eli-claw-${Date.now().toString(36)}-${Math.random().toString(36).slice(2, 7)}`;
   215→  const address = `${localPart}@${domain}`;
   216→  const password = `Claw${Date.now().toString(36)}!X${Math.random().toString(36).slice(2, 8)}`;
   217→
   218→  // Create account
   219→  const accRes = await fetch(`${MT_BASE}/accounts`, {
   220→    method: 'POST',
   221→    headers: { 'Content-Type': 'application/json' },
   222→    body: JSON.stringify({ address, password }),
   223→  });
   224→  if (!accRes.ok) throw new Error(`mail.tm create account failed: ${accRes.status}`);
   225→  const accData = await accRes.json();
   226→
   227→  // Get JWT
   228→  const tokRes = await fetch(`${MT_BASE}/token`, {
   229→    method: 'POST',
   230→    headers: { 'Content-Type': 'application/json' },
   231→    body: JSON.stringify({ address, password }),
   232→  });
   233→  if (!tokRes.ok) throw new Error(`mail.tm get token failed: ${tokRes.status}`);
   234→  const tokData = await tokRes.json();
   235→
   236→  return {
   237→    id: `mt-${accData.id}`,
   238→    email: address,
   239→    provider: 'mailtm',
   240→    expiresAt: Date.now() + DEFAULT_CONFIG.inboxTtlMs,
   241→    createdAt: Date.now(),
   242→    sessionData: { token: tokData.token, accountId: accData.id, password },
   243→    emailCount: 0,
   244→    status: 'fresh',
   245→  };
   246→}

   248→async function mtCheck(inbox: ClawInbox): Promise<ClawEmail[]> {
   249→  const { token } = inbox.sessionData;
   250→  const res = await fetch(`${MT_BASE}/messages`, {
   251→    headers: { Authorization: `Bearer ${token}` },
   252→  });
   253→  if (!res.ok) throw new Error(`mail.tm check failed: ${res.status}`);
   254→  const data = await res.json();
   255→  const members = data['hydra:member'] || [];
   256→  return members.map((m: any) => ({
   257→    id: `mt-${m.id}`,
   258→    from: m.from?.address || m.from?.name || '',
   259→    subject: m.subject || '',
   260→    bodyText: m.intro || '',
   261→    bodyHtml: '',
   262→    receivedAt: new Date(m.createdAt).getTime(),
   263→  }));
   264→}

   266→async function mtFetchEmail(inbox: ClawInbox, emailId: string): Promise<ClawEmail | null> {
   267→  const { token } = inbox.sessionData;
   268→  const numericId = emailId.replace('mt-', '');
   269→  const res = await fetch(`${MT_BASE}/messages/${numericId}`, {
   270→    headers: { Authorization: `Bearer ${token}` },
   271→  });
   272→  if (!res.ok) return null;
   273→  const data = await res.json();
   274→  // mail.tm returns HTML in data.html[].content
   275→  const htmlParts = data.html || [];
   276→  const textParts = data.text || [];
   277→  return {
   278→    id: emailId,
   279→    from: data.from?.address || '',
   280→    subject: data.subject || '',
   281→    bodyText: textParts.map((p: any) => p.content).join('\n'),
   282→    bodyHtml: htmlParts.map((p: any) => p.content).join('\n'),
   283→    receivedAt: new Date(data.createdAt).getTime(),
   284→  };
   285→}

   319→// ─── Key Extraction ──────────────────────────────────────────────
   320→
   321→const KEY_PATTERNS: Array<{ name: string; pattern: RegExp; envVar: string }> = [
   322→  { name: 'gemini', pattern: /AIza[0-9A-Za-z_-]{35}/, envVar: 'GEMINI_API_KEY' },
   323→  { name: 'openai', pattern: /sk-[a-zA-Z0-9]{20,}/, envVar: 'OPENAI_API_KEY' },
   324→  { name: 'anthropic', pattern: /sk-ant-[a-zA-Z0-9-]{20,}/, envVar: 'ANTHROPIC_API_KEY' },
   325→  { name: 'google-generic', pattern: /AQ\.[a-zA-Z0-9_-]{30,}/, envVar: 'GEMINI_API_KEY' },
   326→];
   327→
   328→function stripHtml(html: string): string {
   329→  return html
   330→    .replace(/<pre[^>]*>/gi, '')
   331→    .replace(/<[^>]+>/g, ' ')
   332→    .replace(/&amp;/g, '&')
   333→    .replace(/&lt;/g, '<')
   334→    .replace(/&gt;/g, '>')
   335→    .replace(/&quot;/g, '"')
   336→    .replace(/&#039;/g, "'")
   337→    .replace(/\s+/g, ' ')
   338→    .trim();
   339→}
   340→
   341→function extractKeysFromEmail(email: ClawEmail): Array<{ service: string; key: string; envVar: string }> {
   342→  const plainText = `${email.subject} ${email.bodyText} ${stripHtml(email.bodyHtml)}`;
   343→  const results: Array<{ service: string; key: string; envVar: string }> = [];
   344→
   345→  for (const kp of KEY_PATTERNS) {
   346→    const matches = plainText.match(new RegExp(kp.pattern.source, 'g'));
   347→    if (matches) {
   348→      for (const match of matches) {
   349→        results.push({ service: kp.name, key: match, envVar: kp.envVar });
   350→      }
   351→    }
   352→  }
   353→
   354→  return results;
   355→}

   357→// ─── Open Claw Engine ─────────────────────────────────────────────
   358→
   359→export class OpenClaw {
   360→  private config: ClawConfig;
   361→  private inboxes: ClawInbox[] = [];
   362→  private stats: ClawState['providerStats'];
   363→  private totalGenerated = 0;
   364→  private totalEmailsRead = 0;
   365→  private totalKeysExtracted = 0;
   366→  private lastKeyExtracted: string | null = null;
   367→  private pollTimer: ReturnType<typeof setInterval> | null = null;
   368→  private keyCallback: ((service: string, key: string, envVar: string) => void) | null = null;
   369→  private providerHealth: Record<ProviderName, ProviderHealth>;
   370→  private roundRobinIdx = 0;  // OmniKey: round-robin across equal-penalty providers
   371→
   372→  constructor(config: Partial<ClawConfig> = {}) {
   373→    this.config = { ...DEFAULT_CONFIG, ...config };
   374→    this.stats = {
   375→      guerrilla: { generated: 0, errors: 0, emailsRead: 0 },
   376→      mailtm: { generated: 0, errors: 0, emailsRead: 0 },
   377→      openinbox: { generated: 0, errors: 0, emailsRead: 0 },
   378→    };
   379→    this.providerHealth = {
   380→      guerrilla: { status: 'ok', latencyMs: 0, lastProbeAt: 0, consecutiveFailures: 0, penalty: 0, tier: 0 },
   381→      mailtm: { status: 'ok', latencyMs: 0, lastProbeAt: 0, consecutiveFailures: 0, penalty: 0, tier: 0 },
   382→      openinbox: { status: 'ok', latencyMs: 0, lastProbeAt: 0, consecutiveFailures: 0, penalty: 0, tier: 1 },
   383→    };
   384→  }
   385→
   386→  // ─── Two-Phase Provider Selection (Agent-Reach + OmniKey) ──────
   387→  // Phase 1: Probe ALL providers in parallel
   388→  // Phase 2: Select best by ok → penalty asc → latency asc, round-robin tiebreak
   389→
   390→  async probeAllProviders(): Promise<void> {
   391→    const [gm, mt, oi] = await Promise.all([probeGuerrilla(), probeMailTm(), probeOpenInbox()]);
   392→
   393→    // Penalty decay (OmniKey: -1 per 2 min)
   394→    for (const [name, fresh] of [['guerrilla', gm], ['mailtm', mt], ['openinbox', oi]] as const) {
   395→      const prev = this.providerHealth[name[0] as ProviderName];
   396→      const minsSinceProbe = prev.lastProbeAt > 0 ? (Date.now() - prev.lastProbeAt) / 120_000 : 0;
   397→      const decay = Math.floor(minsSinceProbe);
   398→      const decayedPenalty = Math.max(0, prev.penalty - decay);
   399→
   400→      if (fresh.status !== 'ok') {
   401→        fresh.consecutiveFailures = prev.consecutiveFailures + 1;
   402→        fresh.penalty = Math.min(decayedPenalty + 3, 10);
   403→      } else {
   404→        fresh.consecutiveFailures = 0;
   405→        fresh.penalty = Math.max(decayedPenalty - 1, 0);
   406→      }
   407→    }
   408→
   409→    this.providerHealth = { guerrilla: gm, mailtm: mt, openinbox: oi };
   410→    console.log(`[CLAW] Probed:`, Object.entries(this.providerHealth).map(
   411→      ([n, h]) => `${n}=${h.status}(${h.latencyMs}ms,p${h.penalty})`
   412→    ).join(' | '));
   413→  }
   414→
   415→  async getBestProvider(): Promise<ProviderName> {
   416→    const stale = Object.values(this.providerHealth).every(h => Date.now() - h.lastProbeAt > 120_000);
   417→    if (stale) await this.probeAllProviders();
   418→
   419→    const sorted = Object.entries(this.providerHealth)
   420→      .filter(([, h]) => h.status === 'ok' && h.consecutiveFailures < 3)
   421→      .sort(([, a], [, b]) => a.penalty - b.penalty || a.latencyMs - b.latencyMs);
   422→
   423→    if (sorted.length > 0) {
   424→      const bestPenalty = sorted[0][1].penalty;
   425→      const topTier = sorted.filter(([, h]) => h.penalty === bestPenalty);
   426→      const idx = this.roundRobinIdx % topTier.length;
   427→      this.roundRobinIdx++;
   428→      return topTier[idx][0] as ProviderName;
   429→    }
   430→
   431→    // Fallback: try providers with lowest consecutive failures
   432→    const fallback = Object.entries(this.providerHealth)
   433→      .sort(([, a], [, b]) => a.consecutiveFailures - b.consecutiveFailures);
   434→    return (fallback[0]?.[0] || 'guerrilla') as ProviderName;
   435→  }
   436→
   437→  /**
   438→   * Record provider success/failure for penalty scoring (OmniKey pattern)
   439→   */
   440→  recordProviderResult(provider: ProviderName, success: boolean) {
   441→    const h = this.providerHealth[provider];
   442→    if (!h) return;
   443→    if (success) {
   444→      h.penalty = Math.max(0, h.penalty - 1);
   445→      h.consecutiveFailures = 0;
   446→    } else {
   447→      h.penalty = Math.min(h.penalty + 3, 10);
   448→      h.consecutiveFailures++;
   449→    }
   450→  }
   451→
   452→  /**
   453→   * Register a callback for when a key is extracted
   454→   */
   455→  onKey(callback: (service: string, key: string, envVar: string) => void) {
   456→    this.keyCallback = callback;
   457→  }
   458→
   459→  /**
   460→   * Generate a new inbox. Two-phase selection (Agent-Reach):
   461→   * Probe all providers → select best by penalty + latency + round-robin
   462→   */
   463→  async generate(provider?: ProviderName): Promise<ClawInbox> {
   464→    const targetProvider = provider || await this.getBestProvider();
   465→
   466→    try {
   467→      let inbox: ClawInbox;
   468→      switch (targetProvider) {
   469→        case 'guerrilla': inbox = await gmCreate(); break;
   470→        case 'mailtm':   inbox = await mtCreate(); break;
   471→        case 'openinbox': inbox = await oiCreate(); break;
   472→      }
   473→
   474→      this.recordProviderResult(targetProvider, true);
   475→      this.inboxes.push(inbox);
   476→      this.totalGenerated++;
   477→      this.stats[targetProvider].generated++;
   478→
   479→      // Trim pool
   480→      if (this.inboxes.length > this.config.maxInboxes) {
   481→        this.inboxes = this.inboxes.slice(-this.config.maxInboxes);
   482→      }
   483→
   484→      console.log(`[CLAW] Generated ${targetProvider} inbox: ${inbox.email} (expires in ${Math.round((inbox.expiresAt - Date.now()) / 60000)}min)`);
   485→      return inbox;
   486→    } catch (err) {
   487→      this.recordProviderResult(targetProvider, false);
   488→      this.stats[targetProvider].errors++;
   489→      console.warn(`[CLAW] ${targetProvider} generation failed:`, (err as Error).message);
   490→      throw err;
   491→    }
   492→  }
   493→
   494→  /**
   495→   * Check a specific inbox for new emails
   496→   */
   497→  async checkInbox(inbox: ClawInbox): Promise<ClawEmail[]> {
   498→    // Skip expired
   499→    if (Date.now() > inbox.expiresAt) {
   500→      inbox.status = 'expired';
   501→      return [];
   502→    }
   503→
   504→    inbox.status = 'polling';
   505→
   506→    try {
   507→      let emails: ClawEmail[];
   508→      switch (inbox.provider) {
   509→        case 'guerrilla': emails = await gmCheck(inbox); break;
   510→        case 'mailtm':   emails = await mtCheck(inbox); break;
   511→        case 'openinbox': emails = await oiCheck(inbox); break;
   512→        default: return [];
   513→      }
   514→
   515→      inbox.emailCount = emails.length;
   516→      this.totalEmailsRead += emails.length;
   517→      this.stats[inbox.provider].emailsRead += emails.length;
   518→
   519→      // Auto-extract keys from new emails
   520→      for (const email of emails) {
   521→        const keys = extractKeysFromEmail(email);
   522→        for (const k of keys) {
   523→          console.log(`[CLAW] KEY EXTRACTED [${k.service}]: ${k.key.slice(0, 12)}... from ${inbox.email}`);
   524→          this.totalKeysExtracted++;
   525→          this.lastKeyExtracted = k.key;
   526→          process.env[k.envVar] = k.key;
   527→          this.keyCallback?.(k.service, k.key, k.envVar);
   528→        }
   529→      }
   530→
   531→      return emails;
   532→    } catch (err) {
   533→      inbox.status = 'error';
   534→      this.stats[inbox.provider].errors++;
   535→      console.warn(`[CLAW] Check error [${inbox.provider}]:`, (err as Error).message);
   536→      return [];
   537→    }
   538→  }

   540→  /**
   541→   * Fetch full email body for a specific email
   542→   */
   543→  async fetchEmail(inbox: ClawInbox, emailId: string): Promise<ClawEmail | null> {
   544→    try {
   545→      switch (inbox.provider) {
   546→        case 'guerrilla': return gmFetchEmail(inbox, emailId);
   547→        case 'mailtm':   return mtFetchEmail(inbox, emailId);
   548→        default: return null; // OpenInbox can't fetch without API key
   549→      }
   550→    } catch (err) {
   551→      console.warn(`[CLAW] Fetch email error:`, (err as Error).message);
   552→      return null;
   553→    }
   554→  }

   556→  /**
   557→   * Poll ALL active inboxes for new emails
   558→   */
   559→  async pollAll(): Promise<Map<string, ClawEmail[]>> {
   560→    const results = new Map<string, ClawEmail[]>();
   561→    const now = Date.now();
   562→
   563→    for (const inbox of this.inboxes) {
   564→      if (inbox.status === 'expired' || now > inbox.expiresAt) {
   565→        inbox.status = 'expired';
   566→        continue;
   567→      }
   568→      const emails = await this.checkInbox(inbox);
   569→      if (emails.length > 0) {
   570→        results.set(inbox.email, emails);
   571→      }
   572→    }
   573→
   574→    return results;
   575→  }

   577→  /**
   578→   * Start background polling loop
   579→   */
   580→  startPolling() {
   581→    if (this.pollTimer) return;
   582→    console.log(`[CLAW] Polling started (every ${this.config.pollIntervalMs / 1000}s, ${this.inboxes.length} inboxes)`);
   583→    this.pollTimer = setInterval(() => {
   584→      this.pollAll().then(results => {
   585→        if (results.size > 0) {
   586→          console.log(`[CLAW] Poll: ${results.size} inbox(es) have new emails`);
   587→        }
   588→      }).catch(err => {
   589→        console.error('[CLAW] Poll error:', (err as Error).message);
   590→      });
   591→    }, this.config.pollIntervalMs);
   592→  }
   593→
   594→  stopPolling() {
   595→    if (this.pollTimer) {
   596→      clearInterval(this.pollTimer);
   597→      this.pollTimer = null;
   598→    }
   599→  }

   601→  /**
   602→   * Generate + immediately start polling a new inbox
   603→   */
   604→  async spawn(provider?: 'guerrilla' | 'mailtm' | 'openinbox'): Promise<ClawInbox> {
   605→    const inbox = await this.generate(provider);
   606→    // If this is the first inbox, start the poll loop
   607→    if (!this.pollTimer) {
   608→      this.startPolling();
   609→    }
   610→    return inbox;
   611→  }

   614→   * Get fresh (non-expired) inbox — generates one if none available
   615→   */
   616→  async getFreshInbox(): Promise<ClawInbox> {
   617→    const fresh = this.inboxes.find(
   618→      i => i.status !== 'expired' && Date.now() < i.expiresAt - 5 * 60 * 1000
   619→    );
   620→    if (fresh) return fresh;
   621→    return this.spawn();
   622→  }
