   357→   357→function stripHtml(html: string): string {
   358→   358→  return html
   359→   359→    .replace(/<pre[^>]*>/gi, '')
   360→   360→    .replace(/<[^>]+>/g, ' ')
   361→   361→    .replace(/&amp;/g, '&')
   362→   362→    .replace(/&lt;/g, '<')
   363→   363→    .replace(/&gt;/g, '>')
   364→   364→    .replace(/&quot;/g, '"')
   365→   365→    .replace(/&#039;/g, "'")
   366→   366→    .replace(/\s+/g, ' ')
   367→   367→    .trim();
   368→   368→}
   369→   369→
   370→   370→function extractKeysFromEmail(email: ClawEmail): Array<{ service: string; key: string; envVar: string }> {
   371→   371→  const plainText = `${email.subject} ${email.bodyText} ${stripHtml(email.bodyHtml)}`;
   372→   372→  const results: Array<{ service: string; key: string; envVar: string }> = [];
   373→   373→
   374→   374→  for (const kp of KEY_PATTERNS) {
   375→   375→    const matches = plainText.match(new RegExp(kp.pattern.source, 'g'));
   376→   376→    if (matches) {
   377→   377→      for (const match of matches) {
   378→   378→        results.push({ service: kp.name, key: match, envVar: kp.envVar });
   379→   379→      }
   380→   380→    }
   381→   381→  }
   382→   382→
   383→   383→  return results;
   384→   384→}

   431→   431→  async probeAllProviders(): Promise<void> {
   432→   432→    const [gm, mt, oi] = await Promise.all([probeGuerrilla(), probeMailTm(), probeOpenInbox()]);
   433→   433→
   434→   434→    // Penalty decay (OmniKey: -1 per 2 min)
   435→   435→    for (const [name, fresh] of [['guerrilla', gm], ['mailtm', mt], ['openinbox', oi]] as const) {
   436→   436→      const prev = this.providerHealth[name[0] as ProviderName];
   437→   437→      const minsSinceProbe = prev.lastProbeAt > 0 ? (Date.now() - prev.lastProbeAt) / 120_000 : 0;
   438→   438→      const decay = Math.floor(minsSinceProbe);
   439→   439→      const decayedPenalty = Math.max(0, prev.penalty - decay);
   440→   440→
   441→   441→      if (fresh.status !== 'ok') {
   442→   442→        fresh.consecutiveFailures = prev.consecutiveFailures + 1;
   443→   443→        fresh.penalty = Math.min(decayedPenalty + 3, 10);
   444→   444→      } else {
   445→   445→        fresh.consecutiveFailures = 0;
   446→   446→        fresh.penalty = Math.max(decayedPenalty - 1, 0);
   447→   447→      }
   448→   448→    }
   449→   449→
   450→   450→    this.providerHealth = { guerrilla: gm, mailtm: mt, openinbox: oi };
   451→   451→    console.log(`[CLAW] Probed:`, Object.entries(this.providerHealth).map(
   452→   452→      ([n, h]) => `${n}=${h.status}(${h.latencyMs}ms,p${h.penalty})`
   453→   453→    ).join(' | '));
   454→   454→  }
   455→   455→
   456→   456→  async getBestProvider(skipProviders?: Set<string>): Promise<ProviderName> {
   457→   457→    const stale = Object.values(this.providerHealth).every(h => Date.now() - h.lastProbeAt > 120_000);
   458→   458→    if (stale) await this.probeAllProviders();
   459→   459→
   460→   460→    // OmniMail: provider resolution chain — filter by skip-set, then sort
   461→   461→    const candidates = Object.entries(this.providerHealth)
   462→   462→      .filter(([name, h]) => h.status === 'ok' && h.consecutiveFailures < 3)
   463→   463→      .filter(([name]) => !skipProviders?.has(name))
   464→   464→      .sort(([, a], [, b]) => a.penalty - b.penalty || a.latencyMs - b.latencyMs);
   465→   465→
   466→   466→    if (candidates.length > 0) {
   467→   467→      const bestPenalty = candidates[0][1].penalty;
   468→   468→      const topTier = candidates.filter(([, h]) => h.penalty === bestPenalty);
   469→   469→      const idx = this.roundRobinIdx % topTier.length;
   470→   470→      this.roundRobinIdx++;
   471→   471→      return topTier[idx][0] as ProviderName;
   472→   472→    }
   473→   473→
   474→   474→    // OmniMail: fallback — ignore skip-set, use lowest consecutive failures
   475→   475→    if (skipProviders?.size) {
   476→   476→      console.warn(`[CLAW] All providers in skip-set, falling back to best available`);
   477→   477→      return this.getBestProvider();
   478→   478→    }
   479→   479→
   480→   480→    const fallback = Object.entries(this.providerHealth)
   481→   481→      .sort(([, a], [, b]) => a.consecutiveFailures - b.consecutiveFailures);
   482→   482→    return (fallback[0]?.[0] || 'guerrilla') as ProviderName;
   483→   483→  }

   495→   495→      h.penalty = Math.min(h.penalty + 3, 10);

   497→   497→    }

   507→   507→  /**
   508→   508→   * Generate a new inbox. Two-phase selection (Agent-Reach):
   509→   509→   * Probe all providers → select best by penalty + latency + round-robin
   510→   510→   */
   511→   511→  async generate(provider?: ProviderName): Promise<ClawInbox> {
   512→   512→    const targetProvider = provider || await this.getBestProvider();
   513→   513→
   514→   514→    try {
   515→   515→      let inbox: ClawInbox;
   516→   516→      switch (targetProvider) {
   517→   517→        case 'guerrilla': inbox = await gmCreate(); break;
   518→   518→        case 'mailtm':   inbox = await mtCreate(); break;
   519→   519→        case 'openinbox': inbox = await oiCreate(); break;
   520→   520→      }
   521→   521→
   522→   522→      this.recordProviderResult(targetProvider, true);
   523→   523→      this.inboxes.push(inbox);
   524→   524→      this.totalGenerated++;
   525→   525→      this.stats[targetProvider].generated++;
   526→   526→
   527→   527→      // Trim pool
   528→   528→      if (this.inboxes.length > this.config.maxInboxes) {
   529→   529→        this.inboxes = this.inboxes.slice(-this.config.maxInboxes);
   530→   530→      }
   531→   531→
   532→   532→      console.log(`[CLAW] Generated ${targetProvider} inbox: ${inbox.email} (expires in ${Math.round((inbox.expiresAt - Date.now()) / 60000)}min)`);
   533→   533→      return inbox;
   534→   534→    } catch (err) {
   535→   535→      this.recordProviderResult(targetProvider, false);
   536→   536→      this.stats[targetProvider].errors++;
   537→   537→      // OmniMail: classify error for upstream handling
   538→   538→      const errMsg = (err as Error).message || '';
   539→   539→      const retryable = [429, 408, 500, 502, 503, 504].some(c => errMsg.includes(String(c)))
   540→   540→        || /timeout|econnrefused|econnreset/i.test(errMsg);
   541→   541→      console.warn(`[CLAW] ${targetProvider} generation failed (${retryable ? 'retryable' : 'non-retryable'}):`, errMsg);
   542→   542→      (err as any)._clawRetryable = retryable;
   543→   543→      throw err;
   544→   544→    }
   545→   545→  }

   549→   549→   */
   550→   550→  async checkInbox(inbox: ClawInbox): Promise<ClawEmail[]> {
   551→   551→    // Skip expired
   552→   552→    if (Date.now() > inbox.expiresAt) {
   553→   553→      inbox.status = 'expired';
   554→   554→      return [];
   555→   555→    }
   556→   556→
   557→   557→    inbox.status = 'polling';
   558→   558→
   559→   559→    try {
   560→   560→      let emails: ClawEmail[];
   561→   561→      switch (inbox.provider) {
   562→   562→        case 'guerrilla': emails = await gmCheck(inbox); break;
   563→   563→        case 'mailtm':   emails = await mtCheck(inbox); break;
   564→   564→        case 'openinbox': emails = await oiCheck(inbox); break;
   565→   565→        default: return [];
   566→   566→      }
   567→   567→
   568→   568→      inbox.emailCount = emails.length;
   569→   569→      this.totalEmailsRead += emails.length;
   570→   570→      this.stats[inbox.provider].emailsRead += emails.length;
   571→   571→
   572→   572→      // Auto-extract keys from new emails — Tier 1: pending queue + validation
   573→   573→      for (const email of emails) {
   574→   574→        const keys = extractKeysFromEmail(email);
   575→   575→        for (const k of keys) {
   576→   576→          console.log(`[CLAW] KEY EXTRACTED [${k.service}]: ${k.key.slice(0, 12)}... from ${inbox.email}`);
   577→   577→          this.totalKeysExtracted++;
   578→   578→          this.lastKeyExtracted = k.key;
   579→   579→
   580→   580→          const pendingId = `pending-${Date.now()}-${Math.random().toString(36).slice(2, 6)}`;
   581→   581→          const pending: PendingKey = {
   582→   582→            id: pendingId,
   583→   583→            service: k.service,
   584→   584→            key: k.key,
   585→   585→            envVar: k.envVar,
   586→   586→            extractedAt: Date.now(),
   587→   587→            inboxId: inbox.id,
   588→   588→            inboxEmail: inbox.email,
   589→   589→            status: 'pending',
   590→   590→          };
   591→   591→          this.pendingKeys.push(pending);
   592→   592→          if (this.pendingKeys.length > 20) this.pendingKeys = this.pendingKeys.slice(-20);
   593→   593→
   594→   594→          audit('key.extracted', `${k.service} key from ${inbox.email}`, {
   595→   595→            pendingId, keyPreview: k.key.slice(0, 12) + '...', inboxId: inbox.id,
   596→   596→          });
   597→   597→
   598→   598→          if (this.autoApproveKeys) {
   599→   599→            // Auto-approve: validate then inject
   600→   600→            await this.validateAndInject(pending);
   601→   601→          }
   602→   602→
   603→   603→          // Notify callback (with pendingId so caller can approve/reject)
   604→   604→          this.keyCallback?.(k.service, k.key, k.envVar, pendingId);
   605→   605→        }
   606→   606→      }
   607→   607→
   608→   608→      return emails;
   609→   609→    } catch (err) {
   610→   610→      inbox.status = 'error';
   611→   611→      this.stats[inbox.provider].errors++;
   612→   612→      console.warn(`[CLAW] Check error [${inbox.provider}]:`, (err as Error).message);
   613→   613→      return [];
   614→   614→    }
   615→   615→  }

   731→   731→  async validateAndInject(pending: PendingKey): Promise<boolean> {
   732→   732→    try {
   733→   733→      // Quick format check first (cheaper than API call)
   734→   734→      const svc = SERVICES_MAP[pending.service];
   735→   735→      if (svc?.pattern && !svc.pattern.test(pending.key)) {
   736→   736→        pending.status = 'rejected';
   737→   737→        pending.validationError = 'Format mismatch';
   738→   738→        pending.validatedAt = Date.now();
   739→   739→        audit('key.rejected', `${pending.service} key failed format check`, { pendingId: pending.id });
   740→   740→        return false;
   741→   741→      }
   742→   742→
   743→   743→      // For Gemini keys, make a test call
   744→   744→      if (pending.service === 'gemini' || pending.envVar === 'GEMINI_API_KEY') {
   745→   745→        try {
   746→   746→          const { GoogleGenerativeAI } = await import('@google/generative-ai');
   747→   747→          const genAI = new GoogleGenerativeAI(pending.key);
   748→   748→          const model = genAI.getGenerativeModel({ model: 'gemini-2.0-flash' });
   749→   749→          const result = await model.generateContent({
   750→   750→            contents: [{ role: 'user', parts: [{ text: 'Say OK' }] }],
   751→   751→          });
   752→   752→          const text = result?.response?.text();
   753→   753→          if (!text || text.length === 0) throw new Error('Empty response');
   754→   754→        } catch (err) {
   755→   755→          pending.status = 'rejected';
   756→   756→          pending.validationError = (err as Error).message?.slice(0, 200);
   757→   757→          pending.validatedAt = Date.now();
   758→   758→          audit('key.rejected', `${pending.service} key failed validation: ${(err as Error).message?.slice(0, 100)}`, {
   759→   759→            pendingId: pending.id,

   761→   761→          return false;
   762→   762→        }
   763→   763→      }
   764→   764→
   765→   765→      // Passed validation — inject
   766→   766→      pending.status = 'approved';
   767→   767→      pending.validatedAt = Date.now();
   768→   768→      process.env[pending.envVar] = pending.key;
   769→   769→      audit('key.approved', `${pending.service} key validated and injected`, { pendingId: pending.id });
   770→   770→      return true;
   771→   771→    } catch (err) {
   772→   772→      pending.status = 'rejected';
   773→   773→      pending.validationError = (err as Error).message;
   774→   774→      pending.validatedAt = Date.now();
   775→   775→      audit('key.rejected', `${pending.service} key validation error: ${(err as Error).message?.slice(0, 100)}`, {
   776→   776→        pendingId: pending.id,
   777→   777→      });
   778→   778→      return false;
   779→   779→    }
   780→   780→  }
