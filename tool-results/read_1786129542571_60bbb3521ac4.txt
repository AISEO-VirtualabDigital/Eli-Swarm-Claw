     1→     1→/**
     2→     2→ * Open Claw Engine v2 — Infinite Email Generator + Autonomous Reader
     3→     3→ * 
     4→     4→ * "Open Claw" = self-sufficient, zero-cost, multi-provider email system.
     5→     5→ * 
     6→     6→ * Wired patterns:
     7→     7→ *   OmniKey — penalty tracker with decay, round-robin, skip-set on failure
     8→     8→ *   OmniMail — provider resolution chain, retryable error classification
     9→     9→ *   Agent-Reach — probe-don't-guess health checks
    10→    10→ *   OmniRoute (diegosouzapw) — provider adapter abstraction
    11→    11→ *   browser-use — decoupled browser task generation
    12→    12→ * 
    13→    13→ * Providers:
    14→    14→ *   1. Guerrilla Mail (primary) — session-based, instant, 1hr TTL, no registration
    15→    15→ *   2. mail.tm (secondary) — account-based, JWT auth, @web-library.net
    16→    16→ *   3. OpenInbox (tertiary) — creation-only (free), can't read without paid key
    17→    17→ * 
    18→    18→ * Provider resolution chain (OmniMail pattern):
    19→    19→ *   probe all → filter by skip-set → sort by penalty+latency → round-robin tiebreak

   334→   334→function stripHtml(html: string): string {
   335→   335→  return html
   336→   336→    .replace(/<pre[^>]*>/gi, '')
   337→   337→    .replace(/<[^>]+>/g, ' ')
   338→   338→    .replace(/&amp;/g, '&')
   339→   339→    .replace(/&lt;/g, '<')
   340→   340→    .replace(/&gt;/g, '>')
   341→   341→    .replace(/&quot;/g, '"')
   342→   342→    .replace(/&#039;/g, "'")
   343→   343→    .replace(/\s+/g, ' ')
   344→   344→    .trim();
   345→   345→}

   347→   347→function extractKeysFromEmail(email: ClawEmail): Array<{ service: string; key: string; envVar: string }> {
   348→   348→  const plainText = `${email.subject} ${email.bodyText} ${stripHtml(email.bodyHtml)}`;
   349→   349→  const results: Array<{ service: string; key: string; envVar: string }> = [];
   350→   350→
   351→   351→  for (const kp of KEY_PATTERNS) {
   352→   352→    const matches = plainText.match(new RegExp(kp.pattern.source, 'g'));
   353→   353→    if (matches) {
   354→   354→      for (const match of matches) {
   355→   355→        results.push({ service: kp.name, key: match, envVar: kp.envVar });
   356→   356→      }
   357→   357→    }
   358→   358→  }
   359→   359→
   360→   360→  return results;
   361→   361→}

   365→   365→export class OpenClaw {
   366→   366→  private config: ClawConfig;
   367→   367→  private inboxes: ClawInbox[] = [];
   368→   368→  private stats: ClawState['providerStats'];
   369→   369→  private totalGenerated = 0;
   370→   370→  private totalEmailsRead = 0;
   371→   371→  private totalKeysExtracted = 0;
   372→   372→  private lastKeyExtracted: string | null = null;
   373→   373→  private pollTimer: ReturnType<typeof setInterval> | null = null;
   374→   374→  private keyCallback: ((service: string, key: string, envVar: string) => void) | null = null;
   375→   375→  private providerHealth: Record<ProviderName, ProviderHealth>;
   376→   376→  private roundRobinIdx = 0;  // OmniKey: round-robin across equal-penalty providers
   377→   377→
   378→   378→  constructor(config: Partial<ClawConfig> = {}) {
   379→   379→    this.config = { ...DEFAULT_CONFIG, ...config };
   380→   380→    this.stats = {
   381→   381→      guerrilla: { generated: 0, errors: 0, emailsRead: 0 },
   382→   382→      mailtm: { generated: 0, errors: 0, emailsRead: 0 },
   383→   383→      openinbox: { generated: 0, errors: 0, emailsRead: 0 },
   384→   384→    };
   385→   385→    this.providerHealth = {
   386→   386→      guerrilla: { status: 'ok', latencyMs: 0, lastProbeAt: 0, consecutiveFailures: 0, penalty: 0, tier: 0 },
   387→   387→      mailtm: { status: 'ok', latencyMs: 0, lastProbeAt: 0, consecutiveFailures: 0, penalty: 0, tier: 0 },
   388→   388→      openinbox: { status: 'ok', latencyMs: 0, lastProbeAt: 0, consecutiveFailures: 0, penalty: 0, tier: 1 },
   389→   389→    };
   390→   390→  }
   391→   391→
   392→   392→  // ─── Two-Phase Provider Selection (Agent-Reach + OmniKey) ──────
   393→   393→  // Phase 1: Probe ALL providers in parallel
   394→   394→  // Phase 2: Select best by ok → penalty asc → latency asc, round-robin tiebreak
   395→   395→
   396→   396→  async probeAllProviders(): Promise<void> {
   397→   397→    const [gm, mt, oi] = await Promise.all([probeGuerrilla(), probeMailTm(), probeOpenInbox()]);
   398→   398→
   399→   399→    // Penalty decay (OmniKey: -1 per 2 min)
   400→   400→    for (const [name, fresh] of [['guerrilla', gm], ['mailtm', mt], ['openinbox', oi]] as const) {
   401→   401→      const prev = this.providerHealth[name[0] as ProviderName];
   402→   402→      const minsSinceProbe = prev.lastProbeAt > 0 ? (Date.now() - prev.lastProbeAt) / 120_000 : 0;
   403→   403→      const decay = Math.floor(minsSinceProbe);
   404→   404→      const decayedPenalty = Math.max(0, prev.penalty - decay);
   405→   405→
   406→   406→      if (fresh.status !== 'ok') {
   407→   407→        fresh.consecutiveFailures = prev.consecutiveFailures + 1;
   408→   408→        fresh.penalty = Math.min(decayedPenalty + 3, 10);
   409→   409→      } else {
   410→   410→        fresh.consecutiveFailures = 0;
   411→   411→        fresh.penalty = Math.max(decayedPenalty - 1, 0);
   412→   412→      }
   413→   413→    }
   414→   414→
   415→   415→    this.providerHealth = { guerrilla: gm, mailtm: mt, openinbox: oi };
   416→   416→    console.log(`[CLAW] Probed:`, Object.entries(this.providerHealth).map(
   417→   417→      ([n, h]) => `${n}=${h.status}(${h.latencyMs}ms,p${h.penalty})`
   418→   418→    ).join(' | '));
   419→   419→  }
   420→   420→
   421→   421→  async getBestProvider(skipProviders?: Set<string>): Promise<ProviderName> {
   422→   422→    const stale = Object.values(this.providerHealth).every(h => Date.now() - h.lastProbeAt > 120_000);
   423→   423→    if (stale) await this.probeAllProviders();
   424→   424→
   425→   425→    // OmniMail: provider resolution chain — filter by skip-set, then sort
   426→   426→    const candidates = Object.entries(this.providerHealth)
   427→   427→      .filter(([name, h]) => h.status === 'ok' && h.consecutiveFailures < 3)
   428→   428→      .filter(([name]) => !skipProviders?.has(name))
   429→   429→      .sort(([, a], [, b]) => a.penalty - b.penalty || a.latencyMs - b.latencyMs);
   430→   430→
   431→   431→    if (candidates.length > 0) {
   432→   432→      const bestPenalty = candidates[0][1].penalty;
   433→   433→      const topTier = candidates.filter(([, h]) => h.penalty === bestPenalty);
   434→   434→      const idx = this.roundRobinIdx % topTier.length;
   435→   435→      this.roundRobinIdx++;
   436→   436→      return topTier[idx][0] as ProviderName;
   437→   437→    }
   438→   438→
   439→   439→    // OmniMail: fallback — ignore skip-set, use lowest consecutive failures
   440→   440→    if (skipProviders?.size) {
   441→   441→      console.warn(`[CLAW] All providers in skip-set, falling back to best available`);
   442→   442→      return this.getBestProvider();
   443→   443→    }
   444→   444→
   445→   445→    const fallback = Object.entries(this.providerHealth)
   446→   446→      .sort(([, a], [, b]) => a.consecutiveFailures - b.consecutiveFailures);
   447→   447→    return (fallback[0]?.[0] || 'guerrilla') as ProviderName;
   448→   448→  }

   450→   450→  /**
   451→   451→   * Record provider success/failure for penalty scoring (OmniKey pattern)
   452→   452→   */
   453→   453→  recordProviderResult(provider: ProviderName, success: boolean) {
   454→   454→    const h = this.providerHealth[provider];
   455→   455→    if (!h) return;
   456→   456→    if (success) {
   457→   457→      h.penalty = Math.max(0, h.penalty - 1);
   458→   458→      h.consecutiveFailures = 0;
   459→   459→    } else {
   460→   460→      h.penalty = Math.min(h.penalty + 3, 10);
   461→   461→      h.consecutiveFailures++;
   462→   462→    }
   463→   463→  }

   472→   472→  /**
   473→   473→   * Generate a new inbox. Two-phase selection (Agent-Reach):
   474→   474→   * Probe all providers → select best by penalty + latency + round-robin
   475→   475→   */
   476→   476→  async generate(provider?: ProviderName): Promise<ClawInbox> {
   477→   477→    const targetProvider = provider || await this.getBestProvider();
   478→   478→
   479→   479→    try {
   480→   480→      let inbox: ClawInbox;
   481→   481→      switch (targetProvider) {
   482→   482→        case 'guerrilla': inbox = await gmCreate(); break;
   483→   483→        case 'mailtm':   inbox = await mtCreate(); break;
   484→   484→        case 'openinbox': inbox = await oiCreate(); break;
   485→   485→      }
   486→   486→
   487→   487→      this.recordProviderResult(targetProvider, true);
   488→   488→      this.inboxes.push(inbox);
   489→   489→      this.totalGenerated++;
   490→   490→      this.stats[targetProvider].generated++;
   491→   491→
   492→   492→      // Trim pool
   493→   493→      if (this.inboxes.length > this.config.maxInboxes) {
   494→   494→        this.inboxes = this.inboxes.slice(-this.config.maxInboxes);
   495→   495→      }
   496→   496→
   497→   497→      console.log(`[CLAW] Generated ${targetProvider} inbox: ${inbox.email} (expires in ${Math.round((inbox.expiresAt - Date.now()) / 60000)}min)`);
   498→   498→      return inbox;
   499→   499→    } catch (err) {
   500→   500→      this.recordProviderResult(targetProvider, false);
   501→   501→      this.stats[targetProvider].errors++;
   502→   502→      // OmniMail: classify error for upstream handling
   503→   503→      const errMsg = (err as Error).message || '';
   504→   504→      const retryable = [429, 408, 500, 502, 503, 504].some(c => errMsg.includes(String(c)))
   505→   505→        || /timeout|econnrefused|econnreset/i.test(errMsg);
   506→   506→      console.warn(`[CLAW] ${targetProvider} generation failed (${retryable ? 'retryable' : 'non-retryable'}):`, errMsg);
   507→   507→      (err as any)._clawRetryable = retryable;
   508→   508→      throw err;
   509→   509→    }
   510→   510→  }

   512→   512→  /**
   513→   513→   * Check a specific inbox for new emails
   514→   514→   */
   515→   515→  async checkInbox(inbox: ClawInbox): Promise<ClawEmail[]> {
   516→   516→    // Skip expired
   517→   517→    if (Date.now() > inbox.expiresAt) {
   518→   518→      inbox.status = 'expired';
   519→   519→      return [];
   520→   520→    }
   521→   521→
   522→   522→    inbox.status = 'polling';
   523→   523→
   524→   524→    try {
   525→   525→      let emails: ClawEmail[];
   526→   526→      switch (inbox.provider) {
   527→   527→        case 'guerrilla': emails = await gmCheck(inbox); break;
   528→   528→        case 'mailtm':   emails = await mtCheck(inbox); break;
   529→   529→        case 'openinbox': emails = await oiCheck(inbox); break;
   530→   530→        default: return [];
   531→   531→      }
   532→   532→
   533→   533→      inbox.emailCount = emails.length;
   534→   534→      this.totalEmailsRead += emails.length;
   535→   535→      this.stats[inbox.provider].emailsRead += emails.length;
   536→   536→
   537→   537→      // Auto-extract keys from new emails
   538→   538→      for (const email of emails) {
   539→   539→        const keys = extractKeysFromEmail(email);
   540→   540→        for (const k of keys) {
   541→   541→          console.log(`[CLAW] KEY EXTRACTED [${k.service}]: ${k.key.slice(0, 12)}... from ${inbox.email}`);
   542→   542→          this.totalKeysExtracted++;
   543→   543→          this.lastKeyExtracted = k.key;
   544→   544→          process.env[k.envVar] = k.key;
   545→   545→          this.keyCallback?.(k.service, k.key, k.envVar);
   546→   546→        }
   547→   547→      }
   548→   548→
   549→   549→      return emails;
   550→   550→    } catch (err) {
   551→   551→      inbox.status = 'error';
   552→   552→      this.stats[inbox.provider].errors++;
   553→   553→      console.warn(`[CLAW] Check error [${inbox.provider}]:`, (err as Error).message);
   554→   554→      return [];
   555→   555→    }
   556→   556→  }

   574→   574→  /**
   575→   575→   * Poll ALL active inboxes for new emails
   576→   576→   */
   577→   577→  async pollAll(): Promise<Map<string, ClawEmail[]>> {
   578→   578→    const results = new Map<string, ClawEmail[]>();
   579→   579→    const now = Date.now();
   580→   580→
   581→   581→    for (const inbox of this.inboxes) {
   582→   582→      if (inbox.status === 'expired' || now > inbox.expiresAt) {
   583→   583→        inbox.status = 'expired';
   584→   584→        continue;
   585→   585→      }
   586→   586→      const emails = await this.checkInbox(inbox);
   587→   587→      if (emails.length > 0) {
   588→   588→        results.set(inbox.email, emails);
   589→   589→      }
   590→   590→    }
   591→   591→
   592→   592→    return results;
   593→   593→  }
   594→   594→
   595→   595→  /**
   596→   596→   * Start background polling loop
   597→   597→   */
   598→   598→  startPolling() {
   599→   599→    if (this.pollTimer) return;
   600→   600→    console.log(`[CLAW] Polling started (every ${this.config.pollIntervalMs / 1000}s, ${this.inboxes.length} inboxes)`);
   601→   601→    this.pollTimer = setInterval(() => {
   602→   602→      this.pollAll().then(results => {
   603→   603→        if (results.size > 0) {
   604→   604→          console.log(`[CLAW] Poll: ${results.size} inbox(es) have new emails`);
   605→   605→        }
   606→   606→      }).catch(err => {
   607→   607→        console.error('[CLAW] Poll error:', (err as Error).message);
   608→   608→      });
   609→   609→    }, this.config.pollIntervalMs);
   610→   610→  }
   611→   611→
   612→   612→  stopPolling() {
   613→   613→    if (this.pollTimer) {
   614→   614→      clearInterval(this.pollTimer);
   615→   615→      this.pollTimer = null;
   616→   616→    }
   617→   617→  }

   619→   619→  /**
   620→   620→   * Generate + immediately start polling a new inbox
   621→   621→   */
   622→   622→  async spawn(provider?: 'guerrilla' | 'mailtm' | 'openinbox'): Promise<ClawInbox> {
   623→   623→    const inbox = await this.generate(provider);
   624→   624→    // If this is the first inbox, start the poll loop

   626→   626→      this.startPolling();
   627→   627→    }
   628→   628→    return inbox;
   629→   629→  }
