     1→/**
     2→ * Open Claw Engine v2 — Infinite Email Generator + Autonomous Reader
     3→ * 
     4→ * "Open Claw" = self-sufficient, zero-cost, multi-provider email system.
     5→ * 
     6→ * Wired patterns:
     7→ *   OmniKey — penalty tracker with decay, round-robin, skip-set on failure
     8→ *   OmniMail — provider resolution chain, retryable error classification
     9→ *   Agent-Reach — probe-don't-guess health checks
    10→ *   OmniRoute (diegosouzapw) — provider adapter abstraction
    11→ *   browser-use — decoupled browser task generation
    12→ * 
    13→ * Providers:
    14→ *   1. Guerrilla Mail (primary) — session-based, instant, 1hr TTL, no registration
    15→ *   2. mail.tm (secondary) — account-based, JWT auth, @web-library.net
    16→ *   3. OpenInbox (tertiary) — creation-only (free), can't read without paid key
    17→ * 
    18→ * Provider resolution chain (OmniMail pattern):
    19→ *   probe all → filter by skip-set → sort by penalty+latency → round-robin tiebreak

   240→async function mtCreate(): Promise<ClawInbox> {
   241→  const domains = await mtGetDomains();
   242→  const domain = domains[0] || 'web-library.net';
   243→  const localPart = `eli-claw-${Date.now().toString(36)}-${Math.random().toString(36).slice(2, 7)}`;
   244→  const address = `${localPart}@${domain}`;
   245→  const password = `Claw${Date.now().toString(36)}!X${Math.random().toString(36).slice(2, 8)}`;
   246→
   247→  // Create account
   248→  const accRes = await fetch(`${MT_BASE}/accounts`, {
   249→    method: 'POST',
   250→    headers: { 'Content-Type': 'application/json' },
   251→    body: JSON.stringify({ address, password }),
   252→  });
   253→  if (!accRes.ok) throw new Error(`mail.tm create account failed: ${accRes.status}`);
   254→  const accData = await accRes.json();
   255→
   256→  // Get JWT
   257→  const tokRes = await fetch(`${MT_BASE}/token`, {
   258→    method: 'POST',
   259→    headers: { 'Content-Type': 'application/json' },
   260→    body: JSON.stringify({ address, password }),
   261→  });
   262→  if (!tokRes.ok) throw new Error(`mail.tm get token failed: ${tokRes.status}`);
   263→  const tokData = await tokRes.json();
   264→
   265→  return {
   266→    id: `mt-${accData.id}`,
   267→    email: address,
   268→    provider: 'mailtm',
   269→    expiresAt: Date.now() + DEFAULT_CONFIG.inboxTtlMs,
   270→    createdAt: Date.now(),
   271→    sessionData: { token: tokData.token, accountId: accData.id, password },
   272→    emailCount: 0,
   273→    status: 'fresh',
   274→  };
   275→}

   350→const KEY_PATTERNS: Array<{ name: string; pattern: RegExp; envVar: string }> = [
   351→  { name: 'gemini', pattern: /AIza[0-9A-Za-z_-]{35}/, envVar: 'GEMINI_API_KEY' },
   352→  { name: 'openai', pattern: /sk-[a-zA-Z0-9]{20,}/, envVar: 'OPENAI_API_KEY' },
   353→  { name: 'anthropic', pattern: /sk-ant-[a-zA-Z0-9-]{20,}/, envVar: 'ANTHROPIC_API_KEY' },
   354→  { name: 'google-generic', pattern: /AQ\.[a-zA-Z0-9_-]{30,}/, envVar: 'GEMINI_API_KEY' },
   355→];
   356→
   357→function stripHtml(html: string): string {
   358→  return html
   359→    .replace(/<pre[^>]*>/gi, '')
   360→    .replace(/<[^>]+>/g, ' ')
   361→    .replace(/&amp;/g, '&')
   362→    .replace(/&lt;/g, '<')
   363→    .replace(/&gt;/g, '>')
   364→    .replace(/&quot;/g, '"')
   365→    .replace(/&#039;/g, "'")
   366→    .replace(/\s+/g, ' ')
   367→    .trim();
   368→}
   369→
   370→function extractKeysFromEmail(email: ClawEmail): Array<{ service: string; key: string; envVar: string }> {
   371→  const plainText = `${email.subject} ${email.bodyText} ${stripHtml(email.bodyHtml)}`;
   372→  const results: Array<{ service: string; key: string; envVar: string }> = [];
   373→
   374→  for (const kp of KEY_PATTERNS) {
   375→    const matches = plainText.match(new RegExp(kp.pattern.source, 'g'));
   376→    if (matches) {
   377→      for (const match of matches) {
   378→        results.push({ service: kp.name, key: match, envVar: kp.envVar });
   379→      }
   380→    }
   381→  }
   382→
   383→  return results;
   384→}

   388→export class OpenClaw {
   389→  private config: ClawConfig;
   390→  private inboxes: ClawInbox[] = [];
   391→  private stats: ClawState['providerStats'];
   392→  private totalGenerated = 0;
   393→  private totalEmailsRead = 0;

   395→  private lastKeyExtracted: string | null = null;
   396→  private pollTimer: ReturnType<typeof setInterval> | null = null;
   397→  private keyCallback: ((service: string, key: string, envVar: string, pendingId: string) => void) | null = null;
   398→  private providerHealth: Record<ProviderName, ProviderHealth>;
   399→  private roundRobinIdx = 0;
   400→  private pendingKeys: PendingKey[] = [];
   401→  private autoApproveKeys = false; // Tier 1: default OFF — keys require manual approval
   402→
   403→  /**
   404→   * Enable/disable auto-approval of extracted keys.
   405→   * When OFF (default): keys go to pending queue, require POST /api/omni?action=approve
   406→   * When ON: keys are validated and auto-injected (legacy behavior)
   407→   */
   408→  setAutoApprove(enabled: boolean) {
   409→    this.autoApproveKeys = enabled;
   410→    console.log(`[CLAW] Auto-approve ${enabled ? 'ENABLED' : 'DISABLED'} — keys will ${enabled ? 'auto-inject' : 'require approval'}`);
   411→  }
   412→
   413→  constructor(config: Partial<ClawConfig> = {}) {
   414→    this.config = { ...DEFAULT_CONFIG, ...config };
   415→    this.stats = {
   416→      guerrilla: { generated: 0, errors: 0, emailsRead: 0 },
   417→      mailtm: { generated: 0, errors: 0, emailsRead: 0 },
   418→      openinbox: { generated: 0, errors: 0, emailsRead: 0 },
   419→    };
   420→    this.providerHealth = {
   421→      guerrilla: { status: 'ok', latencyMs: 0, lastProbeAt: 0, consecutiveFailures: 0, penalty: 0, tier: 0 },
   422→      mailtm: { status: 'ok', latencyMs: 0, lastProbeAt: 0, consecutiveFailures: 0, penalty: 0, tier: 0 },
   423→      openinbox: { status: 'ok', latencyMs: 0, lastProbeAt: 0, consecutiveFailures: 0, penalty: 0, tier: 1 },
   424→    };
   425→  }
   426→
   427→  // ─── Two-Phase Provider Selection (Agent-Reach + OmniKey) ──────
   428→  // Phase 1: Probe ALL providers in parallel
   429→  // Phase 2: Select best by ok → penalty asc → latency asc, round-robin tiebreak
   430→
   431→  async probeAllProviders(): Promise<void> {
   432→    const [gm, mt, oi] = await Promise.all([probeGuerrilla(), probeMailTm(), probeOpenInbox()]);
   433→
   434→    // Penalty decay (OmniKey: -1 per 2 min)
   435→    for (const [name, fresh] of [['guerrilla', gm], ['mailtm', mt], ['openinbox', oi]] as const) {
   436→      const prev = this.providerHealth[name[0] as ProviderName];
   437→      const minsSinceProbe = prev.lastProbeAt > 0 ? (Date.now() - prev.lastProbeAt) / 120_000 : 0;
   438→      const decay = Math.floor(minsSinceProbe);
   439→      const decayedPenalty = Math.max(0, prev.penalty - decay);
   440→
   441→      if (fresh.status !== 'ok') {
   442→        fresh.consecutiveFailures = prev.consecutiveFailures + 1;
   443→        fresh.penalty = Math.min(decayedPenalty + 3, 10);
   444→      } else {
   445→        fresh.consecutiveFailures = 0;
   446→        fresh.penalty = Math.max(decayedPenalty - 1, 0);
   447→      }
   448→    }
   449→
   450→    this.providerHealth = { guerrilla: gm, mailtm: mt, openinbox: oi };
   451→    console.log(`[CLAW] Probed:`, Object.entries(this.providerHealth).map(
   452→      ([n, h]) => `${n}=${h.status}(${h.latencyMs}ms,p${h.penalty})`
   453→    ).join(' | '));
   454→  }
   455→
   456→  async getBestProvider(skipProviders?: Set<string>): Promise<ProviderName> {
   457→    const stale = Object.values(this.providerHealth).every(h => Date.now() - h.lastProbeAt > 120_000);
   458→    if (stale) await this.probeAllProviders();
   459→
   460→    // OmniMail: provider resolution chain — filter by skip-set, then sort
   461→    const candidates = Object.entries(this.providerHealth)
   462→      .filter(([name, h]) => h.status === 'ok' && h.consecutiveFailures < 3)
   463→      .filter(([name]) => !skipProviders?.has(name))
   464→      .sort(([, a], [, b]) => a.penalty - b.penalty || a.latencyMs - b.latencyMs);
   465→
   466→    if (candidates.length > 0) {
   467→      const bestPenalty = candidates[0][1].penalty;
   468→      const topTier = candidates.filter(([, h]) => h.penalty === bestPenalty);
   469→      const idx = this.roundRobinIdx % topTier.length;
   470→      this.roundRobinIdx++;
   471→      return topTier[idx][0] as ProviderName;
   472→    }
   473→
   474→    // OmniMail: fallback — ignore skip-set, use lowest consecutive failures
   475→    if (skipProviders?.size) {
   476→      console.warn(`[CLAW] All providers in skip-set, falling back to best available`);
   477→      return this.getBestProvider();
   478→    }
   479→
   480→    const fallback = Object.entries(this.providerHealth)
   481→      .sort(([, a], [, b]) => a.consecutiveFailures - b.consecutiveFailures);
   482→    return (fallback[0]?.[0] || 'guerrilla') as ProviderName;
   483→  }

   485→  /**
   486→   * Record provider success/failure for penalty scoring (OmniKey pattern)
   487→   */
   488→  recordProviderResult(provider: ProviderName, success: boolean) {
   489→    const h = this.providerHealth[provider];
   490→    if (!h) return;
   491→    if (success) {
   492→      h.penalty = Math.max(0, h.penalty - 1);
   493→      h.consecutiveFailures = 0;
   494→    } else {
   495→      h.penalty = Math.min(h.penalty + 3, 10);
   496→      h.consecutiveFailures++;
   497→    }
   498→  }

   506→
   507→  /**
   508→   * Generate a new inbox. Two-phase selection (Agent-Reach):
   509→   * Probe all providers → select best by penalty + latency + round-robin
   510→   */
   511→  async generate(provider?: ProviderName): Promise<ClawInbox> {
   512→    const targetProvider = provider || await this.getBestProvider();
   513→
   514→    try {
   515→      let inbox: ClawInbox;
   516→      switch (targetProvider) {
   517→        case 'guerrilla': inbox = await gmCreate(); break;
   518→        case 'mailtm':   inbox = await mtCreate(); break;
   519→        case 'openinbox': inbox = await oiCreate(); break;
   520→      }
   521→
   522→      this.recordProviderResult(targetProvider, true);
   523→      this.inboxes.push(inbox);
   524→      this.totalGenerated++;
   525→      this.stats[targetProvider].generated++;
   526→
   527→      // Trim pool
   528→      if (this.inboxes.length > this.config.maxInboxes) {
   529→        this.inboxes = this.inboxes.slice(-this.config.maxInboxes);
   530→      }
   531→
   532→      console.log(`[CLAW] Generated ${targetProvider} inbox: ${inbox.email} (expires in ${Math.round((inbox.expiresAt - Date.now()) / 60000)}min)`);
   533→      return inbox;
   534→    } catch (err) {
   535→      this.recordProviderResult(targetProvider, false);
   536→      this.stats[targetProvider].errors++;
   537→      // OmniMail: classify error for upstream handling
   538→      const errMsg = (err as Error).message || '';
   539→      const retryable = [429, 408, 500, 502, 503, 504].some(c => errMsg.includes(String(c)))
   540→        || /timeout|econnrefused|econnreset/i.test(errMsg);
   541→      console.warn(`[CLAW] ${targetProvider} generation failed (${retryable ? 'retryable' : 'non-retryable'}):`, errMsg);
   542→      (err as any)._clawRetryable = retryable;
   543→      throw err;
   544→    }
   545→  }

   547→  /**
   548→   * Check a specific inbox for new emails
   549→   */
   550→  async checkInbox(inbox: ClawInbox): Promise<ClawEmail[]> {
   551→    // Skip expired
   552→    if (Date.now() > inbox.expiresAt) {
   553→      inbox.status = 'expired';
   554→      return [];
   555→    }
   556→
   557→    inbox.status = 'polling';
   558→
   559→    try {
   560→      let emails: ClawEmail[];
   561→      switch (inbox.provider) {
   562→        case 'guerrilla': emails = await gmCheck(inbox); break;
   563→        case 'mailtm':   emails = await mtCheck(inbox); break;
   564→        case 'openinbox': emails = await oiCheck(inbox); break;
   565→        default: return [];
   566→      }
   567→
   568→      inbox.emailCount = emails.length;
   569→      this.totalEmailsRead += emails.length;
   570→      this.stats[inbox.provider].emailsRead += emails.length;
   571→
   572→      // Auto-extract keys from new emails — Tier 1: pending queue + validation
   573→      for (const email of emails) {
   574→        const keys = extractKeysFromEmail(email);
   575→        for (const k of keys) {
   576→          console.log(`[CLAW] KEY EXTRACTED [${k.service}]: ${k.key.slice(0, 12)}... from ${inbox.email}`);
   577→          this.totalKeysExtracted++;
   578→          this.lastKeyExtracted = k.key;
   579→
   580→          const pendingId = `pending-${Date.now()}-${Math.random().toString(36).slice(2, 6)}`;
   581→          const pending: PendingKey = {
   582→            id: pendingId,
   583→            service: k.service,
   584→            key: k.key,
   585→            envVar: k.envVar,
   586→            extractedAt: Date.now(),
   587→            inboxId: inbox.id,
   588→            inboxEmail: inbox.email,
   589→            status: 'pending',
   590→          };
   591→          this.pendingKeys.push(pending);
   592→          if (this.pendingKeys.length > 20) this.pendingKeys = this.pendingKeys.slice(-20);
   593→
   594→          audit('key.extracted', `${k.service} key from ${inbox.email}`, {
   595→            pendingId, keyPreview: k.key.slice(0, 12) + '...', inboxId: inbox.id,
   596→          });
   597→
   598→          if (this.autoApproveKeys) {
   599→            // Auto-approve: validate then inject
   600→            await this.validateAndInject(pending);
   601→          }
   602→
   603→          // Notify callback (with pendingId so caller can approve/reject)
   604→          this.keyCallback?.(k.service, k.key, k.envVar, pendingId);
   605→        }
   606→      }
   607→
   608→      return emails;
   609→    } catch (err) {
   610→      inbox.status = 'error';
   611→      this.stats[inbox.provider].errors++;
   612→      console.warn(`[CLAW] Check error [${inbox.provider}]:`, (err as Error).message);
   613→      return [];
   614→    }
   615→  }

   633→  /**
   634→   * Poll ALL active inboxes for new emails
   635→   */
   636→  async pollAll(): Promise<Map<string, ClawEmail[]>> {
   637→    const results = new Map<string, ClawEmail[]>();
   638→    const now = Date.now();
   639→
   640→    for (const inbox of this.inboxes) {
   641→      if (inbox.status === 'expired' || now > inbox.expiresAt) {
   642→        inbox.status = 'expired';
   643→        continue;
   644→      }
   645→      const emails = await this.checkInbox(inbox);
   646→      if (emails.length > 0) {
   647→        results.set(inbox.email, emails);
   648→      }
   649→    }
   650→
   651→    return results;
   652→  }
   653→
   654→  /**
   655→   * Start background polling loop
   656→   */
   657→  startPolling() {
   658→    if (this.pollTimer) return;
   659→    console.log(`[CLAW] Polling started (every ${this.config.pollIntervalMs / 1000}s, ${this.inboxes.length} inboxes)`);
   660→    this.pollTimer = setInterval(() => {
   661→      this.pollAll().then(results => {
   662→        if (results.size > 0) {
   663→          console.log(`[CLAW] Poll: ${results.size} inbox(es) have new emails`);
   664→        }
   665→      }).catch(err => {
   666→        console.error('[CLAW] Poll error:', (err as Error).message);
   667→      });
   668→    }, this.config.pollIntervalMs);
   669→  }
   670→
   671→  stopPolling() {
   672→    if (this.pollTimer) {
   673→      clearInterval(this.pollTimer);
   674→      this.pollTimer = null;
   675→    }
   676→  }

   678→  /**
   679→   * Generate + immediately start polling a new inbox
   680→   */
   681→  async spawn(provider?: 'guerrilla' | 'mailtm' | 'openinbox'): Promise<ClawInbox> {
   682→    const inbox = await this.generate(provider);
   683→    // If this is the first inbox, start the poll loop
   684→    if (!this.pollTimer) {
   685→      this.startPolling();
   686→    }
   687→    return inbox;
   688→  }

   727→  /**
   728→   * Validate a key by making a test API call, then inject into process.env.
   729→   * Returns true if valid, false if rejected.
   730→   */
   731→  async validateAndInject(pending: PendingKey): Promise<boolean> {
   732→    try {
   733→      // Quick format check first (cheaper than API call)
   734→      const svc = SERVICES_MAP[pending.service];
   735→      if (svc?.pattern && !svc.pattern.test(pending.key)) {
   736→        pending.status = 'rejected';
   737→        pending.validationError = 'Format mismatch';
   738→        pending.validatedAt = Date.now();
   739→        audit('key.rejected', `${pending.service} key failed format check`, { pendingId: pending.id });
   740→        return false;
   741→      }
   742→
   743→      // For Gemini keys, make a test call
   744→      if (pending.service === 'gemini' || pending.envVar === 'GEMINI_API_KEY') {
   745→        try {
   746→          const { GoogleGenerativeAI } = await import('@google/generative-ai');
   747→          const genAI = new GoogleGenerativeAI(pending.key);
   748→          const model = genAI.getGenerativeModel({ model: 'gemini-2.0-flash' });
   749→          const result = await model.generateContent({
   750→            contents: [{ role: 'user', parts: [{ text: 'Say OK' }] }],
   751→          });
   752→          const text = result?.response?.text();
   753→          if (!text || text.length === 0) throw new Error('Empty response');
   754→        } catch (err) {
   755→          pending.status = 'rejected';
   756→          pending.validationError = (err as Error).message?.slice(0, 200);
   757→          pending.validatedAt = Date.now();
   758→          audit('key.rejected', `${pending.service} key failed validation: ${(err as Error).message?.slice(0, 100)}`, {
   759→            pendingId: pending.id,
   760→          });
   761→          return false;
   762→        }
   763→      }
   764→
   765→      // Passed validation — inject
   766→      pending.status = 'approved';
   767→      pending.validatedAt = Date.now();
   768→      process.env[pending.envVar] = pending.key;
   769→      audit('key.approved', `${pending.service} key validated and injected`, { pendingId: pending.id });
   770→      return true;
   771→    } catch (err) {
   772→      pending.status = 'rejected';
   773→      pending.validationError = (err as Error).message;
   774→      pending.validatedAt = Date.now();
   775→      audit('key.rejected', `${pending.service} key validation error: ${(err as Error).message?.slice(0, 100)}`, {
   776→        pendingId: pending.id,
   777→      });
   778→      return false;
   779→    }
   780→  }

   783→  async approvePendingKey(pendingId: string): Promise<boolean> {
   784→    const pending = this.pendingKeys.find(k => k.id === pendingId && k.status === 'pending');
   785→    if (!pending) return false;

   789→  /** Manually reject a pending key */
   790→  rejectPendingKey(pendingId: string): boolean {

   792→    if (!pending) return false;
   793→    pending.status = 'rejected';
   794→    pending.validatedAt = Date.now();
   795→    pending.validationError = 'Manual rejection';
   796→    audit('key.rejected', `${pending.service} key manually rejected`, { pendingId });
   797→    return true;
   798→  }
