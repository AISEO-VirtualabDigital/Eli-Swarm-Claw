   162→// ─── Helper: Count-up animation hook (fixed count) ───────────
   163→function useMetricCounts(targets: number[], duration = 800) {
   164→  const [counts, setCounts] = useState(targets.map(() => 0));
   165→  const [started, setStarted] = useState(false);
   166→
   167→  useEffect(() => {
   168→    if (!started) return;
   169→    const startTime = performance.now();
   170→  let rafId: number;
   171→    const step = (now: number) => {
   172→      const elapsed = now - startTime;
   173→      const progress = Math.min(elapsed / duration, 1);
   174→      const eased = 1 - Math.pow(1 - progress, 3);
   175→      setCounts(targets.map((t) => Math.round(eased * t)));
   176→      if (progress < 1) rafId = requestAnimationFrame(step);
   177→    };
   178→    rafId = requestAnimationFrame(step);
   179→    return () => cancelAnimationFrame(rafId);
   180→  }, [started, duration, ...targets]);
   181→
   182→  return { counts, start: () => setStarted(true) };
   183→}

   185→// ─── Helper: Live time ──────────────────────────────────────────
   186→function useLiveTime() {
   187→  const [time, setTime] = useState('');
   188→  useEffect(() => {
   189→    const update = () => {
   190→      const now = new Date();
   191→      setTime(
   192→        now.toLocaleTimeString('en-US', {
   193→          hour: '2-digit',
   194→          minute: '2-digit',
   195→          second: '2-digit',
   196→          hour12: false,
   197→        })
   198→      );
   199→    };
   200→    update();
   201→    const interval = setInterval(update, 1000);
   202→    return () => clearInterval(interval);
   203→  }, []);
   204→  return time;
   205→}

   587→function ChatView() {
   588→  const [messages, setMessages] = useState<ChatMessage[]>([]);
   589→  const [input, setInput] = useState('');

   591→  const messagesEndRef = useRef<HTMLDivElement>(null);
   592→  const textareaRef = useRef<HTMLTextAreaElement>(null);
   593→
   594→  const scrollToBottom = useCallback(() => {
   595→    messagesEndRef.current?.scrollIntoView({ behavior: 'smooth' });
   596→  }, []);
   597→
   598→  useEffect(() => {
   599→    scrollToBottom();
   600→  }, [messages, isLoading, scrollToBottom]);
   601→
   602→  // Welcome message on mount
   603→  useEffect(() => {
   604→    setMessages([
   605→      {
   606→        id: 'welcome',
   607→        role: 'eli',
   608→        content: ELI_WELCOME,
   609→        timestamp: new Date(),
   610→      },
   611→    ]);
   612→  }, []);
   613→
   614→  const handleSend = async () => {
   615→    const trimmed = input.trim();
   616→    if (!trimmed || isLoading) return;
   617→
   618→    const userMsg: ChatMessage = {
   619→      id: `user-${Date.now()}`,
   620→      role: 'user',
   621→      content: trimmed,
   622→      timestamp: new Date(),
   623→    };
   624→
   625→    setMessages((prev) => [...prev, userMsg]);
   626→    setInput('');
   627→    setIsLoading(true);
   628→
   629→    // Reset textarea height
   630→    if (textareaRef.current) {
   631→      textareaRef.current.style.height = 'auto';
   632→    }
   633→
   634→    try {
   635→      const history = messages
   636→        .filter((m) => m.id !== 'welcome')
   637→        .map((m) => ({ role: m.role, content: m.content }));
   638→
   639→      const res = await fetch('/api/eli-chat', {
   640→        method: 'POST',
   641→        headers: { 'Content-Type': 'application/json' },
   642→        body: JSON.stringify({ message: trimmed, history }),
   643→      });
   644→
   645→      const data = await res.json();
   646→
   647→      const eliMsg: ChatMessage = {
   648→        id: `eli-${Date.now()}`,
   649→        role: 'eli',
   650→        content: data.response || 'I encountered an issue processing your request. Please try again.',
   651→        sources: data.sources,
   652→        timestamp: new Date(),
   653→      };
   654→
   655→      setMessages((prev) => [...prev, eliMsg]);
   656→    } catch {
   657→      const errorMsg: ChatMessage = {
   658→        id: `error-${Date.now()}`,
   659→        role: 'eli',
   660→        content: 'Sorry, I encountered a connection error. Please check that the server is running and try again.',
   661→        timestamp: new Date(),

   700→  return (
