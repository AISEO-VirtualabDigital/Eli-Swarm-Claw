    96→def add_heading(text, style, level=0):
    97→    key = f'h_{hashlib.md5(text.encode()).hexdigest()[:8]}'
    98→    p = Paragraph(f'<a name="{key}"/>{text}', style)
    99→    p.bookmark_name = key
   100→    p.bookmark_level = level
   101→    p.bookmark_text = text
   102→    p.bookmark_key = key
   103→    return p

   105→def safe_keep(elements):
   106→    total = 0
   107→    for el in elements:
   108→        w, h = el.wrap(W - 2*MARGIN, H)
   109→        total += h
   110→    if total <= A4[1] * 0.4:
   111→        return [KeepTogether(elements)]
   112→    elif len(elements) >= 2:
   113→        return [KeepTogether(elements[:2])] + list(elements[2:])
   114→    return list(elements)

   116→def make_table(headers, rows, col_widths=None):
   117→    available = W - 2*MARGIN
   118→    if not col_widths:
   119→        n = len(headers)
   120→        col_widths = [available / n] * n
   121→    data = [[Paragraph(h, ParagraphStyle('TH', fontName='FreeSerif-Bold', fontSize=9.5, leading=14, textColor=colors.white)) for h in headers]]
   122→    for row in rows:
   123→        data.append([Paragraph(str(c), ParagraphStyle('TD', fontName='FreeSerif', fontSize=9, leading=13, textColor=TEXT_PRIMARY)) for c in row])
   124→    t = Table(data, colWidths=col_widths, repeatRows=1)
   125→    style_cmds = [
   126→        ('BACKGROUND', (0, 0), (-1, 0), HEADER_FILL),
   127→        ('TEXTCOLOR', (0, 0), (-1, 0), colors.white),
   128→        ('VALIGN', (0, 0), (-1, -1), 'TOP'),
   129→        ('TOPPADDING', (0, 0), (-1, -1), 6),
   130→        ('BOTTOMPADDING', (0, 0), (-1, -1), 6),
   131→        ('LEFTPADDING', (0, 0), (-1, -1), 8),
   132→        ('RIGHTPADDING', (0, 0), (-1, -1), 8),
   133→        ('GRID', (0, 0), (-1, -1), 0.5, BORDER),
   134→    ]
   135→    for i in range(1, len(data)):
   136→        if i % 2 == 0:
   137→            style_cmds.append(('BACKGROUND', (0, i), (-1, i), TABLE_STRIPE))
   138→    t.setStyle(TableStyle(style_cmds))
   139→    return t

   268→story.append(Paragraph(

The three tiers are: <b>Green (autonomous pass)</b>, <b>Amber (logged and sampled)</b>, and <b>Red (mandatory human approval)</b>. Each tier has distinct processing rules, latency characteristics, and audit requirements.', STY['body']))
   270→
   271→story.append(Paragraph(
   272→    'The <b>Green tier</b> handles read-only operations within an agent\'s declared domain. When the Keyword Agent queries the keyword_clusters table to retrieve previously computed clusters, the kernel verifies that keyword_clusters is listed in the Keyword Agent\'s SKILL.md IPC Policy under allowed read tables, confirms the operation is read-only, and passes the request through without further evaluation. Green tier operations complete in under one millisecond of added latency, which is negligible compared to the PostgreSQL query time. The kernel logs the operation for audit purposes but does not generate an alert. Green tier operations account for an estimated 80-85% of all IPC traffic in a typical SEO workflow.', STY['body']))
   273→
   274→story.append(Paragraph(
   275→    'The <b>Amber tier</b> handles write operations and cross-agent data access. When the Technical SEO Agent writes a new audit result to the tech_seo_audits table, the kernel verifies the write permission, checks resource limits (is the agent within its memory and CPU allocation?), validates the data schema against the SKILL.md output schema, and logs the operation with full context. The kernel also applies statistical sampling: a configurable percentage (default 10%) of Amber operations are flagged for asynchronous review by the QA Agent. This provides continuous quality assurance without blocking the workflow. Amber tier operations add approximately 5-10 milliseconds of latency, which is acceptable for write operations that already incur PostgreSQL write latency.', STY['body']))
   276→
   277→story.append(Paragraph(
   278→    'The <b>Red tier</b> handles operations that are irreversible, high-impact, or cross-domain. Examples include: deleting data from shared tables, modifying another agent\'s records, executing external API calls that have cost implications (such as paid SERP API queries), or any operation that exceeds the agent\'s declared resource limits.    309→story.append(add_heading('<b>4.1 Transport Layer: gRPC over Unix Domain Sockets</b>', STY['h2'], 1))
   310→story.append(Paragraph(
   311→    'The current Eli-OS prototype uses a basic HTTP-based IPC mechanism between the Rust kernel and the Python Eli Claw application.    312→
   313→story.append(Paragraph(
   314→    'The gRPC service definition (in protobuf format) specifies four core RPC methods. The <b>ReportResult</b> method allows agents to report operation results back to the kernel for audit logging and event broadcasting. The <b>Escalate</b> method allows agents to raise escalation events when they encounter conditions defined in their SKILL.md Escalation Triggers.    315→
   316→story.append(Paragraph(
   317→    'The choice of gRPC over alternative IPC mechanisms was driven by several factors specific to the Eli-Swarm-Claw architecture. Second, gRPC\'s built-in deadline and cancellation semantics allow the kernel to enforce the max_duration_seconds resource limit from the SKILL.md IPC Policy. When an agent\'s operation exceeds its time allocation, the kernel cancels the gRPC stream and returns a ResourceExceeded response.
