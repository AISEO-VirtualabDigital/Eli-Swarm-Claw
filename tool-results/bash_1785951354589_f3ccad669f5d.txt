   254	  // --- Settings ----------------------------------------------------------
   255	  async loadSettings(): Promise<void> {
   256	    this.settings = Object.assign(
   257	      {},
   258	      DEFAULT_SETTINGS,
   259	      (await this.loadData()) as Partial<SkillLayerSettings> | null,
   260	    );
   261	    // Migration: the harness selector (the M1 global `--harness` plus the M4–M7
   262	    // per-skill harness machinery) is replaced by the per-skill AGENT selector;
   263	    // omnigent now picks the harness itself. Strip every legacy harness key
   264	    // fail-closed (a plain `delete` cannot throw) so a stale data.json can never
   265	    // reintroduce harness behavior. A skill that had a harness selected simply
   266	    // reverts to the Default agent (it has no `skillAgent` entry). Everything
   267	    // else in data.json is preserved by the Object.assign above.
   268	    const raw = this.settings as unknown as Record<string, unknown>;
   269	    for (const key of [
   270	      // NOTE (M15): `skillHarness` is NO LONGER stripped — it is repurposed as
   271	      // the per-skill omnigent `--harness` map (string values). Any stale value
   272	      // from the removed M4–M7 harness selector (a different shape) is made safe
   273	      // by `resolveHarness`, which fails closed against the hardcoded harness
   274	      // allowlist, so a non-member simply emits no `--harness`.
   275	      "discoveredHarnesses",
   276	      "customHarnesses",
   277	      "omnigentHarness",
   278	      // M11: the user-configurable invocation template is gone — the "Copy
   279	      // invocation" action now uses a FIXED natural-language form
   280	      // (`Use the <name> skill.`). Strip it fail-closed so a stale data.json can
   281	      // never reintroduce it; every OTHER setting (scanRoots, pins, skillAgent,
   282	      // omnigentBinaryPath, omnigentServerUrl, appendVaultAnchor, …) is preserved
   283	      // by the Object.assign above. NOTE (M19): `omnigentServerUrl` is BACK as a
   284	      // real setting (per-launch `--server` target), so it is NOT stripped.
   285	      "invocationTemplate",
   286	    ]) {
   287	      delete raw[key];
   288	    }
   289	
   290	    // M18: remove the home-directory (global) skill roots an earlier build
   291	    // auto-added (external kind, disabled) — they cluttered settings and mixed
   292	    // machine-global skills with the user's in-vault skills. Only the exact
   293	    // auto-added paths are dropped, so a user's own custom external roots stay.
   294	    const homeRoots = new Set(homeSkillRootPaths(os.homedir()));
   295	    this.settings.scanRoots = this.settings.scanRoots.filter(
   296	      (r) => !(r.kind === "external" && homeRoots.has(r.path)),
   297	    );
   298	    // Union the per-tool VAULT skill scan roots (from the Agentfiles tool map)
   299	    // into whatever's stored, so in-vault skills across Claude/Cursor/Codex/etc.
   300	    // are discovered out of the box. Additive + idempotent: existing roots (incl.
   301	    // the user's custom ones and their enabled state) are preserved.
   302	    const havePaths = new Set(this.settings.scanRoots.map((r) => r.path));
   303	    for (const root of defaultSkillScanRoots()) {
   304	      if (!havePaths.has(root.path)) {
   305	        this.settings.scanRoots.push(root);
   306	        havePaths.add(root.path);
   307	      }
   308	    }
   309	    // Migration: the earlier default panel width (420) was too narrow for all six
   310	    // tabs; bump a still-default value up to the new 520. A width the user chose
   311	    // themselves (anything other than the old default) is left untouched.
   312	    if (this.settings.panelWidth === 420) this.settings.panelWidth = 520;
   313	  }

   319	  /**
   320	   * Seed the bundled example skill on first run. No-op when already seeded

   323	   * clobbered and a deleted one isn't recreated). Writes via Node `fs` because

   326	  private async seedExampleSkill(): Promise<void> {
   327	    if (this.settings.seededExample) return;

   330	    try {

   332	      if (!fs.existsSync(abs)) {
   333	        await fs.promises.mkdir(nodePath.dirname(abs), { recursive: true });
   334	        await fs.promises.writeFile(abs, EXAMPLE_SKILL_BODY, "utf8");
   335	      }
   336	      // Mark seeded even if the file already existed, so we stop probing.
   337	      this.settings.seededExample = true;
   338	      await this.saveSettings();
   339	      await this.rescan();
   340	    } catch (err) {
   341	      console.error("[skill-layer] example skill seed failed:", err);

   372	  /**
   373	   * Create a tool folder (e.g. `.claude/skills`) in the vault if absent, then
   374	   * rescan so its (empty) section appears. Returns a user-facing error string, or
   375	   * null on success. Desktop-gated; `seg` must be one of the known segments.
   376	   */
   377	  async createToolFolder(seg: string): Promise<string | null> {
   378	    const base = this.detector.vaultBasePath();
   379	    if (!base || !this.detector.canScanExternal()) return "Requires the desktop app.";
   380	    if (!this.addableFolderSegments("skill").includes(seg) &&
   381	        !this.addableFolderSegments("command").includes(seg)) {
   382	      return "Unknown folder.";
   383	    }
   384	    try {
   385	      await fs.promises.mkdir(nodePath.join(base, seg), { recursive: true });
   386	      await this.rescan();
   387	      return null;
   388	    } catch (err) {
   389	      console.error("[skill-layer] createToolFolder failed:", err);
   390	      return "Could not create the folder.";
   391	    }
   392	  }

   394	  /**
   395	   * Create a new SKILL inside the tool folder `folderSeg`: `<folderSeg>/<name>/
   396	   * SKILL.md` with a minimal frontmatter stub, then open it in Obsidian (like the
   397	   * Open file button). Returns an error string, or null on success. The name is
   398	   * sanitized to a safe folder name; refuses to overwrite an existing file.
   399	   */
   400	  async createSkillInFolder(folderSeg: string, rawName: string): Promise<string | null> {
   401	    const base = this.detector.vaultBasePath();
   402	    if (!base || !this.detector.canScanExternal()) return "Requires the desktop app.";
   403	    const name = this.sanitizeItemName(rawName);
   404	    if (!name) return "Please enter a valid name.";
   405	    const relPath = `${folderSeg}/${name}/SKILL.md`;
   406	    const abs = nodePath.join(base, relPath);
   407	    if (fs.existsSync(abs)) return `A skill named "${name}" already exists here.`;
   408	    const body = `---\nname: ${name}\ndescription: \n---\n\n# ${name}\n\n`;
   409	    try {
   410	      await fs.promises.mkdir(nodePath.dirname(abs), { recursive: true });
   411	      await fs.promises.writeFile(abs, body, "utf8");
   412	    } catch (err) {
   413	      console.error("[skill-layer] createSkillInFolder failed:", err);
   414	      return "Could not create the skill file.";
   415	    }
   416	    await this.rescan();
   417	    await this.openCreatedFile(relPath, abs);
   418	    return null;
   419	  }

   447	  /** A safe folder/file base name from user input (kebab-ish, no path separators). */
   448	  private sanitizeItemName(raw: string): string {
   449	    return (raw ?? "")
   450	      .trim()
   451	      .replace(/[^A-Za-z0-9._ -]+/g, "")
   452	      .replace(/\s+/g, "-")
   453	      .replace(/^[-.]+|[-.]+$/g, "")
   454	      .slice(0, 60);
   455	  }

   457	  /**
   458	   * Open a just-created vault file in Obsidian. Tool folders are hidden
   459	   * dot-folders, so reuse the same temporary-reveal path as `openSkill`
   460	   * (`openHiddenSkillTemporarily`); fall back to the OS app if it never indexes.
   461	   */
   462	  private async openCreatedFile(vaultPath: string, absPath: string): Promise<void> {
   463	    const tfile = this.detector.resolveTFile(vaultPath);
   464	    if (tfile instanceof TFile) {
   465	      await this.openTFileAndRevealFolder(tfile);
   466	      return;
   467	    }
   468	    if (this.canRevealHiddenFolders() && !this.settings.showHiddenFolders) {
   469	      if (await this.openHiddenSkillTemporarily(vaultPath)) return;
   470	    }
   471	    try {
   472	      const result = await shell.openPath(absPath);
   473	      if (result) new Notice(`Could not open file: ${result}`);
   474	    } catch (err) {
   475	      console.error("[skill-layer] openCreatedFile failed:", err);
   476	    }
   477	  }

   606	  // --- Browser view ------------------------------------------------------
   607	  async activateView(): Promise<void> {
   608	    const { workspace } = this.app;
   609	    const existing = workspace.getLeavesOfType(SKILL_LAYER_VIEW);
   610	    if (existing.length > 0) {
   611	      await workspace.revealLeaf(existing[0]);
   612	      // Revealing an existing leaf does NOT re-fire onOpen, so refresh from
   613	      // disk here to pick up external edits (incl. dot-folder / external roots).
   614	      await this.rescan();
   615	      this.applyPanelWidth();
   616	      return;
   617	    }
   618	    const leaf = workspace.getRightLeaf(false);
   619	    if (!leaf) return;
   620	    // A freshly created leaf's onOpen() runs the rescan itself.
   621	    await leaf.setViewState({ type: SKILL_LAYER_VIEW, active: true });
   622	    await workspace.revealLeaf(leaf);
   623	    this.applyPanelWidth();
   624	  }

   626	  /**
   627	   * Open the browser side panel at the configured `panelWidth` so it always uses
   628	   * a consistent width rather than whatever the sidebar was last dragged to. The
   629	   * right-sidebar width has no public API, so this sets the width on the right
   630	   * split's container element (undocumented internals) behind a guarded cast and
   631	   * a `requestAnimationFrame` (so it lands after the reveal lays out). Best-effort
   632	   * — silently no-ops if the internals aren't shaped as expected or on mobile.
   633	   */
   634	  private applyPanelWidth(): void {
   635	    const width = this.settings.panelWidth;
   636	    if (!width || width <= 0) return;
   637	    const rightSplit = (this.app.workspace as unknown as {
   638	      rightSplit?: { containerEl?: HTMLElement; collapsed?: boolean };
   639	    }).rightSplit;
   640	    const el = rightSplit?.containerEl;
   641	    if (!el) return;
   642	    window.requestAnimationFrame(() => {
   643	      try {
   644	        // setCssStyles (not a direct .style assignment) per Obsidian guidelines.
   645	        el.setCssStyles({ width: `${width}px` });
   646	        // Nudge Obsidian to reconcile the split sizes to the new width.
   647	        this.app.workspace.requestSaveLayout?.();
   648	      } catch (err) {
   649	        console.error("[skill-layer] applyPanelWidth failed:", err);
   650	      }
   651	    });
   652	  }

   654	  /**
   655	   * Ribbon behavior: toggle the Skill and Harness Manager pane.
   656	   * - Not open                → open + reveal (via activateView).
   657	   * - Open but not active      → reveal/focus it (via activateView).
   658	   * - Open AND active/visible  → close (detach its leaves).
   659	   *
   660	   * The command-palette "Open skills browser" entry stays on activateView
   661	   * (pure open+reveal, never closes) — closing on a command named "Open"
   662	   * would be surprising. Only the ribbon toggles.
   663	   */
   664	  async toggleView(): Promise<void> {
   665	    const { workspace } = this.app;
   666	    const existing = workspace.getLeavesOfType(SKILL_LAYER_VIEW);
   667	    const activeView = workspace.getActiveViewOfType(SkillBrowserView);
   668	    const isActiveVisible =
   669	      activeView !== null && existing.includes(activeView.leaf);
   670	    const action = decideToggleAction(existing.length > 0, isActiveVisible);
   671	    if (action === "close") {
   672	      workspace.detachLeavesOfType(SKILL_LAYER_VIEW);
   673	      return;
   674	    }
   675	    // "open" and "reveal" both funnel through activateView (open+reveal+rescan).
   676	    await this.activateView();
   677	  }

   781	  /**
   782	   * Scan `<vault>/.claude/agents` and `~/.claude/agents` for `*.md` subagents
   783	   * into the cache (no view refresh). Name = frontmatter `name:` (else filename
   784	   * stem); project entries win over global on a name clash. Missing dirs yield
   785	   * zero (no error); all fs access is wrapped so this never throws.
   786	   */
   787	  private scanClaudeAgents(): void {
   788	    const base = this.detector.vaultBasePath();
   789	    const home = os.homedir();
   790	    // M18: scan EVERY tool's agents folder (.claude/agents, .cursor/agents,
   791	    // .codex/agents, …) — project (vault-relative) first, then home — so agents
   792	    // from any assistant appear. Project wins over global on a name clash.
   793	    const dirs: { dir: string; source: "project" | "global" }[] = [];
   794	    for (const seg of agentFolderSegments()) {
   795	      if (base) dirs.push({ dir: nodePath.join(base, seg), source: "project" });
   796	    }
   797	    for (const seg of agentFolderSegments()) {
   798	      dirs.push({ dir: joinHome(home, seg), source: "global" });
   799	    }
   800	
   801	    const byName = new Map<string, ClaudeAgent>();
   802	    for (const { dir, source } of dirs) {
   803	      let entries: string[];
   804	      try {
   805	        entries = fs.readdirSync(dir);
   806	      } catch {
   807	        continue; // missing dir → skip
   808	      }
   809	      for (const entry of entries.sort()) {
   810	        if (!/\.md$/i.test(entry)) continue;
   811	        const abs = nodePath.join(dir, entry);
   812	        let text: string;
   813	        try {
   814	          if (!fs.statSync(abs).isFile()) continue;
   815	          text = fs.readFileSync(abs, "utf8");
   816	        } catch {
   817	          continue;
   818	        }
   819	        const meta = parseClaudeAgentFrontmatter(text);
   820	        const name = meta.name && meta.name.trim() ? meta.name.trim() : entry.replace(/\.md$/i, "");
   821	        // Project (scanned first) wins; don't let global clobber it.
   822	        if (byName.has(name)) continue;
   823	        byName.set(name, {
   824	          name,
   825	          path: abs,
   826	          source,
   827	          ...(meta.description ? { description: meta.description } : {}),
   828	        });
   829	      }
   830	    }
   831	    this.claudeAgents = Array.from(byName.values()).sort((a, b) =>
   832	      a.name.localeCompare(b.name),
   833	    );
   834	  }

   876	  /**
   877	   * Scan every tool's commands folder (`.claude/commands`, `.codex/prompts`, …)
   878	   * — project (vault-relative) first, then home — for `*.md` files, each of
   879	   * which is one command. Unlike skills there's no SKILL.md rule: any markdown in
   880	   * a commands folder is a command. Built as Skill objects with kind:"command" so
   881	   * they reuse the whole row UI / per-item state. Deduped by absolute path
   882	   * (project wins). All fs access wrapped; never throws.
   883	   */
   884	  private scanCommands(): void {
   885	    const base = this.detector.vaultBasePath();
   886	    const home = os.homedir();
   887	    const dirs: { dir: string; vaultRel: string | null }[] = [];
   888	    for (const seg of commandFolderSegments()) {
   889	      if (base) dirs.push({ dir: nodePath.join(base, seg), vaultRel: seg });
   890	    }
   891	    for (const seg of commandFolderSegments()) {
   892	      dirs.push({ dir: joinHome(home, seg), vaultRel: null });
   893	    }
   894	
   895	    const byId = new Map<string, Skill>();
   896	    for (const { dir, vaultRel } of dirs) {
   897	      let entries: string[];
   898	      try {
   899	        entries = fs.readdirSync(dir);
   900	      } catch {
   901	        continue; // missing dir → skip
   902	      }
   903	      for (const entry of entries.sort()) {
   904	        if (!/\.md$/i.test(entry)) continue;
   905	        const abs = nodePath.join(dir, entry);
   906	        let text: string;
   907	        try {
   908	          if (!fs.statSync(abs).isFile()) continue;
   909	          text = fs.readFileSync(abs, "utf8");
   910	        } catch {
   911	          continue;
   912	        }
   913	        if (byId.has(abs)) continue;
   914	        const fm = parseFrontmatter(text);
   915	        const stem = entry.replace(/\.md$/i, "");
   916	        const name = fm.name && fm.name.trim() ? fm.name.trim() : stem;
   917	        const description =
   918	          fm.description && fm.description.trim()
   919	            ? fm.description.trim()
   920	            : firstHeading(text) ?? "(no description)";
   921	        const vaultPath = vaultRel ? `${vaultRel}/${entry}` : null;
   922	        const relForTag = vaultPath ?? `${nodePath.basename(dir)}/${entry}`;
   923	        byId.set(abs, {
   924	          id: abs,
   925	          kind: "command",
   926	          name,
   927	          description,
   928	          path: abs,
   929	          vaultPath,
   930	          sourceRoot: dir,
   931	          sourceLabel: inferSourceLabel(abs),
   932	          detection: vaultRel ? "adapter" : "external",
   933	          tags: resolveSkillTags({
   934	            relativePath: relForTag,
   935	            description,
   936	            frontmatterTags: fm.tags ?? [],
   937	          }),
   938	        });
   939	      }
   940	    }
   941	    this.commands = Array.from(byId.values()).sort((a, b) =>
   942	      a.name.localeCompare(b.name),
   943	    );
   944	  }

   946	  /**
   947	   * The <select> option value reflecting a skill's stored agent choice, used to
   948	   * preselect the dropdown. Falls back to Default when the stored value is not a
   949	   * currently-selectable option (unknown built-in, or a custom path no longer in
   950	   * the discovered set) so the UI never shows an orphaned selection — launch
   951	   * already fails closed independently via `resolveAgentLaunch`.
   952	   */
   953	  agentOptionValue(id: string): string {
   954	    const stored = this.settings.skillAgent[id];
   955	    if (!stored || typeof stored !== "object") return encodeAgentChoice(null);
   956	    if (stored.kind === "builtin") {
   957	      return isAllowedBuiltinAgent(stored.name)
   958	        ? encodeAgentChoice(stored)
   959	        : encodeAgentChoice(null);
   960	    }
   961	    if (stored.kind === "custom") {
   962	      return this.customAgents.some((a) => a.path === stored.path)
   963	        ? encodeAgentChoice(stored)
   964	        : encodeAgentChoice(null);
   965	    }
   966	    return encodeAgentChoice(null);
   967	  }

   969	  /**
   970	   * Persist a skill's agent choice from the dropdown's encoded option value. The
   971	   * value is decoded then VALIDATED before storing: a built-in name must be in
   972	   * the hardcoded allowlist; a custom path must pass `isValidCustomAgentPath`
   973	   * against the scan dir AND be one of the currently-discovered agents. Anything
   974	   * else (incl. Default) deletes the key so data.json stays clean. Launch
   975	   * re-validates independently, so storage is defense-in-depth, not the gate.
   976	   */
   977	  async setSkillAgent(id: string, encoded: string): Promise<void> {
   978	    const choice = decodeAgentChoice(encoded);
   979	    let store: SkillAgent | null = null;
   980	    if (choice.kind === "builtin" && isAllowedBuiltinAgent(choice.name)) {
   981	      store = { kind: "builtin", name: choice.name };
   982	    } else if (choice.kind === "custom") {
   983	      const dir = this.agentConfigDir();
   984	      const known = this.customAgents.some((a) => a.path === choice.path);
   985	      if (dir && known && isValidCustomAgentPath(choice.path, dir)) {
   986	        store = { kind: "custom", path: choice.path };
   987	      }
   988	    }
   989	    if (store) {
   990	      this.settings.skillAgent[id] = store;
   991	    } else {
   992	      delete this.settings.skillAgent[id];
   993	    }
   994	    await this.saveSettings();
   995	  }

   997	  /**
   998	   * The <select> option value reflecting a skill's stored HARNESS choice (M15),
   999	   * for preselecting the dropdown. Resolves to a built-in omnigent harness name,
  1000	   * a `custom:<id>` value (only if that custom harness still exists), or the
  1001	   * Default sentinel — so the UI never shows an orphaned selection. Launch
  1002	   * re-validates via `resolveSkillHarness`.
  1003	   */
  1004	  harnessOptionValue(id: string): string {
  1005	    const choice = parseHarnessValue(this.settings.skillHarness[id]);
  1006	    if (choice.kind === "omnigent") return choice.name;
  1007	    if (
  1008	      choice.kind === "custom" &&
  1009	      this.settings.harnesses.some((h) => h.id === choice.id)
  1010	    ) {
  1011	      return encodeCustomHarnessChoice(choice.id);
  1012	    }
  1013	    return HARNESS_DEFAULT_VALUE;
  1014	  }
  1015	
  1016	  /**
  1017	   * Human-readable label for a skill's effective AGENT (M16) — shown on the row
  1018	   * so the assignment is visible without opening Configuration. Mirrors
  1019	   * `agentOptionValue`'s fail-closed fallback: a stale/unknown choice reads as
  1020	   * "Default". Returns "Default", a built-in name (polly/debby), or a custom
  1021	   * agent's display name.
  1022	   */
  1023	  agentLabelFor(id: string): string {
  1024	    const choice = decodeAgentChoice(this.agentOptionValue(id));

  1259	  /**
  1260	   * Run a bash script by id, per its own launch mode:
  1261	   *   - terminal → open the user's default terminal running the body (visible).
  1262	   *   - headless → spawn `bash -c <body>` (or `cmd /c` on Windows) detached via
  1263	   *     the shared hardened spawn surface, Notices only.
  1264	   * The body is user-authored and runs only on this explicit click (same trust
  1265	   * model as custom harnesses). cwd = vault. Desktop-gated.
  1266	   */
  1267	  runBashScript(id: string): void {
  1268	    if (!this.detector.canScanExternal()) {
  1269	      new Notice("Skill and Harness Manager: running scripts requires the desktop app.");
  1270	      return;
  1271	    }
  1272	    const cwd = this.detector.vaultBasePath();
  1273	    if (!cwd) {
  1274	      new Notice("Skill and Harness Manager: could not resolve the vault path; not running.");
  1275	      return;
  1276	    }
  1277	    const script = this.settings.bashScripts.find((s) => s.id === id);
  1278	    if (!script) return;
  1279	
  1280	    if (script.launchMode === "terminal") {
  1281	      this.runRawInTerminal(script.body, cwd, `script-${script.id}`);
  1282	      new Notice(`Running "${script.label}" in your terminal…`);
  1283	      return;
  1284	    }
  1285	    // Headless: bash -c <body> (cmd /c on Windows). The body is a single inert
  1286	    // argv element (shell:false), so it is never re-tokenized by a shell WE spawn
  1287	    // — `bash -c` interprets it, which is the intended behavior for a script.
  1288	    const argv =
  1289	      process.platform === "win32"
  1290	        ? ["cmd.exe", "/c", script.body]
  1291	        : ["/bin/bash", "-c", script.body];
  1292	    this.spawnOmnigent(
  1293	      argv,
  1294	      cwd,
  1295	      `Running "${script.label}" in the background — check for output via Notices.`,
  1296	      script.label,
  1297	    );
  1298	  }

  1333	  /**
  1334	   * Discover the harnesses omnigent has configured by running
  1335	   * `omnigent config list` and parsing its "Credentials (by harness)" section.
  1336	   * Best-effort + fail-quiet: desktop-gated, the omnigent binary resolved
  1337	   * silently (no Notice), a bounded stdout capture with a hard timeout, and any
  1338	   * failure leaves the cache empty (the dropdown then falls back to the full
  1339	   * allowlist). This is a READ-ONLY omnigent query (config list), not a launch —
  1340	   * distinct from the hardened launch spawn, and it never passes user input.
  1341	   */
  1342	  private async discoverConfiguredHarnesses(): Promise<void> {
  1343	    const finish = (list: ConfiguredHarness[]) => {
  1344	      this.configuredHarnesses = list;
  1345	      this.harnessesDiscovered = true;
  1346	      this.refreshViews();
  1347	    };
  1348	    if (!this.detector.canScanExternal()) return finish([]);
  1349	    const resolution = resolveOmnigentBinary({
  1350	      override: this.settings.omnigentBinaryPath,
  1351	      homedir: os.homedir(),
  1352	      exists: (p) => fs.existsSync(p),
  1353	    });
  1354	    if (resolution.status !== "ok") return finish([]);
  1355	
  1356	    const env = this.launchEnv();
  1357	    let child;
  1358	    try {
  1359	      // Fixed args only; no user input. stdout piped for parsing; stderr/stdin ignored.
  1360	      child = spawn(resolution.path, ["config", "list"], {
  1361	        env,
  1362	        shell: false,
  1363	        windowsHide: true,
  1364	        stdio: ["ignore", "pipe", "ignore"],
  1365	      });
  1366	    } catch (err) {
  1367	      console.error("[skill-layer] harness discovery spawn threw:", err);
  1368	      return finish([]);
  1369	    }
  1370	    let stdout = "";
  1371	    let done = false;
  1372	    const settle = (list: ConfiguredHarness[]) => {
  1373	      if (done) return;
  1374	      done = true;
  1375	      finish(list);
  1376	    };
  1377	    // Hard timeout so a hung `omnigent` never leaves discovery pending.
  1378	    const timer = window.setTimeout(() => {
  1379	      try {
  1380	        child.kill();
  1381	      } catch {
  1382	        /* ignore */
  1383	      }
  1384	      settle([]);
  1385	    }, 8000);
  1386	    child.stdout?.on("data", (chunk: Buffer) => {
  1387	      // Bound the captured output defensively.
  1388	      if (stdout.length < 64_000) stdout += chunk.toString();
  1389	    });
  1390	    child.on("error", (err) => {
  1391	      console.error("[skill-layer] harness discovery error:", err);
  1392	      window.clearTimeout(timer);
  1393	      settle([]);
  1394	    });
  1395	    child.on("close", (code) => {
  1396	      window.clearTimeout(timer);
  1397	      settle(code === 0 ? parseConfiguredHarnesses(stdout) : []);
  1398	    });
  1399	    child.unref?.();
  1400	  }
