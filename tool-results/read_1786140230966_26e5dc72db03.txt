     1→     1→/**
     2→     2→ * Open Claw Engine v2 — Infinite Email Generator + Autonomous Reader
     3→     3→ * 
     4→     4→ * "Open Claw" = self-sufficient, zero-cost, multi-provider email system.
     5→     5→ * 
     6→     6→ * Wired patterns:
     7→     7→ *   OmniKey — penalty tracker with decay, round-robin, skip-set on failure
     8→     8→ *   OmniMail — provider resolution chain, retryable error classification
     9→     9→ *   Agent-Reach — probe-don't-guess health checks
    10→    10→ *   OmniRoute (diegosouzapw) — provider adapter abstraction
    11→    11→ *   browser-use — decoupled browser task generation
    12→    12→ * 
    13→    13→ * Providers:
    14→    14→ *   1. Guerrilla Mail (primary) — session-based, instant, 1hr TTL, no registration

    16→    16→ *   3. OpenInbox (tertiary) — creation-only (free), can't read without paid key
    17→    17→ * 
    18→    18→ * Provider resolution chain (OmniMail pattern):
    19→    19→ *   probe all → filter by skip-set → sort by penalty+latency → round-robin tiebreak

   358→   358→function stripHtml(html: string): string {
   359→   359→  return html
   360→   360→    .replace(/<pre[^>]*>/gi, '')
   361→   361→    .replace(/<[^>]+>/g, ' ')
   362→   362→    .replace(/&amp;/g, '&')
   363→   363→    .replace(/&lt;/g, '<')
   364→   364→    .replace(/&gt;/g, '>')
   365→   365→    .replace(/&quot;/g, '"')
   366→   366→    .replace(/&#039;/g, "'")
   367→   367→    .replace(/\s+/g, ' ')
   368→   368→    .trim();
   369→   369→}
   370→   370→
   371→   371→function extractKeysFromEmail(email: ClawEmail): Array<{ service: string; key: string; envVar: string }> {
   372→   372→  const plainText = `${email.subject} ${email.bodyText} ${stripHtml(email.bodyHtml)}`;
   373→   373→  const results: Array<{ service: string; key: string; envVar: string }> = [];
   374→   374→
   375→   375→  for (const kp of KEY_PATTERNS) {
   376→   376→    const matches = plainText.match(new RegExp(kp.pattern.source, 'g'));
   377→   377→    if (matches) {
   378→   378→      for (const match of matches) {
   379→   379→        results.push({ service: kp.name, key: match, envVar: kp.envVar });
   380→   380→      }
   381→   381→    }
   382→   382→  }
   383→   383→
   384→   384→  return results;
   385→   385→}

   429→   429→  // Phase 1: Probe ALL providers in parallel
   430→   430→  // Phase 2: Select best by ok → penalty asc → latency asc, round-robin tiebreak
   431→   431→
   432→   432→  async probeAllProviders(): Promise<void> {
   433→   433→    const [gm, mt, oi] = await Promise.all([probeGuerrilla(), probeMailTm(), probeOpenInbox()]);
   434→   434→
   435→   435→    // Penalty decay (OmniKey: -1 per 2 min)
   436→   436→    for (const [name, fresh] of [['guerrilla', gm], ['mailtm', mt], ['openinbox', oi]] as const) {
   437→   437→      const prev = this.providerHealth[name[0] as ProviderName];
   438→   438→      const minsSinceProbe = prev.lastProbeAt > 0 ? (Date.now() - prev.lastProbeAt) / 120_000 : 0;
   439→   439→      const decay = Math.floor(minsSinceProbe);
   440→   440→      const decayedPenalty = Math.max(0, prev.penalty - decay);
   441→   441→
   442→   442→      if (fresh.status !== 'ok') {
   443→   443→        fresh.consecutiveFailures = prev.consecutiveFailures + 1;
   444→   444→        fresh.penalty = Math.min(decayedPenalty + 3, 10);
   445→   445→      } else {
   446→   446→        fresh.consecutiveFailures = 0;
   447→   447→        fresh.penalty = Math.max(decayedPenalty - 1, 0);
   448→   448→      }
   449→   449→    }
   450→   450→
   451→   451→    this.providerHealth = { guerrilla: gm, mailtm: mt, openinbox: oi };
   452→   452→    console.log(`[CLAW] Probed:`, Object.entries(this.providerHealth).map(
   453→   453→      ([n, h]) => `${n}=${h.status}(${h.latencyMs}ms,p${h.penalty})`
   454→   454→    ).join(' | '));
   455→   455→  }
   456→   456→
   457→   457→  async getBestProvider(skipProviders?: Set<string>): Promise<ProviderName> {
   458→   458→    const stale = Object.values(this.providerHealth).every(h => Date.now() - h.lastProbeAt > 120_000);
   459→   459→    if (stale) await this.probeAllProviders();
   460→   460→
   461→   461→    // OmniMail: provider resolution chain — filter by skip-set, then sort
   462→   462→    const candidates = Object.entries(this.providerHealth)
   463→   463→      .filter(([name, h]) => h.status === 'ok' && h.consecutiveFailures < 3)
   464→   464→      .filter(([name]) => !skipProviders?.has(name))
   465→   465→      .sort(([, a], [, b]) => a.penalty - b.penalty || a.latencyMs - b.latencyMs);
   466→   466→
   467→   467→    if (candidates.length > 0) {
   468→   468→      const bestPenalty = candidates[0][1].penalty;
   469→   469→      const topTier = candidates.filter(([, h]) => h.penalty === bestPenalty);
   470→   470→      const idx = this.roundRobinIdx % topTier.length;
   471→   471→      this.roundRobinIdx++;
   472→   472→      return topTier[idx][0] as ProviderName;
   473→   473→    }
   474→   474→
   475→   475→    // OmniMail: fallback — ignore skip-set, use lowest consecutive failures
   476→   476→    if (skipProviders?.size) {
   477→   477→      console.warn(`[CLAW] All providers in skip-set, falling back to best available`);
   478→   478→      return this.getBestProvider();
   479→   479→    }
   480→   480→
   481→   481→    const fallback = Object.entries(this.providerHealth)
   482→   482→      .sort(([, a], [, b]) => a.consecutiveFailures - b.consecutiveFailures);
   483→   483→    return (fallback[0]?.[0] || 'guerrilla') as ProviderName;
   484→   484→  }

   496→   496→      h.penalty = Math.min(h.penalty + 3, 10);
   497→   497→      h.consecutiveFailures++;
   498→   498→    }

   509→   509→   * Generate a new inbox. Two-phase selection (Agent-Reach):
   510→   510→   * Probe all providers → select best by penalty + latency + round-robin
   511→   511→   */
   512→   512→  async generate(provider?: ProviderName): Promise<ClawInbox> {
   513→   513→    const targetProvider = provider || await this.getBestProvider();
   514→   514→
   515→   515→    try {
   516→   516→      let inbox: ClawInbox;
   517→   517→      switch (targetProvider) {
   518→   518→        case 'guerrilla': inbox = await gmCreate(); break;
   519→   519→        case 'mailtm':   inbox = await mtCreate(); break;
   520→   520→        case 'openinbox': inbox = await oiCreate(); break;
   521→   521→      }
   522→   522→
   523→   523→      this.recordProviderResult(targetProvider, true);
   524→   524→      this.inboxes.push(inbox);
   525→   525→      this.totalGenerated++;
   526→   526→      this.stats[targetProvider].generated++;
   527→   527→
   528→   528→      // Trim pool
   529→   529→      if (this.inboxes.length > this.config.maxInboxes) {
   530→   530→        this.inboxes = this.inboxes.slice(-this.config.maxInboxes);
   531→   531→      }
   532→   532→
   533→   533→      console.log(`[CLAW] Generated ${targetProvider} inbox: ${inbox.email} (expires in ${Math.round((inbox.expiresAt - Date.now()) / 60000)}min)`);
   534→   534→      return inbox;
   535→   535→    } catch (err) {
   536→   536→      this.recordProviderResult(targetProvider, false);
   537→   537→      this.stats[targetProvider].errors++;
   538→   538→      // OmniMail: classify error for upstream handling
   539→   539→      const errMsg = (err as Error).message || '';
   540→   540→      const retryable = [429, 408, 500, 502, 503, 504].some(c => errMsg.includes(String(c)))
   541→   541→        || /timeout|econnrefused|econnreset/i.test(errMsg);
   542→   542→      console.warn(`[CLAW] ${targetProvider} generation failed (${retryable ? 'retryable' : 'non-retryable'}):`, errMsg);
   543→   543→      (err as any)._clawRetryable = retryable;
   544→   544→      throw err;
   545→   545→    }
   546→   546→  }

   551→   551→  async checkInbox(inbox: ClawInbox): Promise<ClawEmail[]> {
   552→   552→    // Skip expired
   553→   553→    if (Date.now() > inbox.expiresAt) {
   554→   554→      inbox.status = 'expired';
   555→   555→      return [];
   556→   556→    }
   557→   557→
   558→   558→    inbox.status = 'polling';
   559→   559→
   560→   560→    try {
   561→   561→      let emails: ClawEmail[];
   562→   562→      switch (inbox.provider) {
   563→   563→        case 'guerrilla': emails = await gmCheck(inbox); break;
   564→   564→        case 'mailtm':   emails = await mtCheck(inbox); break;
   565→   565→        case 'openinbox': emails = await oiCheck(inbox); break;
   566→   566→        default: return [];
   567→   567→      }
   568→   568→
   569→   569→      inbox.emailCount = emails.length;
   570→   570→      this.totalEmailsRead += emails.length;
   571→   571→      this.stats[inbox.provider].emailsRead += emails.length;
   572→   572→
   573→   573→      // Auto-extract keys from new emails — Tier 1: pending queue + validation
   574→   574→      for (const email of emails) {
   575→   575→        const keys = extractKeysFromEmail(email);
   576→   576→        for (const k of keys) {
   577→   577→          console.log(`[CLAW] KEY EXTRACTED [${k.service}]: ${k.key.slice(0, 12)}... from ${inbox.email}`);
   578→   578→          this.totalKeysExtracted++;
   579→   579→          this.lastKeyExtracted = k.key;
   580→   580→
   581→   581→          const pendingId = `pending-${Date.now()}-${Math.random().toString(36).slice(2, 6)}`;
   582→   582→          const pending: PendingKey = {
   583→   583→            id: pendingId,
   584→   584→            service: k.service,
   585→   585→            key: k.key,
   586→   586→            envVar: k.envVar,
   587→   587→            extractedAt: Date.now(),
   588→   588→            inboxId: inbox.id,
   589→   589→            inboxEmail: inbox.email,
   590→   590→            status: 'pending',
   591→   591→          };
   592→   592→          this.pendingKeys.push(pending);
   593→   593→          if (this.pendingKeys.length > MAX_PENDING_KEYS) this.pendingKeys = this.pendingKeys.slice(-MAX_PENDING_KEYS);
   594→   594→
   595→   595→          audit('key.extracted', `${k.service} key from ${inbox.email}`, {
   596→   596→            pendingId, keyPreview: k.key.slice(0, 12) + '...', inboxId: inbox.id,
   597→   597→          });
   598→   598→
   599→   599→          if (this.autoApproveKeys) {
   600→   600→            // Auto-approve: validate then inject
   601→   601→            await this.validateAndInject(pending);
   602→   602→          }
   603→   603→
   604→   604→          // Notify callback (with pendingId so caller can approve/reject)
   605→   605→          this.keyCallback?.(k.service, k.key, k.envVar, pendingId);
   606→   606→        }
   607→   607→      }
   608→   608→
   609→   609→      return emails;
   610→   610→    } catch (err) {
   611→   611→      inbox.status = 'error';
   612→   612→      this.stats[inbox.provider].errors++;
   613→   613→      console.warn(`[CLAW] Check error [${inbox.provider}]:`, (err as Error).message);
   614→   614→      return [];
   615→   615→    }
   616→   616→  }

   732→   732→  async validateAndInject(pending: PendingKey): Promise<boolean> {
