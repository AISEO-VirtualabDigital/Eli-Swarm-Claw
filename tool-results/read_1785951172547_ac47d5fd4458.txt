   255→   255→  async loadSettings(): Promise<void> {
   256→   256→    this.settings = Object.assign(
   257→   257→      {},
   258→   258→      DEFAULT_SETTINGS,
   259→   259→      (await this.loadData()) as Partial<SkillLayerSettings> | null,

   263→   263→    // omnigent now picks the harness itself. Strip every legacy harness key

   265→   265→    // reintroduce harness behavior. A skill that had a harness selected simply
   266→   266→    // reverts to the Default agent (it has no `skillAgent` entry). Everything
   267→   267→    // else in data.json is preserved by the Object.assign above.
   268→   268→    const raw = this.settings as unknown as Record<string, unknown>;
   269→   269→    for (const key of [
   270→   270→      // NOTE (M15): `skillHarness` is NO LONGER stripped — it is repurposed as
   271→   271→      // the per-skill omnigent `--harness` map (string values). Any stale value
   272→   272→      // from the removed M4–M7 harness selector (a different shape) is made safe
   273→   273→      // by `resolveHarness`, which fails closed against the hardcoded harness
   274→   274→      // allowlist, so a non-member simply emits no `--harness`.
   275→   275→      "discoveredHarnesses",
   276→   276→      "customHarnesses",
   277→   277→      "omnigentHarness",
   278→   278→      // M11: the user-configurable invocation template is gone — the "Copy
   279→   279→      // invocation" action now uses a FIXED natural-language form
   280→   280→      // (`Use the <name> skill.`). Strip it fail-closed so a stale data.json can
   281→   281→      // never reintroduce it; every OTHER setting (scanRoots, pins, skillAgent,
   282→   282→      // omnigentBinaryPath, omnigentServerUrl, appendVaultAnchor, …) is preserved
   283→   283→      // by the Object.assign above. NOTE (M19): `omnigentServerUrl` is BACK as a
   284→   284→      // real setting (per-launch `--server` target), so it is NOT stripped.
   285→   285→      "invocationTemplate",
   286→   286→    ]) {
   287→   287→      delete raw[key];
   288→   288→    }

   291→   291→    // auto-added (external kind, disabled) — they cluttered settings and mixed
   292→   292→    // machine-global skills with the user's in-vault skills. Only the exact

   377→   377→  async createToolFolder(seg: string): Promise<string | null> {

   381→   381→        !this.addableFolderSegments("command").includes(seg)) {
   382→   382→      return "Unknown folder.";
   383→   383→    }

   385→   385→      await fs.promises.mkdir(nodePath.join(base, seg), { recursive: true });
   386→   386→      await this.rescan();
   387→   387→      return null;
   388→   388→    } catch (err) {
   389→   389→      console.error("[skill-layer] createToolFolder failed:", err);

   394→   394→  /**
   395→   395→   * Create a new SKILL inside the tool folder `folderSeg`: `<folderSeg>/<name>/
   396→   396→   * SKILL.md` with a minimal frontmatter stub, then open it in Obsidian (like the
   397→   397→   * Open file button). Returns an error string, or null on success. The name is
   398→   398→   * sanitized to a safe folder name; refuses to overwrite an existing file.
   399→   399→   */
   400→   400→  async createSkillInFolder(folderSeg: string, rawName: string): Promise<string | null> {
   401→   401→    const base = this.detector.vaultBasePath();
   402→   402→    if (!base || !this.detector.canScanExternal()) return "Requires the desktop app.";
   403→   403→    const name = this.sanitizeItemName(rawName);
   404→   404→    if (!name) return "Please enter a valid name.";
   405→   405→    const relPath = `${folderSeg}/${name}/SKILL.md`;
   406→   406→    const abs = nodePath.join(base, relPath);
   407→   407→    if (fs.existsSync(abs)) return `A skill named "${name}" already exists here.`;
   408→   408→    const body = `---\nname: ${name}\ndescription: \n---\n\n# ${name}\n\n`;
   409→   409→    try {
   410→   410→      await fs.promises.mkdir(nodePath.dirname(abs), { recursive: true });
   411→   411→      await fs.promises.writeFile(abs, body, "utf8");
   412→   412→    } catch (err) {
   413→   413→      console.error("[skill-layer] createSkillInFolder failed:", err);
   414→   414→      return "Could not create the skill file.";
   415→   415→    }
   416→   416→    await this.rescan();
   417→   417→    await this.openCreatedFile(relPath, abs);
   418→   418→    return null;
   419→   419→  }
   420→   420→
   421→   421→  /**
   422→   422→   * Create a new COMMAND inside the tool folder `folderSeg`: `<folderSeg>/
   423→   423→   * <name>.md` (any markdown in a commands folder is a command), then open it in
   424→   424→   * Obsidian. Returns an error string, or null on success.
   425→   425→   */
   426→   426→  async createCommandInFolder(folderSeg: string, rawName: string): Promise<string | null> {
   427→   427→    const base = this.detector.vaultBasePath();
   428→   428→    if (!base || !this.detector.canScanExternal()) return "Requires the desktop app.";
   429→   429→    const name = this.sanitizeItemName(rawName);
   430→   430→    if (!name) return "Please enter a valid name.";
   431→   431→    const relPath = `${folderSeg}/${name}.md`;
   432→   432→    const abs = nodePath.join(base, relPath);
   433→   433→    if (fs.existsSync(abs)) return `A command named "${name}" already exists here.`;
   434→   434→    const body = `# ${name}\n\n`;
   435→   435→    try {
   436→   436→      await fs.promises.mkdir(nodePath.dirname(abs), { recursive: true });
   437→   437→      await fs.promises.writeFile(abs, body, "utf8");
   438→   438→    } catch (err) {
   439→   439→      console.error("[skill-layer] createCommandInFolder failed:", err);
   440→   440→      return "Could not create the command file.";
   441→   441→    }
   442→   442→    await this.rescan();
   443→   443→    await this.openCreatedFile(relPath, abs);
   444→   444→    return null;
   445→   445→  }

   606→   606→  // --- Browser view ------------------------------------------------------
   607→   607→  async activateView(): Promise<void> {
   608→   608→    const { workspace } = this.app;
   609→   609→    const existing = workspace.getLeavesOfType(SKILL_LAYER_VIEW);
   610→   610→    if (existing.length > 0) {
   611→   611→      await workspace.revealLeaf(existing[0]);
   612→   612→      // Revealing an existing leaf does NOT re-fire onOpen, so refresh from
   613→   613→      // disk here to pick up external edits (incl. dot-folder / external roots).
   614→   614→      await this.rescan();
   615→   615→      this.applyPanelWidth();
   616→   616→      return;
   617→   617→    }
   618→   618→    const leaf = workspace.getRightLeaf(false);
   619→   619→    if (!leaf) return;
   620→   620→    // A freshly created leaf's onOpen() runs the rescan itself.
   621→   621→    await leaf.setViewState({ type: SKILL_LAYER_VIEW, active: true });
   622→   622→    await workspace.revealLeaf(leaf);
   623→   623→    this.applyPanelWidth();
   624→   624→  }
   625→   625→
   626→   626→  /**
   627→   627→   * Open the browser side panel at the configured `panelWidth` so it always uses
   628→   628→   * a consistent width rather than whatever the sidebar was last dragged to. The
   629→   629→   * right-sidebar width has no public API, so this sets the width on the right
   630→   630→   * split's container element (undocumented internals) behind a guarded cast and
   631→   631→   * a `requestAnimationFrame` (so it lands after the reveal lays out). Best-effort
   632→   632→   * — silently no-ops if the internals aren't shaped as expected or on mobile.
   633→   633→   */
   634→   634→  private applyPanelWidth(): void {
   635→   635→    const width = this.settings.panelWidth;
   636→   636→    if (!width || width <= 0) return;
   637→   637→    const rightSplit = (this.app.workspace as unknown as {
   638→   638→      rightSplit?: { containerEl?: HTMLElement; collapsed?: boolean };
   639→   639→    }).rightSplit;
   640→   640→    const el = rightSplit?.containerEl;
   641→   641→    if (!el) return;
   642→   642→    window.requestAnimationFrame(() => {
   643→   643→      try {
   644→   644→        // setCssStyles (not a direct .style assignment) per Obsidian guidelines.
   645→   645→        el.setCssStyles({ width: `${width}px` });
   646→   646→        // Nudge Obsidian to reconcile the split sizes to the new width.
   647→   647→        this.app.workspace.requestSaveLayout?.();
   648→   648→      } catch (err) {
   649→   649→        console.error("[skill-layer] applyPanelWidth failed:", err);
   650→   650→      }
   651→   651→    });
   652→   652→  }
   653→   653→
   654→   654→  /**
   655→   655→   * Ribbon behavior: toggle the Skill and Harness Manager pane.
   656→   656→   * - Not open                → open + reveal (via activateView).
   657→   657→   * - Open but not active      → reveal/focus it (via activateView).
   658→   658→   * - Open AND active/visible  → close (detach its leaves).
   659→   659→   *
   660→   660→   * The command-palette "Open skills browser" entry stays on activateView
   661→   661→   * (pure open+reveal, never closes) — closing on a command named "Open"
   662→   662→   * would be surprising. Only the ribbon toggles.
   663→   663→   */
   664→   664→  async toggleView(): Promise<void> {
   665→   665→    const { workspace } = this.app;
   666→   666→    const existing = workspace.getLeavesOfType(SKILL_LAYER_VIEW);
   667→   667→    const activeView = workspace.getActiveViewOfType(SkillBrowserView);
   668→   668→    const isActiveVisible =
   669→   669→      activeView !== null && existing.includes(activeView.leaf);
   670→   670→    const action = decideToggleAction(existing.length > 0, isActiveVisible);
   671→   671→    if (action === "close") {
   672→   672→      workspace.detachLeavesOfType(SKILL_LAYER_VIEW);
   673→   673→      return;
   674→   674→    }
   675→   675→    // "open" and "reveal" both funnel through activateView (open+reveal+rescan).
   676→   676→    await this.activateView();
   677→   677→  }

   780→   780→
   781→   781→  /**
   782→   782→   * Scan `<vault>/.claude/agents` and `~/.claude/agents` for `*.md` subagents
   783→   783→   * into the cache (no view refresh). Name = frontmatter `name:` (else filename
   784→   784→   * stem); project entries win over global on a name clash. Missing dirs yield
   785→   785→   * zero (no error); all fs access is wrapped so this never throws.
   786→   786→   */
   787→   787→  private scanClaudeAgents(): void {
   788→   788→    const base = this.detector.vaultBasePath();
   789→   789→    const home = os.homedir();
   790→   790→    // M18: scan EVERY tool's agents folder (.claude/agents, .cursor/agents,
   791→   791→    // .codex/agents, …) — project (vault-relative) first, then home — so agents
   792→   792→    // from any assistant appear. Project wins over global on a name clash.
   793→   793→    const dirs: { dir: string; source: "project" | "global" }[] = [];
   794→   794→    for (const seg of agentFolderSegments()) {
   795→   795→      if (base) dirs.push({ dir: nodePath.join(base, seg), source: "project" });
   796→   796→    }
   797→   797→    for (const seg of agentFolderSegments()) {
   798→   798→      dirs.push({ dir: joinHome(home, seg), source: "global" });
   799→   799→    }
   800→   800→
   801→   801→    const byName = new Map<string, ClaudeAgent>();
   802→   802→    for (const { dir, source } of dirs) {
   803→   803→      let entries: string[];
   804→   804→      try {
   805→   805→        entries = fs.readdirSync(dir);
   806→   806→      } catch {
   807→   807→        continue; // missing dir → skip
   808→   808→      }
   809→   809→      for (const entry of entries.sort()) {
   810→   810→        if (!/\.md$/i.test(entry)) continue;
   811→   811→        const abs = nodePath.join(dir, entry);
   812→   812→        let text: string;
   813→   813→        try {
   814→   814→          if (!fs.statSync(abs).isFile()) continue;
   815→   815→          text = fs.readFileSync(abs, "utf8");
   816→   816→        } catch {
   817→   817→          continue;
   818→   818→        }
   819→   819→        const meta = parseClaudeAgentFrontmatter(text);
   820→   820→        const name = meta.name && meta.name.trim() ? meta.name.trim() : entry.replace(/\.md$/i, "");
   821→   821→        // Project (scanned first) wins; don't let global clobber it.
   822→   822→        if (byName.has(name)) continue;
   823→   823→        byName.set(name, {
   824→   824→          name,
   825→   825→          path: abs,
   826→   826→          source,
   827→   827→          ...(meta.description ? { description: meta.description } : {}),
   828→   828→        });
   829→   829→      }
   830→   830→    }
   831→   831→    this.claudeAgents = Array.from(byName.values()).sort((a, b) =>
   832→   832→      a.name.localeCompare(b.name),
   833→   833→    );
   834→   834→  }

   876→   876→  /**
   877→   877→   * Scan every tool's commands folder (`.claude/commands`, `.codex/prompts`, …)
   878→   878→   * — project (vault-relative) first, then home — for `*.md` files, each of
   879→   879→   * which is one command. Unlike skills there's no SKILL.md rule: any markdown in
   880→   880→   * a commands folder is a command. Built as Skill objects with kind:"command" so
   881→   881→   * they reuse the whole row UI / per-item state. Deduped by absolute path
   882→   882→   * (project wins). All fs access wrapped; never throws.
   883→   883→   */
   884→   884→  private scanCommands(): void {
   885→   885→    const base = this.detector.vaultBasePath();
   886→   886→    const home = os.homedir();
   887→   887→    const dirs: { dir: string; vaultRel: string | null }[] = [];
   888→   888→    for (const seg of commandFolderSegments()) {
   889→   889→      if (base) dirs.push({ dir: nodePath.join(base, seg), vaultRel: seg });
   890→   890→    }
   891→   891→    for (const seg of commandFolderSegments()) {
   892→   892→      dirs.push({ dir: joinHome(home, seg), vaultRel: null });
   893→   893→    }
   894→   894→
   895→   895→    const byId = new Map<string, Skill>();
   896→   896→    for (const { dir, vaultRel } of dirs) {
   897→   897→      let entries: string[];
   898→   898→      try {
   899→   899→        entries = fs.readdirSync(dir);
   900→   900→      } catch {
   901→   901→        continue; // missing dir → skip
   902→   902→      }
   903→   903→      for (const entry of entries.sort()) {
   904→   904→        if (!/\.md$/i.test(entry)) continue;
   905→   905→        const abs = nodePath.join(dir, entry);
   906→   906→        let text: string;
   907→   907→        try {
   908→   908→          if (!fs.statSync(abs).isFile()) continue;
   909→   909→          text = fs.readFileSync(abs, "utf8");
   910→   910→        } catch {
   911→   911→          continue;
   912→   912→        }
   913→   913→        if (byId.has(abs)) continue;
   914→   914→        const fm = parseFrontmatter(text);
   915→   915→        const stem = entry.replace(/\.md$/i, "");
   916→   916→        const name = fm.name && fm.name.trim() ? fm.name.trim() : stem;
   917→   917→        const description =
   918→   918→          fm.description && fm.description.trim()
   919→   919→            ? fm.description.trim()
   920→   920→            : firstHeading(text) ?? "(no description)";
   921→   921→        const vaultPath = vaultRel ? `${vaultRel}/${entry}` : null;
   922→   922→        const relForTag = vaultPath ?? `${nodePath.basename(dir)}/${entry}`;
   923→   923→        byId.set(abs, {
   924→   924→          id: abs,
   925→   925→          kind: "command",
   926→   926→          name,
   927→   927→          description,
   928→   928→          path: abs,
   929→   929→          vaultPath,
   930→   930→          sourceRoot: dir,
   931→   931→          sourceLabel: inferSourceLabel(abs),
   932→   932→          detection: vaultRel ? "adapter" : "external",
   933→   933→          tags: resolveSkillTags({
   934→   934→            relativePath: relForTag,
   935→   935→            description,
   936→   936→            frontmatterTags: fm.tags ?? [],
   937→   937→          }),
   938→   938→        });
   939→   939→      }
   940→   940→    }
   941→   941→    this.commands = Array.from(byId.values()).sort((a, b) =>
   942→   942→      a.name.localeCompare(b.name),
   943→   943→    );
   944→   944→  }

  1333→  1333→  /**
  1334→  1334→   * Discover the harnesses omnigent has configured by running
  1335→  1335→   * `omnigent config list` and parsing its "Credentials (by harness)" section.
  1336→  1336→   * Best-effort + fail-quiet: desktop-gated, the omnigent binary resolved
  1337→  1337→   * silently (no Notice), a bounded stdout capture with a hard timeout, and any
  1338→  1338→   * failure leaves the cache empty (the dropdown then falls back to the full
  1339→  1339→   * allowlist). This is a READ-ONLY omnigent query (config list), not a launch —
  1340→  1340→   * distinct from the hardened launch spawn, and it never passes user input.
  1341→  1341→   */
  1342→  1342→  private async discoverConfiguredHarnesses(): Promise<void> {
  1343→  1343→    const finish = (list: ConfiguredHarness[]) => {
  1344→  1344→      this.configuredHarnesses = list;
  1345→  1345→      this.harnessesDiscovered = true;
  1346→  1346→      this.refreshViews();
  1347→  1347→    };
  1348→  1348→    if (!this.detector.canScanExternal()) return finish([]);
  1349→  1349→    const resolution = resolveOmnigentBinary({
  1350→  1350→      override: this.settings.omnigentBinaryPath,
  1351→  1351→      homedir: os.homedir(),
  1352→  1352→      exists: (p) => fs.existsSync(p),
  1353→  1353→    });
  1354→  1354→    if (resolution.status !== "ok") return finish([]);
  1355→  1355→
  1356→  1356→    const env = this.launchEnv();
  1357→  1357→    let child;
  1358→  1358→    try {
  1359→  1359→      // Fixed args only; no user input. stdout piped for parsing; stderr/stdin ignored.
  1360→  1360→      child = spawn(resolution.path, ["config", "list"], {
  1361→  1361→        env,
  1362→  1362→        shell: false,
  1363→  1363→        windowsHide: true,
  1364→  1364→        stdio: ["ignore", "pipe", "ignore"],
  1365→  1365→      });
  1366→  1366→    } catch (err) {
  1367→  1367→      console.error("[skill-layer] harness discovery spawn threw:", err);
  1368→  1368→      return finish([]);
  1369→  1369→    }
  1370→  1370→    let stdout = "";
  1371→  1371→    let done = false;
  1372→  1372→    const settle = (list: ConfiguredHarness[]) => {
  1373→  1373→      if (done) return;
  1374→  1374→      done = true;
  1375→  1375→      finish(list);
  1376→  1376→    };
  1377→  1377→    // Hard timeout so a hung `omnigent` never leaves discovery pending.
  1378→  1378→    const timer = window.setTimeout(() => {
  1379→  1379→      try {
  1380→  1380→        child.kill();
  1381→  1381→      } catch {
  1382→  1382→        /* ignore */
  1383→  1383→      }
  1384→  1384→      settle([]);
  1385→  1385→    }, 8000);
  1386→  1386→    child.stdout?.on("data", (chunk: Buffer) => {
  1387→  1387→      // Bound the captured output defensively.
  1388→  1388→      if (stdout.length < 64_000) stdout += chunk.toString();
  1389→  1389→    });
  1390→  1390→    child.on("error", (err) => {
  1391→  1391→      console.error("[skill-layer] harness discovery error:", err);
  1392→  1392→      window.clearTimeout(timer);
  1393→  1393→      settle([]);
  1394→  1394→    });
  1395→  1395→    child.on("close", (code) => {
  1396→  1396→      window.clearTimeout(timer);
  1397→  1397→      settle(code === 0 ? parseConfiguredHarnesses(stdout) : []);
  1398→  1398→    });
  1399→  1399→    child.unref?.();
  1400→  1400→  }

  1676→  1676→   * full path with a sanitized basename, then guarantees uniqueness against a
  1677→  1677→   * live set so two distinct skills can never collide on the same id.

  1681→  1681→    for (let i = 0; i < skillId.length; i++) {
  1682→  1682→      hash = ((hash << 5) + hash + skillId.charCodeAt(i)) | 0;
  1683→  1683→    }

  1685→  1685→      (skillId.replace(/\\/g, "/").split("/").pop() ?? "")
  1686→  1686→        .toLowerCase()
  1687→  1687→        .replace(/[^a-z0-9]+/g, "-")
  1688→  1688→        .replace(/^-+|-+$/g, "")
  1689→  1689→        .slice(0, 24) || "skill";
  1690→  1690→    const stem = `run-${(hash >>> 0).toString(36)}-${slug}`;
  1691→  1691→    let candidate = stem;
