   162→   162→// ─── Helper: Count-up animation hook (fixed count) ───────────
   163→   163→function useMetricCounts(targets: number[], duration = 800) {
   164→   164→  const [counts, setCounts] = useState(targets.map(() => 0));
   165→   165→  const [started, setStarted] = useState(false);
   166→   166→
   167→   167→  useEffect(() => {
   168→   168→    if (!started) return;
   169→   169→    const startTime = performance.now();
   170→   170→  let rafId: number;
   171→   171→    const step = (now: number) => {
   172→   172→      const elapsed = now - startTime;
   173→   173→      const progress = Math.min(elapsed / duration, 1);
   174→   174→      const eased = 1 - Math.pow(1 - progress, 3);
   175→   175→      setCounts(targets.map((t) => Math.round(eased * t)));
   176→   176→      if (progress < 1) rafId = requestAnimationFrame(step);
   177→   177→    };
   178→   178→    rafId = requestAnimationFrame(step);
   179→   179→    return () => cancelAnimationFrame(rafId);
   180→   180→  }, [started, duration, ...targets]);
   181→   181→
   182→   182→  return { counts, start: () => setStarted(true) };
   183→   183→}

   186→   186→function useLiveTime() {
   187→   187→  const [time, setTime] = useState('');
   188→   188→  useEffect(() => {
   189→   189→    const update = () => {
   190→   190→      const now = new Date();
   191→   191→      setTime(
   192→   192→        now.toLocaleTimeString('en-US', {
   193→   193→          hour: '2-digit',
   194→   194→          minute: '2-digit',
   195→   195→          second: '2-digit',
   196→   196→          hour12: false,
   197→   197→        })
   198→   198→      );
   199→   199→    };
   200→   200→    update();
   201→   201→    const interval = setInterval(update, 1000);
   202→   202→    return () => clearInterval(interval);
   203→   203→  }, []);
   204→   204→  return time;
   205→   205→}

   587→   587→function ChatView() {
   588→   588→  const [messages, setMessages] = useState<ChatMessage[]>([]);
   589→   589→  const [input, setInput] = useState('');
   590→   590→  const [isLoading, setIsLoading] = useState(false);
   591→   591→  const messagesEndRef = useRef<HTMLDivElement>(null);
   592→   592→  const textareaRef = useRef<HTMLTextAreaElement>(null);
   593→   593→
   594→   594→  const scrollToBottom = useCallback(() => {
   595→   595→    messagesEndRef.current?.scrollIntoView({ behavior: 'smooth' });
   596→   596→  }, []);
   597→   597→
   598→   598→  useEffect(() => {
   599→   599→    scrollToBottom();
   600→   600→  }, [messages, isLoading, scrollToBottom]);
   601→   601→
   602→   602→  // Welcome message on mount
   603→   603→  useEffect(() => {
   604→   604→    setMessages([
   605→   605→      {
   606→   606→        id: 'welcome',
   607→   607→        role: 'eli',
   608→   608→        content: ELI_WELCOME,
   609→   609→        timestamp: new Date(),
   610→   610→      },
   611→   611→    ]);
   612→   612→  }, []);
   613→   613→
   614→   614→  const handleSend = async () => {
   615→   615→    const trimmed = input.trim();
   616→   616→    if (!trimmed || isLoading) return;
   617→   617→
   618→   618→    const userMsg: ChatMessage = {
   619→   619→      id: `user-${Date.now()}`,
   620→   620→      role: 'user',
   621→   621→      content: trimmed,
   622→   622→      timestamp: new Date(),
   623→   623→    };
   624→   624→
   625→   625→    setMessages((prev) => [...prev, userMsg]);
   626→   626→    setInput('');
   627→   627→    setIsLoading(true);
   628→   628→
   629→   629→    // Reset textarea height
   630→   630→    if (textareaRef.current) {
   631→   631→      textareaRef.current.style.height = 'auto';
   632→   632→    }
   633→   633→
   634→   634→    try {
   635→   635→      const history = messages
   636→   636→        .filter((m) => m.id !== 'welcome')
   637→   637→        .map((m) => ({ role: m.role, content: m.content }));
   638→   638→
   639→   639→      const res = await fetch('/api/eli-chat', {
   640→   640→        method: 'POST',
   641→   641→        headers: { 'Content-Type': 'application/json' },
   642→   642→        body: JSON.stringify({ message: trimmed, history }),
   643→   643→      });
   644→   644→
   645→   645→      const data = await res.json();
   646→   646→
   647→   647→      const eliMsg: ChatMessage = {
   648→   648→        id: `eli-${Date.now()}`,
   649→   649→        role: 'eli',

   651→   651→        sources: data.sources,

   655→   655→      setMessages((prev) => [...prev, eliMsg]);
   656→   656→    } catch {

   658→   658→        id: `error-${Date.now()}`,

   698→   698→  };
   699→   699→
