   164→// ─── Helper: Count-up animation hook (fixed count) ───────────
   165→function useMetricCounts(targets: number[], duration = 800) {
   166→  const [counts, setCounts] = useState(targets.map(() => 0));
   167→  const [started, setStarted] = useState(false);
   168→
   169→  useEffect(() => {
   170→    if (!started) return;
   171→    const startTime = performance.now();
   172→  let rafId: number;
   173→    const step = (now: number) => {
   174→      const elapsed = now - startTime;
   175→      const progress = Math.min(elapsed / duration, 1);
   176→      const eased = 1 - Math.pow(1 - progress, 3);
   177→      setCounts(targets.map((t) => Math.round(eased * t)));
   178→      if (progress < 1) rafId = requestAnimationFrame(step);
   179→    };
   180→    rafId = requestAnimationFrame(step);
   181→    return () => cancelAnimationFrame(rafId);
   182→  }, [started, duration, ...targets]);
   183→
   184→  return { counts, start: () => setStarted(true) };
   185→}

   187→// ─── Helper: Live time ──────────────────────────────────────────
   188→function useLiveTime() {
   189→  const [time, setTime] = useState('');
   190→  useEffect(() => {
   191→    const update = () => {
   192→      const now = new Date();
   193→      setTime(
   194→        now.toLocaleTimeString('en-US', {
   195→          hour: '2-digit',
   196→          minute: '2-digit',
   197→          second: '2-digit',
   198→          hour12: false,
   199→        })
   200→      );
   201→    };
   202→    update();
   203→    const interval = setInterval(update, 1000);
   204→    return () => clearInterval(interval);
   205→  }, []);
   206→  return time;
   207→}

   589→function ChatView() {
   590→  const [messages, setMessages] = useState<ChatMessage[]>([]);
   591→  const [input, setInput] = useState('');
   592→  const [isLoading, setIsLoading] = useState(false);
   593→  const messagesEndRef = useRef<HTMLDivElement>(null);
   594→  const textareaRef = useRef<HTMLTextAreaElement>(null);
   595→
   596→  const scrollToBottom = useCallback(() => {
   597→    messagesEndRef.current?.scrollIntoView({ behavior: 'smooth' });
   598→  }, []);
   599→
   600→  useEffect(() => {
   601→    scrollToBottom();
   602→  }, [messages, isLoading, scrollToBottom]);
   603→
   604→  // Welcome message on mount
   605→  useEffect(() => {
   606→    setMessages([
   607→      {
   608→        id: 'welcome',
   609→        role: 'eli',
   610→        content: ELI_WELCOME,
   611→        timestamp: new Date(),
   612→      },
   613→    ]);
   614→  }, []);
   615→
   616→  const handleSend = async () => {
   617→    const trimmed = input.trim();
   618→    if (!trimmed || isLoading) return;
   619→
   620→    const userMsg: ChatMessage = {
   621→      id: `user-${Date.now()}`,
   622→      role: 'user',
   623→      content: trimmed,
   624→      timestamp: new Date(),
   625→    };
   626→
   627→    setMessages((prev) => [...prev, userMsg]);
   628→    setInput('');
   629→    setIsLoading(true);
   630→
   631→    // Reset textarea height
   632→    if (textareaRef.current) {
   633→      textareaRef.current.style.height = 'auto';
   634→    }
   635→
   636→    try {
   637→      const history = messages
   638→        .filter((m) => m.id !== 'welcome')
   639→        .map((m) => ({ role: m.role, content: m.content }));
   640→
   641→      const res = await fetch('/api/eli-chat', {
   642→        method: 'POST',
   643→        headers: { 'Content-Type': 'application/json' },
   644→        body: JSON.stringify({ message: trimmed, history }),
   645→      });
   646→
   647→      const data = await res.json();
   648→
   649→      const eliMsg: ChatMessage = {
   650→        id: `eli-${Date.now()}`,
   651→        role: 'eli',
   652→        content: data.response || 'I encountered an issue processing your request. Please try again.',
   653→        sources: data.sources,
   654→        timestamp: new Date(),
   655→      };
   656→
   657→      setMessages((prev) => [...prev, eliMsg]);
   658→    } catch {
   659→      const errorMsg: ChatMessage = {
   660→        id: `error-${Date.now()}`,
   661→        role: 'eli',
   662→        content: 'Sorry, I encountered a connection error. Please check that the server is running and try again.',
   663→        timestamp: new Date(),
   664→      };
   665→      setMessages((prev) => [...prev, errorMsg]);
   666→    } finally {
   667→      setIsLoading(false);
   668→    }
   669→  };
   670→
   671→  const handleKeyDown = (e: React.KeyboardEvent<HTMLTextAreaElement>) => {
   672→    if (e.key === 'Enter' && !e.shiftKey) {
   673→      e.preventDefault();
   674→      handleSend();
   675→    }
   676→  };
   677→
   678→  const handleTextareaChange = (e: React.ChangeEvent<HTMLTextAreaElement>) => {
   679→    setInput(e.target.value);
   680→    // Auto-resize
   681→    const ta = e.target;
   682→    ta.style.height = 'auto';
   683→    ta.style.height = `${Math.min(ta.scrollHeight, 160)}px`;
   684→  };

   689→      if (line.startsWith('- ') || line.startsWith('• ')) {
   690→        return (
   691→          <div key={i} className="flex gap-2 pl-1">
   692→            <span className="mt-0.5">•</span>
   693→            <span>{line.replace(/^[•-]\s*/, '')}</span>
   694→          </div>
   695→        );
   696→      }
   697→      if (line.trim() === '') return <div key={i} className="h-2" />;
   698→      return <p key={i} className="leading-relaxed">{line}</p>;
   699→    });
   700→  };
   701→
   702→  return (
   703→    <div className="flex flex-col h-full animate-fadeIn">
   704→      {/* Chat Header */}
   705→      <div
   706→        className="px-5 py-3 flex items-center gap-3 flex-shrink-0"
   707→        style={{ borderBottom: '1px solid #e2e8f0' }}
   708→      >
   709→        <div
   710→          className="w-8 h-8 rounded-full flex items-center justify-center text-xs font-bold"
   711→          style={{
   712→            background: 'linear-gradient(135deg, #7c3aed, #6d28d9)',
   713→            boxShadow: '0 0 12px rgba(124,58,237,0.2)',
   714→          }}
   715→        >
   716→          E
   717→        </div>
   718→        <div>
   719→          <div className="text-sm font-semibold text-[#1e293b]">Eli</div>
   720→          <div className="text-[10px] text-[#059669] flex items-center gap-1">
   721→            <div className="w-1.5 h-1.5 rounded-full bg-[#059669]" />
   722→            Growth Intelligence Online
   723→          </div>
   724→        </div>
   725→      </div>
   726→
   727→      {/* Messages */}
   728→      <ScrollArea className="flex-1 px-4 md:px-6 py-4">
   729→        <div className="max-w-3xl mx-auto space-y-4">
   730→          {messages.map((msg) => (
