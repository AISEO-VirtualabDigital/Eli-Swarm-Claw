   162→// ─── Helper: Count-up animation hook ────────────────────────────
   163→function useMetricCounts(targets: number[], duration = 800) {
   164→  const [counts, setCounts] = useState(targets.map(() => 0));
   165→  const [started, setStarted] = useState(false);
   166→
   167→  useEffect(() => {
   168→    if (!started) return;
   169→    const startTime = performance.now();
   170→    let rafId: number;
   171→    const step = (now: number) => {
   172→      const elapsed = now - startTime;
   173→      const progress = Math.min(elapsed / duration, 1);
   174→      const eased = 1 - Math.pow(1 - progress, 3);
   175→      setCounts(targets.map((t) => Math.round(eased * t)));
   176→      if (progress < 1) rafId = requestAnimationFrame(step);
   177→    };
   178→    rafId = requestAnimationFrame(step);
   179→    return () => cancelAnimationFrame(rafId);
   180→  }, [started, duration, ...targets]);
   181→
   182→  return { counts, start: () => setStarted(true) };
   183→}

   185→// ─── Helper: Live time ──────────────────────────────────────────
   186→function useLiveTime() {
   187→  const [time, setTime] = useState('');
   188→  useEffect(() => {
   189→    const update = () => {
   190→      const now = new Date();
   191→      setTime(
   192→        now.toLocaleTimeString('en-US', {
   193→          hour: '2-digit',
   194→          minute: '2-digit',
   195→          second: '2-digit',
   196→          hour12: false,
   197→        })
   198→      );
   199→    };
   200→    update();
   201→    const interval = setInterval(update, 1000);
   202→    return () => clearInterval(interval);
   203→  }, []);
   204→  return time;
   205→}

   437→// ─── Dashboard View (Real Metrics) ──────────────────────────────
   438→function DashboardView() {
   439→  const [stats, setStats] = useState<KnowledgeStats | null>(null);
   440→  const [kwData, setKwData] = useState<{ totalDatasets: number; totalKeywords: number } | null>(null);
   441→  const [skillsCount, setSkillsCount] = useState<number>(0);
   442→
   443→  useEffect(() => {
   444→    fetch('/api/knowledge-stats').then(r => r.json()).then(setStats).catch(() => {});
   445→    fetch('/api/keywords').then(r => r.json()).then(d => setKwData({ totalDatasets: d.totalDatasets, totalKeywords: d.totalKeywords })).catch(() => {});
   446→    fetch('/api/skills').then(r => r.json()).then(d => setSkillsCount(d.total || 0)).catch(() => {});
   447→  }, []);
   448→
   449→  const files = stats?.totalFiles || 0;
   450→  const categories = stats?.totalCategories || 0;
   451→  const keywords = kwData?.totalKeywords || 0;
   452→  const skills = skillsCount;
   453→
   454→  const METRICS = [
   455→    { label: 'Knowledge Files', value: files, icon: Database, color: '#2563eb', glowClass: 'glow-cyan' },
   456→    { label: 'Categories', value: categories, icon: FolderOpen, color: '#7c3aed', glowClass: 'glow-purple' },
   457→    { label: 'Keywords', value: keywords, icon: Tag, color: '#059669', glowClass: 'glow-green' },
   458→    { label: 'SEO Skills', value: skills, icon: Wrench, color: '#d97706', glowClass: 'glow-amber' },
   459→  ];
   460→
   461→  const { counts, start } = useMetricCounts(METRICS.map((m) => m.value));
   462→  useEffect(() => { const t = setTimeout(start, 200); return () => clearTimeout(t); }, [start]);
   463→
   464→  const topCategories = (stats?.categories || [])
