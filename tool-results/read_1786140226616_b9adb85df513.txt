     4→ * "Open Claw" = self-sufficient, zero-cost, multi-provider email system.

     7→ *   OmniKey — penalty tracker with decay, round-robin, skip-set on failure

   242→  const domains = await mtGetDomains();
   243→  const domain = domains[0] || 'web-library.net';
   244→  const localPart = `eli-claw-${Date.now().toString(36)}-${Math.random().toString(36).slice(2, 7)}`;
   245→  const address = `${localPart}@${domain}`;
   246→  const password = `Claw${Date.now().toString(36)}!X${Math.random().toString(36).slice(2, 8)}`;
   247→
   248→  // Create account
   249→  const accRes = await fetch(`${MT_BASE}/accounts`, {
   250→    method: 'POST',
   251→    headers: { 'Content-Type': 'application/json' },
   252→    body: JSON.stringify({ address, password }),
   253→  });
   254→  if (!accRes.ok) throw new Error(`mail.tm create account failed: ${accRes.status}`);
   255→  const accData = await accRes.json();
   256→
   257→  // Get JWT
   258→  const tokRes = await fetch(`${MT_BASE}/token`, {
   259→    method: 'POST',
   260→    headers: { 'Content-Type': 'application/json' },
   261→    body: JSON.stringify({ address, password }),
   262→  });
   263→  if (!tokRes.ok) throw new Error(`mail.tm get token failed: ${tokRes.status}`);
   264→  const tokData = await tokRes.json();
   265→
   266→  return {
   267→    id: `mt-${accData.id}`,
   268→    email: address,
   269→    provider: 'mailtm',
   270→    expiresAt: Date.now() + DEFAULT_CONFIG.inboxTtlMs,
   271→    createdAt: Date.now(),
   272→    sessionData: { token: tokData.token, accountId: accData.id, password },
   273→    emailCount: 0,

   351→const KEY_PATTERNS: Array<{ name: string; pattern: RegExp; envVar: string }> = [
   352→  { name: 'gemini', pattern: /AIza[0-9A-Za-z_-]{35}/, envVar: 'GEMINI_API_KEY' },
   353→  { name: 'openai', pattern: /sk-[a-zA-Z0-9]{20,}/, envVar: 'OPENAI_API_KEY' },
   354→  { name: 'anthropic', pattern: /sk-ant-[a-zA-Z0-9-]{20,}/, envVar: 'ANTHROPIC_API_KEY' },
   355→  { name: 'google-generic', pattern: /AQ\.[a-zA-Z0-9_-]{30,}/, envVar: 'GEMINI_API_KEY' },
   356→];
   357→
   358→function stripHtml(html: string): string {
   359→  return html
   360→    .replace(/<pre[^>]*>/gi, '')
   361→    .replace(/<[^>]+>/g, ' ')
   362→    .replace(/&amp;/g, '&')
   363→    .replace(/&lt;/g, '<')
   364→    .replace(/&gt;/g, '>')
   365→    .replace(/&quot;/g, '"')
   366→    .replace(/&#039;/g, "'")
   367→    .replace(/\s+/g, ' ')
   368→    .trim();
   369→}
   370→
   371→function extractKeysFromEmail(email: ClawEmail): Array<{ service: string; key: string; envVar: string }> {
   372→  const plainText = `${email.subject} ${email.bodyText} ${stripHtml(email.bodyHtml)}`;
   373→  const results: Array<{ service: string; key: string; envVar: string }> = [];
   374→
   375→  for (const kp of KEY_PATTERNS) {
   376→    const matches = plainText.match(new RegExp(kp.pattern.source, 'g'));
   377→    if (matches) {
   378→      for (const match of matches) {
   379→        results.push({ service: kp.name, key: match, envVar: kp.envVar });
   380→      }
   381→    }
   382→  }
   383→
   384→  return results;
   385→}

   389→export class OpenClaw {
   390→  private config: ClawConfig;
   391→  private inboxes: ClawInbox[] = [];
   392→  private stats: ClawState['providerStats'];
   393→  private totalGenerated = 0;
   394→  private totalEmailsRead = 0;
   395→  private totalKeysExtracted = 0;
   396→  private lastKeyExtracted: string | null = null;
   397→  private pollTimer: ReturnType<typeof setInterval> | null = null;
   398→  private keyCallback: ((service: string, key: string, envVar: string, pendingId: string) => void) | null = null;
   399→  private providerHealth: Record<ProviderName, ProviderHealth>;
   400→  private roundRobinIdx = 0;
   401→  private pendingKeys: PendingKey[] = [];
   402→  private autoApproveKeys = false; // Tier 1: default OFF — keys require manual approval
   403→
   404→  /**
   405→   * Enable/disable auto-approval of extracted keys.
   406→   * When OFF (default): keys go to pending queue, require POST /api/omni?action=approve
   407→   * When ON: keys are validated and auto-injected (legacy behavior)
   408→   */
   409→  setAutoApprove(enabled: boolean) {
   410→    this.autoApproveKeys = enabled;
   411→    console.log(`[CLAW] Auto-approve ${enabled ? 'ENABLED' : 'DISABLED'} — keys will ${enabled ? 'auto-inject' : 'require approval'}`);
   412→  }
   413→
   414→  constructor(config: Partial<ClawConfig> = {}) {
   415→    this.config = { ...DEFAULT_CONFIG, ...config };
   416→    this.stats = {
   417→      guerrilla: { generated: 0, errors: 0, emailsRead: 0 },
   418→      mailtm: { generated: 0, errors: 0, emailsRead: 0 },
   419→      openinbox: { generated: 0, errors: 0, emailsRead: 0 },
   420→    };
   421→    this.providerHealth = {
   422→      guerrilla: { status: 'ok', latencyMs: 0, lastProbeAt: 0, consecutiveFailures: 0, penalty: 0, tier: 0 },
   423→      mailtm: { status: 'ok', latencyMs: 0, lastProbeAt: 0, consecutiveFailures: 0, penalty: 0, tier: 0 },
   424→      openinbox: { status: 'ok', latencyMs: 0, lastProbeAt: 0, consecutiveFailures: 0, penalty: 0, tier: 1 },
   425→    };
   426→  }
   427→
   428→  // ─── Two-Phase Provider Selection (Agent-Reach + OmniKey) ──────
   429→  // Phase 1: Probe ALL providers in parallel
   430→  // Phase 2: Select best by ok → penalty asc → latency asc, round-robin tiebreak
   431→
   432→  async probeAllProviders(): Promise<void> {
   433→    const [gm, mt, oi] = await Promise.all([probeGuerrilla(), probeMailTm(), probeOpenInbox()]);
   434→
   435→    // Penalty decay (OmniKey: -1 per 2 min)
   436→    for (const [name, fresh] of [['guerrilla', gm], ['mailtm', mt], ['openinbox', oi]] as const) {
   437→      const prev = this.providerHealth[name[0] as ProviderName];
   438→      const minsSinceProbe = prev.lastProbeAt > 0 ? (Date.now() - prev.lastProbeAt) / 120_000 : 0;
   439→      const decay = Math.floor(minsSinceProbe);
   440→      const decayedPenalty = Math.max(0, prev.penalty - decay);
   441→
   442→      if (fresh.status !== 'ok') {
   443→        fresh.consecutiveFailures = prev.consecutiveFailures + 1;
   444→        fresh.penalty = Math.min(decayedPenalty + 3, 10);
   445→      } else {
   446→        fresh.consecutiveFailures = 0;
   447→        fresh.penalty = Math.max(decayedPenalty - 1, 0);
   448→      }
   449→    }
   450→
   451→    this.providerHealth = { guerrilla: gm, mailtm: mt, openinbox: oi };
   452→    console.log(`[CLAW] Probed:`, Object.entries(this.providerHealth).map(
   453→      ([n, h]) => `${n}=${h.status}(${h.latencyMs}ms,p${h.penalty})`
   454→    ).join(' | '));
   455→  }
   456→
   457→  async getBestProvider(skipProviders?: Set<string>): Promise<ProviderName> {
   458→    const stale = Object.values(this.providerHealth).every(h => Date.now() - h.lastProbeAt > 120_000);
   459→    if (stale) await this.probeAllProviders();
   460→
   461→    // OmniMail: provider resolution chain — filter by skip-set, then sort
   462→    const candidates = Object.entries(this.providerHealth)
   463→      .filter(([name, h]) => h.status === 'ok' && h.consecutiveFailures < 3)
   464→      .filter(([name]) => !skipProviders?.has(name))
   465→      .sort(([, a], [, b]) => a.penalty - b.penalty || a.latencyMs - b.latencyMs);
   466→
   467→    if (candidates.length > 0) {
   468→      const bestPenalty = candidates[0][1].penalty;
   469→      const topTier = candidates.filter(([, h]) => h.penalty === bestPenalty);
   470→      const idx = this.roundRobinIdx % topTier.length;
   471→      this.roundRobinIdx++;
   472→      return topTier[idx][0] as ProviderName;
   473→    }
   474→
   475→    // OmniMail: fallback — ignore skip-set, use lowest consecutive failures
   476→    if (skipProviders?.size) {
   477→      console.warn(`[CLAW] All providers in skip-set, falling back to best available`);
   478→      return this.getBestProvider();
   479→    }
   480→
   481→    const fallback = Object.entries(this.providerHealth)
   482→      .sort(([, a], [, b]) => a.consecutiveFailures - b.consecutiveFailures);
   483→    return (fallback[0]?.[0] || 'guerrilla') as ProviderName;
   484→  }
   485→
   486→  /**
   487→   * Record provider success/failure for penalty scoring (OmniKey pattern)
   488→   */
   489→  recordProviderResult(provider: ProviderName, success: boolean) {
   490→    const h = this.providerHealth[provider];
   491→    if (!h) return;
   492→    if (success) {
   493→      h.penalty = Math.max(0, h.penalty - 1);
   494→      h.consecutiveFailures = 0;
   495→    } else {
   496→      h.penalty = Math.min(h.penalty + 3, 10);
   497→      h.consecutiveFailures++;
   498→    }
   499→  }

   501→  /**

   508→  /**
   509→   * Generate a new inbox. Two-phase selection (Agent-Reach):
   510→   * Probe all providers → select best by penalty + latency + round-robin
   511→   */
   512→  async generate(provider?: ProviderName): Promise<ClawInbox> {
   513→    const targetProvider = provider || await this.getBestProvider();
   514→
   515→    try {
   516→      let inbox: ClawInbox;
   517→      switch (targetProvider) {
   518→        case 'guerrilla': inbox = await gmCreate(); break;
   519→        case 'mailtm':   inbox = await mtCreate(); break;
   520→        case 'openinbox': inbox = await oiCreate(); break;
   521→      }
   522→
   523→      this.recordProviderResult(targetProvider, true);
   524→      this.inboxes.push(inbox);
   525→      this.totalGenerated++;
   526→      this.stats[targetProvider].generated++;
   527→
   528→      // Trim pool
   529→      if (this.inboxes.length > this.config.maxInboxes) {
   530→        this.inboxes = this.inboxes.slice(-this.config.maxInboxes);
   531→      }
   532→
   533→      console.log(`[CLAW] Generated ${targetProvider} inbox: ${inbox.email} (expires in ${Math.round((inbox.expiresAt - Date.now()) / 60000)}min)`);
   534→      return inbox;
   535→    } catch (err) {
   536→      this.recordProviderResult(targetProvider, false);
   537→      this.stats[targetProvider].errors++;
   538→      // OmniMail: classify error for upstream handling
   539→      const errMsg = (err as Error).message || '';
   540→      const retryable = [429, 408, 500, 502, 503, 504].some(c => errMsg.includes(String(c)))
   541→        || /timeout|econnrefused|econnreset/i.test(errMsg);
   542→      console.warn(`[CLAW] ${targetProvider} generation failed (${retryable ? 'retryable' : 'non-retryable'}):`, errMsg);
   543→      (err as any)._clawRetryable = retryable;
   544→      throw err;
   545→    }
   546→  }

   548→  /**
   549→   * Check a specific inbox for new emails
   550→   */
   551→  async checkInbox(inbox: ClawInbox): Promise<ClawEmail[]> {
   552→    // Skip expired
   553→    if (Date.now() > inbox.expiresAt) {
   554→      inbox.status = 'expired';
   555→      return [];
   556→    }
   557→
   558→    inbox.status = 'polling';
   559→
   560→    try {
   561→      let emails: ClawEmail[];
   562→      switch (inbox.provider) {
   563→        case 'guerrilla': emails = await gmCheck(inbox); break;
   564→        case 'mailtm':   emails = await mtCheck(inbox); break;
   565→        case 'openinbox': emails = await oiCheck(inbox); break;
   566→        default: return [];
   567→      }
   568→
   569→      inbox.emailCount = emails.length;
   570→      this.totalEmailsRead += emails.length;
   571→      this.stats[inbox.provider].emailsRead += emails.length;
   572→
   573→      // Auto-extract keys from new emails — Tier 1: pending queue + validation
   574→      for (const email of emails) {
   575→        const keys = extractKeysFromEmail(email);
   576→        for (const k of keys) {
   577→          console.log(`[CLAW] KEY EXTRACTED [${k.service}]: ${k.key.slice(0, 12)}... from ${inbox.email}`);
   578→          this.totalKeysExtracted++;
   579→          this.lastKeyExtracted = k.key;
   580→
   581→          const pendingId = `pending-${Date.now()}-${Math.random().toString(36).slice(2, 6)}`;
   582→          const pending: PendingKey = {
   583→            id: pendingId,
   584→            service: k.service,
   585→            key: k.key,
   586→            envVar: k.envVar,
   587→            extractedAt: Date.now(),
   588→            inboxId: inbox.id,
   589→            inboxEmail: inbox.email,
   590→            status: 'pending',
   591→          };
   592→          this.pendingKeys.push(pending);
   593→          if (this.pendingKeys.length > MAX_PENDING_KEYS) this.pendingKeys = this.pendingKeys.slice(-MAX_PENDING_KEYS);
   594→
   595→          audit('key.extracted', `${k.service} key from ${inbox.email}`, {
   596→            pendingId, keyPreview: k.key.slice(0, 12) + '...', inboxId: inbox.id,
   597→          });
   598→
   599→          if (this.autoApproveKeys) {
   600→            // Auto-approve: validate then inject
   601→            await this.validateAndInject(pending);
   602→          }
   603→
   604→          // Notify callback (with pendingId so caller can approve/reject)
   605→          this.keyCallback?.(k.service, k.key, k.envVar, pendingId);
   606→        }
   607→      }
   608→
   609→      return emails;
   610→    } catch (err) {
   611→      inbox.status = 'error';
   612→      this.stats[inbox.provider].errors++;
   613→      console.warn(`[CLAW] Check error [${inbox.provider}]:`, (err as Error).message);
   614→      return [];
   615→    }
   616→  }

   634→  /**
   635→   * Poll ALL active inboxes for new emails
   636→   */
   637→  async pollAll(): Promise<Map<string, ClawEmail[]>> {
   638→    const results = new Map<string, ClawEmail[]>();
   639→    const now = Date.now();
   640→
   641→    for (const inbox of this.inboxes) {
   642→      if (inbox.status === 'expired' || now > inbox.expiresAt) {
   643→        inbox.status = 'expired';
   644→        continue;
   645→      }
   646→      const emails = await this.checkInbox(inbox);
   647→      if (emails.length > 0) {
   648→        results.set(inbox.email, emails);
   649→      }
   650→    }
   651→
   652→    return results;
   653→  }

   655→  /**
   656→   * Start background polling loop
   657→   */
   658→  startPolling() {
   659→    if (this.pollTimer) return;
   660→    console.log(`[CLAW] Polling started (every ${this.config.pollIntervalMs / 1000}s, ${this.inboxes.length} inboxes)`);
   661→    this.pollTimer = setInterval(() => {
   662→      this.pollAll().then(results => {
   663→        if (results.size > 0) {
   664→          console.log(`[CLAW] Poll: ${results.size} inbox(es) have new emails`);
   665→        }
   666→      }).catch(err => {
   667→        console.error('[CLAW] Poll error:', (err as Error).message);
   668→      });
   669→    }, this.config.pollIntervalMs);
   670→  }
   671→
   672→  stopPolling() {
   673→    if (this.pollTimer) {
   674→      clearInterval(this.pollTimer);
   675→      this.pollTimer = null;
   676→    }
   677→  }

   679→  /**
   680→   * Generate + immediately start polling a new inbox
   681→   */
   682→  async spawn(provider?: 'guerrilla' | 'mailtm' | 'openinbox'): Promise<ClawInbox> {
   683→    const inbox = await this.generate(provider);
   684→    // If this is the first inbox, start the poll loop
   685→    if (!this.pollTimer) {
   686→      this.startPolling();

   688→    return inbox;

   728→  /**
   729→   * Validate a key by making a test API call, then inject into process.env.
   730→   * Returns true if valid, false if rejected.
   731→   */
   732→  async validateAndInject(pending: PendingKey): Promise<boolean> {
   733→    try {
   734→      // Quick format check first (cheaper than API call)
   735→      const svc = SERVICES_MAP[pending.service];
   736→      if (svc?.pattern && !svc.pattern.test(pending.key)) {
   737→        pending.status = 'rejected';
   738→        pending.validationError = 'Format mismatch';
   739→        pending.validatedAt = Date.now();
   740→        audit('key.rejected', `${pending.service} key failed format check`, { pendingId: pending.id });
   741→        return false;
   742→      }
   743→
   744→      // For Gemini keys, make a test call
   745→      if (pending.service === 'gemini' || pending.envVar === 'GEMINI_API_KEY') {
   746→        try {
   747→          const { GoogleGenerativeAI } = await import('@google/generative-ai');
   748→          const genAI = new GoogleGenerativeAI(pending.key);
   749→          const model = genAI.getGenerativeModel({ model: 'gemini-2.0-flash' });
   750→          const result = await model.generateContent({
   751→            contents: [{ role: 'user', parts: [{ text: 'Say OK' }] }],
   752→          });
   753→          const text = result?.response?.text();
   754→          if (!text || text.length === 0) throw new Error('Empty response');
   755→        } catch (err) {
   756→          pending.status = 'rejected';
   757→          pending.validationError = (err as Error).message?.slice(0, 200);
   758→          pending.validatedAt = Date.now();
   759→          audit('key.rejected', `${pending.service} key failed validation: ${(err as Error).message?.slice(0, 100)}`, {
   760→            pendingId: pending.id,
   761→          });
   762→          return false;
   763→        }
   764→      }
   765→
   766→      // Passed validation — inject
   767→      pending.status = 'approved';
   768→      pending.validatedAt = Date.now();
   769→      process.env[pending.envVar] = pending.key;
   770→      audit('key.approved', `${pending.service} key validated and injected`, { pendingId: pending.id });
   771→      return true;
   772→    } catch (err) {
   773→      pending.status = 'rejected';
   774→      pending.validationError = (err as Error).message;
   775→      pending.validatedAt = Date.now();
   776→      audit('key.rejected', `${pending.service} key validation error: ${(err as Error).message?.slice(0, 100)}`, {
   777→        pendingId: pending.id,
   778→      });
   779→      return false;
   780→    }
   781→  }

   790→  /** Manually reject a pending key */
   791→  rejectPendingKey(pendingId: string): boolean {

   793→    if (!pending) return false;

   796→    pending.validationError = 'Manual rejection';
   797→    audit('key.rejected', `${pending.service} key manually rejected`, { pendingId });
   798→    return true;
   799→  }
