/** Strip matching surrounding quotes, else drop a trailing ` #` inline comment. */
function unquoteScalar(s: string): string {
  if (
    s.length >= 2 &&
    ((s[0] === '"' && s[s.length - 1] === '"') ||
      (s[0] === "'" && s[s.length - 1] === "'"))
  ) {
    return s.slice(1, -1);
