   501→   701→        If no target agent is specified (or the specified one is
   502→   702→        unknown), this method sends the sub-task description to the
   503→   703→        Kimi model and asks it to select the best agent.
   504→   704→
   505→   705→        Parameters
   506→   706→        ----------
   507→   707→        sub_task:
   508→   708→            The sub-task to route.
   509→   709→
   510→   710→        Returns
   511→   711→        -------
   512→   712→        str
   513→   713→            The agent name to execute this sub-task.
   514→   714→
   515→   715→        Raises
   516→   716→        ------
   517→   717→        ValueError
   518→   718→            If no valid agent can be determined.
   519→   719→        """
   520→   720→        known_agents = set(self._manifests.keys()) | set(self._agent_instances.keys())
   521→   721→
   522→   722→        # Fast path: agent is already specified and known
   523→   723→        if sub_task.target_agent and sub_task.target_agent in known_agents:
   524→   724→            self._log.debug(
   525→   725→                "route_direct",
   526→   726→                sub_task_id=sub_task.id,
   527→   727→                agent=sub_task.target_agent,
   528→   728→            )
   529→   729→            return sub_task.target_agent
   530→   730→
   531→   731→        # Slow path: ask the model to pick an agent
   532→   732→        if sub_task.target_agent and sub_task.target_agent not in known_agents:
   533→   733→            self._log.warning(
   534→   734→                "route_unknown_agent_fallback_to_model",
   535→   735→                sub_task_id=sub_task.id,
   536→   736→                unknown_agent=sub_task.target_agent,
   537→   737→            )
   538→   738→
   539→   739→        self._log.debug(
   540→   740→            "route_model_fallback",
   541→   741→            sub_task_id=sub_task.id,
   542→   742→        )
   543→   743→
   544→   744→        system = ROUTE_SYSTEM_PROMPT.format(
   545→   745→            agent_registry=self._agent_registry,
   546→   746→        )
   547→   747→
   548→   748→        response_text = await self._call_model(
   549→   749→            system_prompt=system,
   550→   750→            user_prompt=sub_task.description,
   551→   751→            temperature=0.0,
   552→   752→            max_tokens=50,
   553→   753→        )
   554→   754→
   555→   755→        # The model should return just the agent name as a plain string.
   556→   756→        # Strip whitespace, quotes, and code fences.
   557→   757→        agent_name = self._strip_code_fences(response_text).strip().strip('"').strip("'")
   558→   758→
   559→   759→        if agent_name not in known_agents:
   560→   760→            self._log.error(
   561→   761→                "route_model_returned_unknown_agent",
   562→   762→                sub_task_id=sub_task.id,
   563→   763→                model_response=response_text,
   564→   764→                known_agents=sorted(known_agents),
   565→   765→            )
   566→   766→            raise ValueError(
   567→   767→                f"Model returned unknown agent '{agent_name}' for sub-task "
   568→   768→                f"'{sub_task.id}'. Known agents: {sorted(known_agents)}"
   569→   769→            )
   570→   770→
   571→   771→        self._log.info(
   572→   772→            "route_complete",
   573→   773→            sub_task_id=sub_task.id,
   574→   774→            agent=agent_name,
   575→   775→        )
   576→   776→
   577→   777→        return agent_name
   578→   778→
   579→   779→    # -- DAG execution ------------------------------------------------------
   580→   780→
   581→   781→    async def execute_dag(self, dag: TaskDAG) -> Dict[str, Any]:
   582→   782→        """Execute a task DAG by routing and running each sub-task.
   583→   783→
   584→   784→        Execution follows topological order to respect dependencies.
   585→   785→        Completed results are made available to subsequent sub-tasks
   586→   786→        via the ``_completed_results`` dict.
   587→   787→
   588→   788→        Error handling:
   589→   789→
   590→   790→        - **PolicyViolationError**: Logs the violation, attempts to
   591→   791→          modify the request based on ``suggested_resolution``, retries
   592→   792→          once with the modified payload.
   593→   793→        - **EscalationRequiredError**: Logs the escalation, checks if
   594→   794→          re-routing to a different agent can resolve it.  If not,
   595→   795→          collects all completed results and returns them with the
   596→   796→          escalation noted.
   597→   797→
   598→   798→        Parameters
   599→   799→        ----------
   600→   800→        dag:
   601→   801→            The task DAG to execute.
   602→   802→
   603→   803→        Returns
   604→   804→        -------
   605→   805→        dict
   606→   806→            Mapping of ``sub_task_id`` → result dict.  Failed sub-tasks
   607→   807→            have an ``"error"`` key instead of normal results.
   608→   808→        """
   609→   809→        from agents.base import EscalationRequiredError, PolicyViolationError
   610→   810→
   611→   811→        self._log.info(
   612→   812→            "execute_dag_start",
   613→   813→            sub_task_count=len(dag.sub_tasks),
   614→   814→        )
   615→   815→
   616→   816→        # Topologically sort the sub-tasks
   617→   817→        ordered = dag.topological_order()
   618→   818→
   619→   819→        # Result store: sub_task_id → result dict
   620→   820→        results: Dict[str, Any] = {}
   621→   821→        # Completed results available for dependent sub-tasks
   622→   822→        completed_results: Dict[str, Any] = {}
   623→   823→        escalations: List[Dict[str, Any]] = []
   624→   824→
   625→   825→        for sub_task in ordered:
   626→   826→            self._log.info(
   627→   827→                "executing_sub_task",
   628→   828→                sub_task_id=sub_task.id,
   629→   829→                target_agent=sub_task.target_agent,
   630→   830→            )
   631→   831→
   632→   832→            try:
   633→   833→                # Route the sub-task to an agent
   634→   834→                agent_name = await self.route(sub_task)
   635→   835→
   636→   836→                # Get the agent instance
   637→   837→                agent = self._agent_instances.get(agent_name)
   638→   838→                if agent is None:
   639→   839→                    raise ValueError(
   640→   840→                        f"No registered agent instance for '{agent_name}'. "
   641→   841→                        f"Call orchestrator.register_agent('{agent_name}', instance) first."
   642→   842→                    )
   643→   843→
   644→   844→                # Build the task payload, injecting results from dependencies
   645→   845→                task_payload: Dict[str, Any] = {
   646→   846→                    "description": sub_task.description,
   647→   847→                    "sub_task_id": sub_task.id,
   648→   848→                    "dependency_results": {
   649→   849→                        dep_id: completed_results.get(dep_id)
   650→   850→                        for dep_id in sub_task.dependencies
   651→   851→                    },
   652→   852→                }
   653→   853→
   654→   854→                # Execute with policy violation retry
   655→   855→                result = await self._execute_with_retry(
   656→   856→                    agent=agent,
   657→   857→                    sub_task=sub_task,
   658→   858→                    task_payload=task_payload,
   659→   859→                )
   660→   860→
   661→   861→                results[sub_task.id] = result
   662→   862→                completed_results[sub_task.id] = result
   663→   863→
   664→   864→                self._log.info(
   665→   865→                    "sub_task_complete",
   666→   866→                    sub_task_id=sub_task.id,
   667→   867→                    agent=agent_name,
   668→   868→                )
   669→   869→
   670→   870→            except PolicyViolationError as exc:
   671→   871→                # Log and attempt retry with suggested resolution
   672→   872→                self._log.warning(
   673→   873→                    "sub_task_policy_violation",
   674→   874→                    sub_task_id=sub_task.id,
   675→   875→                    agent=sub_task.target_agent,
   676→   876→                    tier=exc.tier,
   677→   877→                    explanation=exc.explanation,
   678→   878→                )
   679→   879→
   680→   880→                # Try to apply the suggested resolution and retry once
   681→   881→                try:
   682→   882→                    modified_payload = self._apply_suggested_resolution(
   683→   883→                        task_payload,
   684→   884→                        exc.suggested_resolution,
   685→   885→                    )
   686→   886→                    agent_name = await self.route(sub_task)
   687→   887→                    agent = self._agent_instances.get(agent_name)
   688→   888→                    if agent is not None:
   689→   889→                        result = await agent.execute(modified_payload)
   690→   890→                        results[sub_task.id] = result
   691→   891→                        completed_results[sub_task.id] = result
   692→   892→                        self._log.info(
   693→   893→                            "sub_task_retry_success",
   694→   894→                            sub_task_id=sub_task.id,
   695→   895→                            agent=agent_name,
   696→   896→                        )
   697→   897→                        continue
   698→   898→                except Exception as retry_exc:
   699→   899→                    self._log.error(
   700→   900→                        "sub_task_retry_failed",
   701→   901→                        sub_task_id=sub_task.id,
   702→   902→                        error=str(retry_exc),
   703→   903→                    )
   704→   904→
   705→   905→                # Retry failed — record the error
   706→   906→                results[sub_task.id] = {
   707→   907→                    "error": "policy_violation",
   708→   908→                    "agent": sub_task.target_agent,
   709→   909→                    "tier": exc.tier,
   710→   910→                    "explanation": exc.explanation,
   711→   911→                    "suggested_resolution": exc.suggested_resolution,
   712→   912→                }
   713→   913→
   714→   914→            except EscalationRequiredError as exc:
   715→   915→                self._log.warning(
   716→   916→                    "sub_task_escalation",
   717→   917→                    sub_task_id=sub_task.id,
   718→   918→                    agent=exc.agent_id,
   719→   919→                    reason=exc.reason,
   720→   920→                )
   721→   921→
   722→   922→                escalations.append({
   723→   923→                    "sub_task_id": sub_task.id,
   724→   924→                    "agent_id": exc.agent_id,
   725→   925→                    "reason": exc.reason,
   726→   926→                    "context": exc.context,
   727→   927→                })
   728→   928→
   729→   929→                # Ask the escalation handler what to do
   730→   930→                decision = await self.handle_escalation({
   731→   931→                    "sub_task_id": sub_task.id,
   732→   932→                    "agent_id": exc.agent_id,
   733→   933→                    "reason": exc.reason,
   734→   934→                    "context": exc.context,
   735→   935→                    "completed_results": dict(completed_results),
   736→   936→                    "remaining_sub_tasks": [
   737→   937→                        st.id for st in ordered
   738→   938→                        if st.id not in completed_results and st.id != sub_task.id
   739→   939→                    ],
   740→   940→                })
   741→   941→
   742→   942→                if decision.action == "reroute" and decision.target_agent:
   743→   943→                    # Try a different agent
   744→   944→                    alt_agent = self._agent_instances.get(decision.target_agent)
   745→   945→                    if alt_agent is not None:
   746→   946→                        self._log.info(
   747→   947→                            "sub_task_rerouted",
   748→   948→                            sub_task_id=sub_task.id,
   749→   949→                            new_agent=decision.target_agent,
   750→   950→                            rationale=decision.rationale,
   751→   951→                        )
   752→   952→                        try:
   753→   953→                            result = await alt_agent.execute(task_payload)
   754→   954→                            results[sub_task.id] = result
   755→   955→                            completed_results[sub_task.id] = result
   756→   956→                            continue
   757→   957→                        except Exception as reroute_exc:
   758→   958→                            self._log.error(
   759→   959→                                "sub_task_reroute_failed",
   760→   960→                                sub_task_id=sub_task.id,
   761→   961→                                error=str(reroute_exc),
   762→   962→                            )
   763→   963→
   764→   964→                if decision.action == "compose":
   765→   965→                    # We have enough results — stop executing and compose
   766→   966→                    self._log.info(
   767→   967→                        "execute_dag_composing_early",
   768→   968→                        completed=len(completed_results),
   769→   969→                        total=len(ordered),
   770→   970→                        rationale=decision.rationale,
   771→   971→                    )
   772→   972→                    results["_orchestrator_note"] = (
   773→   973→                        f"DAG execution stopped early due to escalation. "
   774→   974→                        f"Composing with {len(completed_results)}/{len(ordered)} sub-tasks complete. "
   775→   975→                        f"Rationale: {decision.rationale}"
   776→   976→                    )
   777→   977→                    results["_escalations"] = escalations
   778→   978→                    break
   779→   979→
   780→   980→                if decision.action == "retry" and decision.modified_request:
   781→   981→                    agent = self._agent_instances.get(sub_task.target_agent or "")
   782→   982→                    if agent is not None:
   783→   983→                        try:
   784→   984→                            result = await agent.execute(decision.modified_request)
   785→   985→                            results[sub_task.id] = result
   786→   986→                            completed_results[sub_task.id] = result
   787→   987→                            continue
   788→   988→                        except Exception as retry_exc:
   789→   989→                            self._log.error(
   790→   990→                                "sub_task_escalation_retry_failed",
   791→   991→                                sub_task_id=sub_task.id,
   792→   992→                                error=str(retry_exc),
   793→   993→                            )
   794→   994→
   795→   995→                # escalate_to_human or no action — record and continue
   796→   996→                results[sub_task.id] = {
   797→   997→                    "error": "escalation_required",
   798→   998→                    "agent": exc.agent_id,
   799→   999→                    "reason": exc.reason,
   800→  1000→                    "decision": decision.action,
   801→  1001→                    "rationale": decision.rationale,
   802→  1002→                }
   803→  1003→
   804→  1004→            except Exception as exc:
   805→  1005→                self._log.error(
   806→  1006→                    "sub_task_unexpected_error",
   807→  1007→                    sub_task_id=sub_task.id,
   808→  1008→                    agent=sub_task.target_agent,
   809→  1009→                    error=str(exc),
   810→  1010→                    error_type=type(exc).__name__,
   811→  1011→                )
   812→  1012→                results[sub_task.id] = {
   813→  1013→                    "error": "unexpected_error",
   814→  1014→                    "agent": sub_task.target_agent,
   815→  1015→                    "error_type": type(exc).__name__,
   816→  1016→                    "message": str(exc),
   817→  1017→                }
   818→  1018→
   819→  1019→        # Attach escalations if any occurred
   820→  1020→        if escalations and "_escalations" not in results:
   821→  1021→            results["_escalations"] = escalations
   822→  1022→
   823→  1023→        self._log.info(
   824→  1024→            "execute_dag_complete",
   825→  1025→            total_sub_tasks=len(ordered),
   826→  1026→            successful=sum(1 for v in results.values() if not isinstance(v, dict) or "error" not in v),
   827→  1027→            failed=sum(1 for v in results.values() if isinstance(v, dict) and "error" in v),
   828→  1028→        )
   829→  1029→
   830→  1030→        return results
   831→  1031→
   832→  1032→    async def _execute_with_retry(
   833→  1033→        self,
   834→  1034→        agent: Any,
   835→  1035→        sub_task: SubTask,
   836→  1036→        task_payload: Dict[str, Any],
   837→  1037→    ) -> Dict[str, Any]:
   838→  1038→        """Execute a sub-task with a single retry on policy violation.
   839→  1039→
   840→  1040→        This is the inner execution helper used by ``execute_dag``.
   841→  1041→        The first policy violation triggers an automatic retry with
   842→  1042→        the suggested resolution applied.  Subsequent violations
   843→  1043→        are re-raised for the caller to handle.
   844→  1044→
   845→  1045→        Parameters
   846→  1046→        ----------
   847→  1047→        agent:
   848→  1048→            The agent instance to call ``execute`` on.
   849→  1049→        sub_task:
   850→  1050→            The sub-task being executed.
   851→  1051→        task_payload:
   852→  1052→            The task payload dict.
   853→  1053→
   854→  1054→        Returns
   855→  1055→        -------
   856→  1056→        dict
   857→  1057→            The agent's execution result.
   858→  1058→        """
   859→  1059→        from agents.base import PolicyViolationError
   860→  1060→
   861→  1061→        try:
   862→  1062→            return await agent.execute(task_payload)
   863→  1063→        except PolicyViolationError as exc:
   864→  1064→            self._log.info(
   865→  1065→                "auto_retry_on_policy_violation",
   866→  1066→                sub_task_id=sub_task.id,
   867→  1067→                tier=exc.tier,
   868→  1068→            )
   869→  1069→            modified = self._apply_suggested_resolution(
   870→  1070→                task_payload,
   871→  1071→                exc.suggested_resolution,
   872→  1072→            )
   873→  1073→            # Re-raise — let execute_dag handle the retry properly
   874→  1074→            # so it can update routing and result tracking
   875→  1075→            raise
   876→  1076→
   877→  1077→    @staticmethod
   878→  1078→    def _apply_suggested_resolution(
   879→  1079→        payload: Dict[str, Any],
   880→  1080→        suggested_resolution: str,
   881→  1081→    ) -> Dict[str, Any]:
   882→  1082→        """Attempt to apply the policy engine's suggested resolution.
   883→  1083→
   884→  1084→        This is a best-effort heuristic.  The suggested resolution is
   885→  1085→        a natural-language string from the policy engine; we look for
   886→  1086→        patterns like "use X instead of Y" and apply the substitution.
   887→  1087→
   888→  1088→        Parameters
   889→  1089→        ----------
   890→  1090→        payload:
   891→  1091→            The original task payload.
   892→  1092→        suggested_resolution:
   893→  1093→            The ``suggested_resolution`` from a ``PolicyViolationDetail``.
   894→  1094→
   895→  1095→        Returns
   896→  1096→        -------
   897→  1097→        dict
   898→  1098→            A modified copy of the payload.
   899→  1099→        """
   900→  1100→        modified = dict(payload)
   901→  1101→        modified["_policy_retry"] = True
   902→  1102→        modified["_original_resolution"] = suggested_resolution
   903→  1103→        return modified
   904→  1104→
   905→  1105→    # -- Synthesis ----------------------------------------------------------
   906→  1106→
   907→  1107→    async def synthesize(self, task: str, results: Dict[str, Any]) -> str:
   908→  1108→        """Synthesize sub-task results into a unified response.
   909→  1109→
   910→  1110→        Sends all sub-task results to the Kimi model along with the
   911→  1111→        original task description.  The model produces a single,
   912→  1112→        coherent response that addresses the user's request.
   913→  1113→
   914→  1114→        Parameters
   915→  1115→        ----------
   916→  1116→        task:
   917→  1117→            The user's original task description.
   918→  1118→        results:
   919→  1119→            Mapping of ``sub_task_id`` → result dict from ``execute_dag``.
   920→  1120→
   921→  1121→        Returns
   922→  1122→        -------
   923→  1123→        str
   924→  1124→            The synthesized response in markdown format.
   925→  1125→        """
   926→  1126→        self._log.info("synthesize_start")
   927→  1127→
   928→  1128→        # Build a structured summary of all results
   929→  1129→        results_summary_parts: list[str] = []
   930→  1130→        for sub_id, result in results.items():
   931→  1131→            if sub_id.startswith("_"):
   932→  1132→                # Orchestrator metadata, skip
   933→  1133→                continue
   934→  1134→            if isinstance(result, dict) and "error" in result:
   935→  1135→                results_summary_parts.append(
   936→  1136→                    f"### {sub_id}: FAILED\n"
   937→  1137→                    f"- Error: {result.get('error_type', result['error'])}\n"
   938→  1138→                    f"- Details: {result.get('message', result.get('reason', 'N/A'))}"
   939→  1139→                )
   940→  1140→            else:
   941→  1141→                # Truncate large results to keep within model context
   942→  1142→                result_str = json.dumps(result, default=str, indent=2)
   943→  1143→                if len(result_str) > 2000:
   944→  1144→                    result_str = result_str[:2000] + "\n... [truncated]"
   945→  1145→                results_summary_parts.append(
   946→  1146→                    f"### {sub_id}: SUCCESS\n````json\n{result_str}\n```"
   947→  1147→                )
   948→  1148→
   949→  1149→        # Include escalation notes if present
   950→  1150→        escalations = results.get("_escalations")
   951→  1151→        escalation_note = ""
   952→  1152→        if escalations:
   953→  1153→            escalation_lines = [
   954→  1154→                f"- Sub-task {e.get('sub_task_id')}: {e.get('reason')}"
   955→  1155→                for e in escalations
   956→  1156→            ]
   957→  1157→            escalation_note = (
   958→  1158→                "\n\n## Escalations That Occurred\n"
   959→  1159→                + "\n".join(escalation_lines)
   960→  1160→            )
   961→  1161→
   962→  1162→        orchestrator_note = results.get("_orchestrator_note", "")
   963→  1163→        note_section = ""
   964→  1164→        if orchestrator_note:
   965→  1165→            note_section = f"\n\n## Orchestrator Note\n{orchestrator_note}"
   966→  1166→
   967→  1167→        user_prompt = (
   968→  1168→            f"## Original Task\n{task}\n\n"
   969→  1169→            f"## Sub-Task Results\n\n"
   970→  1170→            f"{chr(10).join(results_summary_parts)}"
   971→  1171→            f"{escalation_note}"
   972→  1172→            f"{note_section}"
   973→  1173→        )
   974→  1174→
   975→  1175→        synthesis = await self._call_model(
   976→  1176→            system_prompt=SYNTHESIZE_SYSTEM_PROMPT,
   977→  1177→            user_prompt=user_prompt,
   978→  1178→            temperature=0.3,
   979→  1179→            max_tokens=4096,
   980→  1180→        )
   981→  1181→
   982→  1182→        self._log.info("synthesize_complete", response_length=len(synthesis))
   983→  1183→
   984→  1184→        return synthesis

   988→  1188→    async def handle_escalation(self, event: Dict[str, Any]) -> EscalationDecision:
   989→  1189→        """Decide how to handle an escalation event.
   990→  1190→
   991→  1191→        Sends the escalation context to the Kimi model and asks it
   992→  1192→        to choose one of four actions: retry, reroute, compose, or
   993→  1193→        escalate_to_human.
   994→  1194→
   995→  1195→        Parameters
   996→  1196→        ----------
   997→  1197→        event:
   998→  1198→            The escalation event dict with keys: ``sub_task_id``,
   999→  1199→            ``agent_id``, ``reason``, ``context``, and optionally
  1000→  1200→            ``completed_results`` and ``remaining_sub_tasks``.
  1001→  1201→
  1002→  1202→        Returns
  1003→  1203→        -------
  1004→  1204→        EscalationDecision
  1005→  1205→            The model's decision with action, rationale, and optional
  1006→  1206→            target agent or modified request.
  1007→  1207→        """
  1008→  1208→        self._log.info(
  1009→  1209→            "handle_escalation_start",
  1010→  1210→            sub_task_id=event.get("sub_task_id"),
  1011→  1211→            agent_id=event.get("agent_id"),
  1012→  1212→            reason=event.get("reason"),
  1013→  1213→        )
  1014→  1214→
  1015→  1215→        system = ESCALATION_SYSTEM_PROMPT.format(
  1016→  1216→            agent_registry=self._agent_registry,
  1017→  1217→        )
  1018→  1218→
  1019→  1219→        # Build a focused context for the model
  1020→  1220→        event_context = {
  1021→  1221→            "sub_task_id": event.get("sub_task_id"),
  1022→  1222→            "agent_id": event.get("agent_id"),
  1023→  1223→            "reason": event.get("reason"),
  1024→  1224→            "context": event.get("context", {}),
  1025→  1225→            "completed_sub_tasks": list(event.get("completed_results", {}).keys()),
  1026→  1226→            "remaining_sub_tasks": event.get("remaining_sub_tasks", []),
  1027→  1227→        }
  1028→  1228→
  1029→  1229→        response_text = await self._call_model(
  1030→  1230→            system_prompt=system,
  1031→  1231→            user_prompt=json.dumps(event_context, default=str, indent=2),
  1032→  1232→            temperature=0.1,
  1033→  1233→            max_tokens=500,
  1034→  1234→        )
  1035→  1235→
  1036→  1236→        # Parse the model's JSON decision
  1037→  1237→        cleaned = self._strip_code_fences(response_text)
  1038→  1238→
  1039→  1239→        try:
  1040→  1240→            decision_data = json.loads(cleaned)
  1041→  1241→        except json.JSONDecodeError as exc:
  1042→  1242→            self._log.error(
  1043→  1243→                "escalation_decision_parse_error",
  1044→  1244→                response_text=response_text[:500],
  1045→  1245→                error=str(exc),
  1046→  1246→            )
  1047→  1247→            # Default to escalate_to_human on parse failure
  1048→  1248→            return EscalationDecision(
  1049→  1249→                action="escalate_to_human",
  1050→  1250→                rationale=(
  1051→  1251→                    f"Failed to parse model escalation decision: {exc}. "
  1052→  1252→                    f"Defaulting to human escalation."
  1053→  1253→                ),
  1054→  1254→            )
  1055→  1255→
  1056→  1256→        decision = EscalationDecision(
  1057→  1257→            action=decision_data.get("action", "escalate_to_human"),
  1058→  1258→            rationale=decision_data.get("rationale", "No rationale provided."),
  1059→  1259→            target_agent=decision_data.get("target_agent"),
  1060→  1260→            modified_request=decision_data.get("modified_request"),
  1061→  1261→        )
  1062→  1262→
  1063→  1263→        # Validate the action
  1064→  1264→        valid_actions = {"retry", "reroute", "compose", "escalate_to_human"}
  1065→  1265→        if decision.action not in valid_actions:
  1066→  1266→            self._log.warning(
  1067→  1267→                "escalation_invalid_action",
  1068→  1268→                action=decision.action,
  1069→  1269→                valid_actions=sorted(valid_actions),
  1070→  1270→            )
  1071→  1271→            decision.action = "escalate_to_human"
  1072→  1272→            decision.rationale += " (Invalid action corrected to escalate_to_human)"
  1073→  1273→
  1074→  1274→        self._log.info(
  1075→  1275→            "escalation_decision_made",
  1076→  1276→            action=decision.action,
  1077→  1277→            rationale=decision.rationale,
  1078→  1278→            target_agent=decision.target_agent,
  1079→  1279→        )
  1080→  1280→
  1081→  1281→        return decision

  1084→  1284→
  1085→  1285→    async def run(self, task: str) -> Dict[str, Any]:
  1086→  1286→        """Execute the full orchestration pipeline.
  1087→  1287→
  1088→  1288→        The main entry point that orchestrates the complete flow:
  1089→  1289→
  1090→  1290→        1. **Decompose** the task into a sub-task DAG.
  1091→  1291→        2. **Execute** the DAG (with error handling and retries).
  1092→  1292→        3. **Synthesize** the results into a unified response.
  1093→  1293→
  1094→  1294→        Parameters
  1095→  1295→        ----------
  1096→  1296→        task:
  1097→  1297→            The user's task description in natural language.
  1098→  1298→
  1099→  1299→        Returns
  1100→  1300→        -------
  1101→  1301→        dict
  1102→  1302→            A result dictionary with keys:
  1103→  1303→
  1104→  1304→            - ``"synthesis"`` — The synthesized response string.
  1105→  1305→            - ``"sub_task_results"`` — Raw sub-task results dict.
  1106→  1306→            - ``"sub_task_count"`` — Total number of sub-tasks.
  1107→  1307→            - ``"successful_count"`` — Number of successful sub-tasks.
  1108→  1308→            - ``"escalations"`` — List of escalation events (if any).
  1109→  1309→            - ``"duration_seconds"`` — Total pipeline duration.
  1110→  1310→        """
  1111→  1311→        start_time = time.monotonic()
  1112→  1312→
  1113→  1313→        self._log.info("orchestration_run_start", task_length=len(task))
  1114→  1314→
  1115→  1315→        try:
  1116→  1316→            # Step 1: Decompose the task into a DAG
  1117→  1317→            dag = await self.decompose(task)
  1118→  1318→
  1119→  1319→            # Step 2: Execute the DAG
  1120→  1320→            sub_task_results = await self.execute_dag(dag)
  1121→  1321→
  1122→  1322→            # Step 3: Synthesize the results
  1123→  1323→            synthesis = await self.synthesize(task, sub_task_results)
  1124→  1324→
  1125→  1325→        except Exception as exc:
  1126→  1326→            self._log.error(
  1127→  1327→                "orchestration_run_error",
  1128→  1328→                error=str(exc),
  1129→  1329→                error_type=type(exc).__name__,
  1130→  1330→            )
  1131→  1331→            # Return a partial result with error information
  1132→  1332→            return {
  1133→  1333→                "synthesis": f"**Orchestration Error**: {exc}",
  1134→  1334→                "sub_task_results": {},
  1135→  1335→                "sub_task_count": 0,
  1136→  1336→                "successful_count": 0,
  1137→  1337→                "escalations": [],
  1138→  1338→                "duration_seconds": time.monotonic() - start_time,
  1139→  1339→                "error": str(exc),
  1140→  1340→                "error_type": type(exc).__name__,
  1141→  1341→            }
  1142→  1342→
  1143→  1343→        duration = time.monotonic() - start_time
  1144→  1344→
  1145→  1345→        # Compute summary stats
  1146→  1346→        successful = sum(
  1147→  1347→            1 for k, v in sub_task_results.items()
  1148→  1348→            if not k.startswith("_") and isinstance(v, dict) and "error" not in v
  1149→  1349→        )
  1150→  1350→        total = sum(1 for k in sub_task_results if not k.startswith("_"))
  1151→  1351→
  1152→  1352→        result: Dict[str, Any] = {
  1153→  1353→            "synthesis": synthesis,
  1154→  1354→            "sub_task_results": sub_task_results,
  1155→  1355→            "sub_task_count": total,
  1156→  1356→            "successful_count": successful,
  1157→  1357→            "escalations": sub_task_results.get("_escalations", []),
  1158→  1358→            "duration_seconds": round(duration, 3),
  1159→  1359→        }
  1160→  1360→
  1161→  1361→        self._log.info(
  1162→  1362→            "orchestration_run_complete",
  1163→  1363→            sub_task_count=total,
  1164→  1364→            successful=successful,
  1165→  1365→            duration_seconds=round(duration, 3),
  1166→  1366→        )
  1167→  1367→
  1168→  1368→        return result

  1172→  1372→    @staticmethod
  1173→  1373→    def _strip_code_fences(text: str) -> str:
  1174→  1374→        """Strip markdown code fences from a model response.
  1175→  1375→
  1176→  1376→        The Kimi model may wrap JSON in ``````json ... `````` blocks.
  1177→  1377→        This method removes those wrappers.
  1178→  1378→
  1179→  1379→        Parameters
  1180→  1380→        ----------
  1181→  1381→        text:
  1182→  1382→            The raw model response.
  1183→  1383→
  1184→  1384→        Returns
  1185→  1385→        -------
  1186→  1386→        str
  1187→  1387→            The text with code fences removed.
  1188→  1388→        """
  1189→  1389→        # Match ```json ... ``` or ``` ... ```
  1190→  1390→        pattern = r"^```(?:json)?\s*\n?(.*?)\n?\s*```$"
  1191→  1391→        match = re.match(pattern, text.strip(), re.DOTALL)
  1192→  1392→        if match:
  1193→  1393→            return match.group(1).strip()
  1194→  1394→        return text.strip()
