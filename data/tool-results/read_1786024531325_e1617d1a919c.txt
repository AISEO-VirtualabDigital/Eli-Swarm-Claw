    78→    78→def _parse_skill_md(skill_md_path: str) -> Dict[str, Any]:
    79→    79→    """Parse a SKILL.md manifest file into a structured dictionary.
    80→    80→
    81→    81→    Extracts the following top-level sections:
    82→    82→
    83→    83→    - **Identity** — ``name``, ``role``, ``domain``, ``version``
    84→    84→    - **Purpose** — the full purpose paragraph
    85→    85→    - **Knowledge Base Scope** — ``sources``, ``exclusions``, ``refresh_policy``
    86→    86→    - **Capabilities** — list of tool definitions
    87→    87→    - **Forbidden Actions** — list of forbidden action strings
    88→    88→    - **Input Schema** — parsed JSON schema dict
    89→    89→    - **Output Schema** — parsed JSON schema dict
    90→    90→    - **Constraints** — list of constraint strings
    91→    91→    - **IPC Policy** — ``allowed_tables``, ``allowed_endpoints``, ``resource_limits``
    92→    92→    - **Escalation Triggers** — list of trigger descriptions
    93→    93→
    94→    94→    Parameters
    95→    95→    ----------
    96→    96→    skill_md_path:
    97→    97→        Filesystem path to the agent's ``SKILL.md`` file.
    98→    98→
    99→    99→    Returns
   100→   100→    -------
   101→   101→    dict
   102→   102→        Structured representation of the manifest.
   103→   103→    """
   104→   104→    path = Path(skill_md_path)
   105→   105→    if not path.exists():
   106→   106→        raise FileNotFoundError(f"SKILL.md not found: {skill_md_path}")
   107→   107→
   108→   108→    text = path.read_text(encoding="utf-8")
   109→   109→    manifest: Dict[str, Any] = {"raw_text": text}
   110→   110→
   111→   111→    # --- Identity ---
   112→   112→    identity_match = re.search(r"## Identity\n(.*?)(?=\n## )", text, re.DOTALL)
   113→   113→    if identity_match:
   114→   114→        block = identity_match.group(1)
   115→   115→        manifest["identity"] = {
   116→   116→            "name": _extract_field(block, "Name"),
   117→   117→            "role": _extract_field(block, "Role"),
   118→   118→            "domain": _extract_field(block, "Domain"),
   119→   119→            "version": _extract_field(block, "Version"),
   120→   120→        }
   121→   121→
   122→   122→    # --- Purpose ---
   123→   123→    purpose_match = re.search(r"## Purpose\n(.*?)(?=\n## )", text, re.DOTALL)
   124→   124→    if purpose_match:
   125→   125→        manifest["purpose"] = purpose_match.group(1).strip()
   126→   126→
   127→   127→    # --- Knowledge Base Scope ---
   128→   128→    kb_match = re.search(r"## Knowledge Base Scope\n(.*?)(?=\n## )", text, re.DOTALL)
   129→   129→    if kb_match:
   130→   130→        block = kb_match.group(1)
   131→   131→        manifest["knowledge_base_scope"] = {
   132→   132→            "sources": _extract_field(block, "Sources"),
   133→   133→            "exclusions": _extract_field(block, "Exclusions"),
   134→   134→            "refresh_policy": _extract_field(block, "Refresh Policy"),
   135→   135→        }
   136→   136→
   137→   137→    # --- Capabilities ---
   138→   138→    caps_match = re.search(r"## Capabilities[^\n]*\n(.*?)(?=\n## )", text, re.DOTALL)
   139→   139→    if caps_match:
   140→   140→        manifest["capabilities"] = _extract_numbered_list(caps_match.group(1))
   141→   141→
   142→   142→    # --- Forbidden Actions ---
   143→   143→    forbidden_match = re.search(r"## Forbidden Actions\n(.*?)(?=\n## )", text, re.DOTALL)
   144→   144→    if forbidden_match:
   145→   145→        manifest["forbidden_actions"] = _extract_numbered_list(forbidden_match.group(1))
   146→   146→
   147→   147→    # --- Input Schema ---
   148→   148→    input_schema_match = re.search(r"## Input Schema\n```json\n(.*?)```", text, re.DOTALL)
   149→   149→    if input_schema_match:
   150→   150→        import json
   151→   151→        try:
   152→   152→            manifest["input_schema"] = json.loads(input_schema_match.group(1).strip())
   153→   153→        except json.JSONDecodeError:
   154→   154→            manifest["input_schema"] = input_schema_match.group(1).strip()
   155→   155→
   156→   156→    # --- Output Schema ---
   157→   157→    output_schema_match = re.search(r"## Output Schema\n```json\n(.*?)```", text, re.DOTALL)
   158→   158→    if output_schema_match:
   159→   159→        import json
   160→   160→        try:
   161→   161→            manifest["output_schema"] = json.loads(output_schema_match.group(1).strip())
   162→   162→        except json.JSONDecodeError:
   163→   163→            manifest["output_schema"] = output_schema_match.group(1).strip()
   164→   164→
   165→   165→    # --- Constraints ---
   166→   166→    constraints_match = re.search(r"## Constraints\n(.*?)(?=\n## )", text, re.DOTALL)
   167→   167→    if constraints_match:
   168→   168→        manifest["constraints"] = _extract_bullet_list(constraints_match.group(1))
   169→   169→
   170→   170→    # --- IPC Policy ---
   171→   171→    ipc_match = re.search(r"## IPC Policy\n(.*?)(?=\n## )", text, re.DOTALL)
   172→   172→    if ipc_match:
   173→   173→        manifest["ipc_policy"] = _parse_ipc_policy(ipc_match.group(1))
   174→   174→
   175→   175→    # --- Escalation Triggers ---
   176→   176→    esc_match = re.search(r"## Escalation Triggers\n(.*?)(?=$|\n# )", text, re.DOTALL)
   177→   177→    if esc_match:
   178→   178→        manifest["escalation_triggers"] = _extract_numbered_list(esc_match.group(1))
   179→   179→
   180→   180→    return manifest
   181→   181→
   182→   182→
   183→   183→def _extract_field(block: str, field_name: str) -> Optional[str]:
   184→   184→    """Extract a ``- Field: value`` entry from a markdown block."""
   185→   185→    pattern = rf"^-\s*{re.escape(field_name)}:\s*(.+)$"
   186→   186→    match = re.search(pattern, block, re.MULTILINE)
   187→   187→    return match.group(1).strip() if match else None
   188→   188→
   189→   189→
   190→   190→def _extract_numbered_list(block: str) -> list[str]:
   191→   191→    """Extract a numbered list (``1. ...``) from a markdown block."""
   192→   192→    return [
   193→   193→        line.strip().lstrip("0123456789.").strip()
   194→   194→        for line in block.strip().splitlines()
   195→   195→        if re.match(r"^\s*\d+\.\s", line)
   196→   196→    ]
   197→   197→
   198→   198→
   199→   199→def _extract_bullet_list(block: str) -> list[str]:
   200→   200→    """Extract a bullet list (``- ...``) from a markdown block."""
   201→   201→    return [
   202→   202→        line.strip().lstrip("-").strip()
   203→   203→        for line in block.strip().splitlines()
   204→   204→        if re.match(r"^\s*-\s", line)
   205→   205→    ]
   206→   206→
   207→   207→
   208→   208→def _parse_ipc_policy(block: str) -> Dict[str, Any]:
   209→   209→    """Parse the IPC Policy section into a structured dict."""
   210→   210→    policy: Dict[str, Any] = {}
   211→   211→
   212→   212→    # Allowed Tables
   213→   213→    tables_match = re.search(r"Allowed Tables:\n(.*?)(?=\n-\s|$)", block, re.DOTALL)
   214→   214→    if tables_match:
   215→   215→        policy["allowed_tables"] = _extract_bullet_list(tables_match.group(1))
   216→   216→
   217→   217→    # Allowed Endpoints
   218→   218→    endpoints_match = re.search(r"Allowed Endpoints:\n(.*?)(?=\n-\s|$)", block, re.DOTALL)
   219→   219→    if endpoints_match:
   220→   220→        policy["allowed_endpoints"] = _extract_bullet_list(endpoints_match.group(1))
   221→   221→
   222→   222→    # Resource Limits
   223→   223→    limits_match = re.search(r"Resource Limits:\s*\{([^}]+)\}", block)
   224→   224→    if limits_match:
   225→   225→        limits_str = "{" + limits_match.group(1) + "}"
   226→   226→        import json
   227→   227→        try:
   228→   228→            policy["resource_limits"] = json.loads(limits_str)
   229→   229→        except json.JSONDecodeError:
   230→   230→            policy["resource_limits"] = limits_str
   231→   231→
   232→   232→    return policy

   239→   239→@dataclass(frozen=True)
   240→   240→class IpcResponse:
   241→   241→    """Mirrors the Rust ``IpcResponse`` returned by the gRPC IPC handler.
   242→   242→
   243→   243→    Attributes
   244→   244→    ----------
   245→   245→    approved:
   246→   246→        Whether the request was approved by the policy engine.
   247→   247→    tier:
   248→   248→        The enforcement tier (``"GREEN"``, ``"AMBER"``, ``"RED"``).
   249→   249→        Only present when approved.
   250→   250→    violation_detail:
   251→   251→        JSON-encoded ``PolicyViolationDetail`` if the request was denied.
   252→   252→    escalation_reason:
   253→   253→        Human-readable reason if the request was escalated.
   254→   254→    raw:
   255→   255→        The full response dict as received from the IPC client.
   256→   256→    """
   257→   257→    approved: bool
   258→   258→    tier: Optional[str] = None
   259→   259→    violation_detail: Optional[Dict[str, Any]] = None
   260→   260→    escalation_reason: Optional[str] = None
   261→   261→    raw: Optional[Dict[str, Any]] = field(default=None, repr=False)
   262→   262→
   263→   263→    @property
   264→   264→    def status(self) -> str:
   265→   265→        """Return a string status: ``'approved'``, ``'policy_violation'``, or ``'escalated'``."""
   266→   266→        if self.approved:
   267→   267→            return "approved"
   268→   268→        if self.violation_detail is not None:
   269→   269→            return "policy_violation"
   270→   270→        if self.escalation_reason is not None:
   271→   271→            return "escalated"
   272→   272→        return "unknown"

   281→   281→    """Raised when the Rust control plane denies an IPC request.
   282→   282→
   283→   283→    Carries the full ``PolicyViolationDetail`` from the policy engine,
   284→   284→    enabling agents (and the orchestrator) to make informed retry or
   285→   285→    escalation decisions.
   286→   286→
   287→   287→    Attributes
   288→   288→    ----------
   289→   289→    agent_id:
   290→   290→        The agent that issued the violating request.

   292→   292→        The IPC operation type that was attempted (``read``, ``write``, etc.).
   293→   293→    target_resource:
   294→   294→        The table, endpoint, or command the agent tried to access.
   295→   295→    violated_section:
   296→   296→        The SKILL.md section that was violated (e.g. ``"IPC Policy"``,
   297→   297→        ``"Forbidden Actions"``).
   298→   298→    rule_text:
   299→   299→        The exact rule text from the SKILL.md that was violated.
   300→   300→    tier:
   301→   301→        The enforcement tier (``"GREEN"``, ``"AMBER"``, ``"RED"``).
   302→   302→    explanation:
   303→   303→        Human-readable explanation of why this is a violation.
   304→   304→    suggested_resolution:
   305→   305→        Suggested fix the agent can attempt.
   306→   306→    """
   307→   307→    agent_id: str
   308→   308→    operation_type: str
   309→   309→    target_resource: str
   310→   310→    violated_section: str
   311→   311→    rule_text: str
   312→   312→    tier: str
   313→   313→    explanation: str
   314→   314→    suggested_resolution: str
   315→   315→
   316→   316→    def __str__(self) -> str:
   317→   317→        return (
   318→   318→            f"PolicyViolationError: Agent '{self.agent_id}' attempted "
   319→   319→            f"{self.operation_type.upper()} on '{self.target_resource}'\n"
   320→   320→            f"  Tier:           {self.tier}\n"
   321→   321→            f"  Violated:       {self.violated_section}\n"
   322→   322→            f"  Rule:           {self.rule_text}\n"
   323→   323→            f"  Explanation:    {self.explanation}\n"
   324→   324→            f"  Suggested Fix:  {self.suggested_resolution}"
   325→   325→        )

   358→   358→# EliAgent — abstract base class

   360→   360→
   361→   361→class EliAgent(ABC):
   362→   362→    """Abstract base class for all 12 Eli-OS SEO agents.
   363→   363→
   364→   364→    Every concrete agent:
   365→   365→
   366→   366→    1. Reads its ``SKILL.md`` manifest at construction time.
   367→   367→    2. Connects to the Rust control plane via the gRPC IPC client.
   368→   368→    3. Routes all inter-agent and resource access through
   369→   369→       ``ipc_request()`` so the policy engine can enforce the
   370→   370→       Green / Amber / Red tier model.
   371→   371→    4. Reports results back to the kernel via ``report_result()``.
   372→   372→    5. Sends heartbeats to signal liveness.
   373→   373→
   374→   374→    Parameters
   375→   375→    ----------
   376→   376→    skill_md_path:
   377→   377→        Absolute path to this agent's ``SKILL.md`` manifest file.
   378→   378→    ipc_client:
   379→   379→        An ``EliIpcClient`` (or ``StubIpcClient``) instance.  If ``None``,
   380→   380→        a ``StubIpcClient`` is created with default settings.
   381→   381→    """
   382→   382→
   383→   383→    def __init__(
   384→   384→        self,
   385→   385→        skill_md_path: str,
   386→   386→        ipc_client: Optional[Any] = None,
   387→   387→    ) -> None:
   388→   388→        # Parse the SKILL.md manifest
   389→   389→        self._manifest: Dict[str, Any] = _parse_skill_md(skill_md_path)
   390→   390→
   391→   391→        # Structured logger bound to this agent
   392→   392→        self._log = structlog.get_logger(
   393→   393→            "eli_agent",
   394→   394→            agent_id=self.agent_id,
   395→   395→            domain=self.domain,
   396→   396→        )
   397→   397→        self._log.info(
   398→   398→            "agent_initialized",
   399→   399→            agent_id=self.agent_id,
   400→   400→            domain=self.domain,
   401→   401→            version=self._manifest.get("identity", {}).get("version", "unknown"),
   402→   402→        )
   403→   403→
   404→   404→        # IPC client — connects to the Rust kernel over gRPC/UDS
   405→   405→        if ipc_client is not None:
   406→   406→            self.ipc_client = ipc_client
   407→   407→        else:
   408→   408→            from .ipc_client import StubIpcClient
   409→   409→            self.ipc_client = StubIpcClient()
   410→   410→
   411→   411→        # Heartbeat task handle (set by heartbeat_loop)
   412→   412→        self._heartbeat_task: Optional[asyncio.Task[None]] = None
   413→   413→
   414→   414→    # -- Properties --------------------------------------------------------
   415→   415→
   416→   416→    @property
   417→   417→    def agent_id(self) -> str:
   418→   418→        """Return the agent's identity name from the parsed SKILL.md.
   419→   419→
   420→   420→        This value must match the key used in the Rust policy engine's
   421→   421→        manifest map.
   422→   422→        """
   423→   423→        return self._manifest.get("identity", {}).get("name", "unknown_agent")
   424→   424→
   425→   425→    @property
   426→   426→    def domain(self) -> str:
   427→   427→        """Return the agent's domain from the parsed SKILL.md."""
   428→   428→        return self._manifest.get("identity", {}).get("domain", "unknown_domain")
   429→   429→
   430→   430→    @property
   431→   431→    def manifest(self) -> Dict[str, Any]:
   432→   432→        """Return the full parsed SKILL.md manifest dictionary."""
   433→   433→        return self._manifest

   437→   437→    @abstractmethod

   454→   454→
   455→   455→        Returns
   456→   456→        -------
   457→   457→        dict
   458→   458→            The execution result, conforming to the agent's SKILL.md
   459→   459→            ``Output Schema``.
   460→   460→        """

   463→   463→    # -- IPC methods --------------------------------------------------------
   464→   464→
   465→   465→    async def ipc_request(
   466→   466→        self,
   467→   467→        operation_type: str,
   468→   468→        target_resource: str,
   469→   469→        payload: Optional[Dict[str, Any]] = None,
   470→   470→    ) -> IpcResponse:
   471→   471→        """Send a policy-gated IPC request to the Rust control plane.
   472→   472→
   473→   473→        Constructs an ``IpcRequest`` with a ULID and timestamp, sends it
   474→   474→        via the gRPC IPC client's ``evaluate()`` method, and interprets
   475→   475→        the response:
   476→   476→
   477→   477→        - **approved** → returns the ``IpcResponse``
   478→   478→        - **policy_violation** → raises :class:`PolicyViolationError`
   479→   479→        - **escalated** → raises :class:`EscalationRequiredError`
   480→   480→
   481→   481→        Parameters
   482→   482→        ----------
   483→   483→        operation_type:
   484→   484→            One of ``"read"``, ``"write"``, ``"execute"``, ``"delete"``.
   485→   485→        target_resource:
   486→   486→            The table name, endpoint URL, or command identifier.
   487→   487→        payload:
   488→   488→            Optional request payload (will be JSON-serialized by the client).
   489→   489→
   490→   490→        Returns
   491→   491→        -------
   492→   492→        IpcResponse
   493→   493→            The approved response from the policy engine.
   494→   494→
   495→   495→        Raises
   496→   496→        ------
   497→   497→        PolicyViolationError
   498→   498→            If the request violates the agent's SKILL.md policy.
   499→   499→        EscalationRequiredError
   500→   500→            If the request requires human operator review.
   501→   501→        """
   502→   502→        # Generate ULID for distributed tracing.
   503→   503→        # Dependency: ``pip install ulid-py``
   504→   504→        try:
   505→   505→            import ulid
   506→   506→            request_ulid = str(ulid.new())
   507→   507→        except ImportError:
   508→   508→            # Fallback: timestamp + random hex
   509→   509→            request_ulid = f"{int(time.time() * 1000):012x}{int(time.time() * 1000) % 0xFFFF:04x}"
   510→   510→
   511→   511→        # Build the request dict matching the Rust IpcRequest struct
   512→   512→        request: Dict[str, Any] = {
   513→   513→            "agent_id": self.agent_id,
   514→   514→            "operation_type": operation_type.upper(),
   515→   515→            "target_resource": target_resource,
   516→   516→            "payload": payload,
   517→   517→            "request_ulid": request_ulid,
   518→   518→            "timestamp": int(time.time()),
   519→   519→        }
   520→   520→
   521→   521→        self._log.debug(
   522→   522→            "ipc_request_sent",
   523→   523→            operation_type=operation_type,
   524→   524→            target_resource=target_resource,
   525→   525→            request_ulid=request_ulid,
   526→   526→        )
   527→   527→
   528→   528→        # Send to the Rust kernel via gRPC
   529→   529→        response_dict = await self.ipc_client.evaluate(request)
   530→   530→
   531→   531→        # Wrap in the typed IpcResponse dataclass
   532→   532→        response = IpcResponse(
   533→   533→            approved=response_dict.get("approved", False),
   534→   534→            tier=response_dict.get("tier"),
   535→   535→            violation_detail=response_dict.get("violation_detail"),
   536→   536→            escalation_reason=response_dict.get("escalation_reason"),
   537→   537→            raw=response_dict,
   538→   538→        )
   539→   539→
   540→   540→        # Dispatch based on the policy verdict
   541→   541→        status = response.status
   542→   542→        self._log.debug(
   543→   543→            "ipc_response_received",
   544→   544→            request_ulid=request_ulid,
   545→   545→            status=status,
   546→   546→            tier=response.tier,
   547→   547→        )
   548→   548→
   549→   549→        if status == "approved":
   550→   550→            return response
   551→   551→
   552→   552→        if status == "policy_violation":
   553→   553→            detail = response.violation_detail or {}
   554→   554→            raise PolicyViolationError(
   555→   555→                agent_id=detail.get("agent_id", self.agent_id),
   556→   556→                operation_type=detail.get("operation_type", operation_type),
   557→   557→                target_resource=detail.get("target_resource", target_resource),
   558→   558→                violated_section=detail.get("violated_section", "unknown"),
   559→   559→                rule_text=detail.get("rule_text", "unknown"),
   560→   560→                tier=detail.get("tier", "RED"),
   561→   561→                explanation=detail.get("explanation", "No explanation provided."),
   562→   562→                suggested_resolution=detail.get(
   563→   563→                    "suggested_resolution", "No resolution suggested."
   564→   564→                ),
   565→   565→            )
   566→   566→
   567→   567→        if status == "escalated":
   568→   568→            raise EscalationRequiredError(
   569→   569→                agent_id=self.agent_id,
   570→   570→                reason=response.escalation_reason or "No reason provided.",
   571→   571→                context=payload or {},
   572→   572→            )
   573→   573→
   574→   574→        # Unknown status — treat as a policy violation for safety
   575→   575→        self._log.warning(
   576→   576→            "ipc_unknown_status",
   577→   577→            request_ulid=request_ulid,
   578→   578→            response=response_dict,
   579→   579→        )
   580→   580→        raise PolicyViolationError(
   581→   581→            agent_id=self.agent_id,
   582→   582→            operation_type=operation_type,
   583→   583→            target_resource=target_resource,
   584→   584→            violated_section="unknown",
   585→   585→            rule_text="Unknown IPC response status",
   586→   586→            tier="RED",
   587→   587→            explanation=f"Received unknown status from IPC handler: {response_dict}",
   588→   588→            suggested_resolution="Check IPC handler logs and retry.",
   589→   589→        )

   640→   640→        if not ack.get("success", False):

   698→   698→    async def heartbeat_loop(self) -> None:
   699→   699→        """Run a background task that sends heartbeats every 30 seconds.
   700→   700→
   701→   701→        The heartbeat carries the agent's current status and resource
   702→   702→        usage to the Rust kernel.  The kernel checks these against the
   703→   703→        agent's declared IPC policy ``Resource Limits`` and may return
   704→   704→        a warning if limits are being approached.
   705→   705→
   706→   706→        This method is designed to be run as an ``asyncio`` background
   707→   707→        task::
   708→   708→
   709→   709→            asyncio.create_task(agent.heartbeat_loop())
   710→   710→
   711→   711→        The loop runs until cancelled (e.g. on agent shutdown).
   712→   712→        """
   713→   713→        import resource as resource_module  # stdlib
   714→   714→
   715→   715→        self._log.info("heartbeat_loop_started")
   716→   716→
   717→   717→        try:
   718→   718→            while True:
   719→   719→                # Gather resource telemetry
   720→   720→                try:
   721→   721→                    # rusage is Unix-only; fall back gracefully
   722→   722→                    usage = resource_module.getrusage(resource_module.RUSAGE_SELF)
   723→   723→                    memory_usage_mb = usage.ru_maxrss / 1024.0  # macOS: bytes; Linux: KB
   724→   724→                    if memory_usage_mb > 100_000:
   725→   725→                        # Likely Linux where ru_maxrss is in KB
   726→   726→                        memory_usage_mb = usage.ru_maxrss / 1024.0
   727→   727→                except (ValueError, AttributeError):
   728→   728→                    memory_usage_mb = 0.0
   729→   729→
   730→   730→                heartbeat_request: Dict[str, Any] = {
   731→   731→                    "agent_id": self.agent_id,
   732→   732→                    "status": "processing",
   733→   733→                    "tasks_completed": 0,  # Subclasses should track this
   734→   734→                    "memory_usage_mb": round(memory_usage_mb, 2),
   735→   735→                }
   736→   736→
   737→   737→                try:
   738→   738→                    ack = await self.ipc_client.heartbeat(heartbeat_request)
   739→   739→                    warning = ack.get("resource_warning")
   740→   740→                    if warning:
   741→   741→                        self._log.warning(
   742→   742→                            "heartbeat_resource_warning",
   743→   743→                            warning=warning,
   744→   744→                        )
   745→   745→                except Exception as exc:
   746→   746→                    # Heartbeat failures should not crash the agent
   747→   747→                    self._log.error(
   748→   748→                        "heartbeat_failed",
   749→   749→                        error=str(exc),
   750→   750→                    )
   751→   751→
   752→   752→                await asyncio.sleep(30)
   753→   753→
   754→   754→        except asyncio.CancelledError:
   755→   755→            self._log.info("heartbeat_loop_stopped")
   756→   756→
   757→   757→    def start_heartbeat(self) -> asyncio.Task[None]:
   758→   758→        """Start the heartbeat loop as a background task.
   759→   759→
   760→   760→        Returns the ``asyncio.Task`` so it can be cancelled on shutdown.
   761→   761→
   762→   762→        Returns
   763→   763→        -------
   764→   764→        asyncio.Task
   765→   765→            The background heartbeat task.
   766→   766→        """
   767→   767→        self._heartbeat_task = asyncio.create_task(self.heartbeat_loop())
   768→   768→        return self._heartbeat_task
   769→   769→
   770→   770→    async def shutdown(self) -> None:
   771→   771→        """Gracefully shut down the agent.
   772→   772→
   773→   773→        Cancels the heartbeat loop and closes the IPC client.
   774→   774→        """
   775→   775→        if self._heartbeat_task is not None:
   776→   776→            self._heartbeat_task.cancel()
   777→   777→            try:
   778→   778→                await self._heartbeat_task
   779→   779→            except asyncio.CancelledError:
   780→   780→                pass
   781→   781→
   782→   782→        if hasattr(self.ipc_client, "close"):
   783→   783→            await self.ipc_client.close()
   784→   784→
   785→   785→        self._log.info("agent_shutdown_complete", agent_id=self.agent_id)
