   128→def make_table(headers, rows, col_widths=None):
   129→    """Create a styled table with header row."""
   130→    avail_w = 460
   131→    if col_widths is None:
   132→        n = len(headers)
   133→        col_widths = [avail_w / n] * n
   134→    
   135→    hdr_cells = [Paragraph(h, styles['TableHeader']) for h in headers]
   136→    data = [hdr_cells]
   137→    for row in rows:
   138→        data.append([Paragraph(str(c), styles['TableCell']) if i == 0 else Paragraph(str(c), styles['TableCellCenter']) if len(str(c)) < 20 else Paragraph(str(c), styles['TableCell']) for i, c in enumerate(row)])
   139→    
   140→    t = Table(data, colWidths=col_widths, repeatRows=1)
   141→    style_cmds = [
   142→        ('BACKGROUND', (0, 0), (-1, 0), HEADER_FILL),
   143→        ('TEXTCOLOR', (0, 0), (-1, 0), WHITE),
   144→        ('FONTNAME', (0, 0), (-1, 0), 'LibSans-Bold'),
   145→        ('FONTSIZE', (0, 0), (-1, 0), 9),
   146→        ('BOTTOMPADDING', (0, 0), (-1, 0), 6),
   147→        ('TOPPADDING', (0, 0), (-1, 0), 6),
   148→        ('GRID', (0, 0), (-1, -1), 0.5, BORDER),
   149→        ('VALIGN', (0, 0), (-1, -1), 'TOP'),
   150→        ('LEFTPADDING', (0, 0), (-1, -1), 6),
   151→        ('RIGHTPADDING', (0, 0), (-1, -1), 6),
   152→        ('TOPPADDING', (0, 1), (-1, -1), 4),
   153→        ('BOTTOMPADDING', (0, 1), (-1, -1), 4),
   154→    ]
   155→    for i in range(1, len(data)):
   156→        if i % 2 == 0:
   157→            style_cmds.append(('BACKGROUND', (0, i), (-1, i), TABLE_STRIPE))
   158→    t.setStyle(TableStyle(style_cmds))
   159→    return t
