    76→def make_table(headers, rows, col_widths=None, header_bg=None):
    77→    if header_bg is None:
    78→        header_bg = HEADER_FILL
    79→    hdr = [Paragraph(h, s_table_header) for h in headers]
    80→    data = [hdr]
    81→    for row in rows:
    82→        data.append([Paragraph(str(c), s_table_cell) if not isinstance(c, Paragraph) else c for c in row])
    83→    if col_widths is None:
    84→        col_widths = [CONTENT_W / len(headers)] * len(headers)
    85→    t = Table(data, colWidths=col_widths, repeatRows=1)
    86→    cmds = [
    87→        ('BACKGROUND', (0, 0), (-1, 0), header_bg),
    88→        ('TEXTCOLOR', (0, 0), (-1, 0), WHITE),
    89→        ('FONTNAME', (0, 0), (-1, 0), 'LibSans-Bold'),
    90→        ('FONTSIZE', (0, 0), (-1, 0), 8),
    91→        ('BOTTOMPADDING', (0, 0), (-1, 0), 5),
    92→        ('TOPPADDING', (0, 0), (-1, 0), 5),
    93→        ('GRID', (0, 0), (-1, -1), 0.4, BORDER_COLOR),
    94→        ('VALIGN', (0, 0), (-1, -1), 'TOP'),
    95→        ('LEFTPADDING', (0, 0), (-1, -1), 4),
    96→        ('RIGHTPADDING', (0, 0), (-1, -1), 4),
    97→        ('TOPPADDING', (0, 1), (-1, -1), 3),
    98→        ('BOTTOMPADDING', (0, 1), (-1, -1), 3),
    99→    ]
   100→    for i in range(1, len(data)):
   101→        if i % 2 == 0:
   102→            cmds.append(('BACKGROUND', (0, i), (-1, i), TABLE_STRIPE))
   103→    t.setStyle(TableStyle(cmds))
   104→    return t
