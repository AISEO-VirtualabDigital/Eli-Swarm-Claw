     1→//! # eli-policy-engine
     2→//!
     3→//! Tiered policy enforcement engine for the Eli-OS control plane.
     4→//!
     5→//! This crate implements the **Green / Amber / Red** enforcement model that
     6→//! replaces the old all-or-nothing blocking approach:
     7→//!
     8→//! - **Green** — Low-risk read operations within the agent's declared IPC
     9→//!   policy. Approved immediately, no escalation.
    10→//!
    11→//! - **Amber** — Write operations within policy, or reads approaching
    12→//!   resource limits. Approved but logged for audit; may trigger soft
    13→//!   throttling.
    14→//!
    15→//! - **Red** — Forbidden actions, cross-domain writes, deletes, external
    16→//!   paid API calls, or resource limit violations. Blocked and escalated
    17→//!   to the human operator.
    18→//!
    19→//! ## Architecture
    20→//!
    21→//! ```text
    22→//!   IpcRequest
    23→//!       │
    24→//!       ▼
    25→//!  ┌─────────────┐     ┌──────────────────┐
    26→//!  │ PolicyEngine │────▶│ CapabilityManifest│ (from SKILL.md)
    27→//!  └──────┬──────┘     └──────────────────┘
    28→//!         │
    29→//!    evaluate()
    30→//!         │
    31→//!         ▼
    32→//!   IpcResponse
    33→//!   ├─ Approved { tier }
    34→//!   ├─ PolicyViolation { detail }
    35→//!   └─ Escalated { reason }
    36→//! ```
    37→//!
    38→//! ## Required dependencies
    39→//!
    40→//! ```toml
    41→//! [dependencies]
    42→//! eli-skill-parser = { path = "../eli-skill-parser" }
    43→//! serde = { version = "1", features = ["derive"] }
    44→//! thiserror = "1"
    45→//! ```
    46→

    53→// Enforcement tier
    54→// ---------------------------------------------------------------------------
    55→
    56→/// The three enforcement tiers that govern agent IPC requests.
    57→///
    58→/// Tier classification determines whether an operation proceeds
    59→/// immediately, proceeds with logging, or is blocked entirely.
    60→#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize, Hash)]
    61→pub enum EnforcementTier {
    62→    /// Low-risk read within declared IPC policy.
    63→    /// The operation proceeds without any additional scrutiny.
    64→    Green,
    65→
    66→    /// Elevated risk: writes within policy, or reads near resource limits.
    67→    /// The operation is allowed but audit-logged and may be throttled.
    68→    Amber,
    69→
    70→    /// High risk: deletes, cross-domain operations, external paid APIs,
    71→    /// or resource exhaustion. The operation is **blocked** and escalated
    72→    /// to a human operator.
    73→    Red,
    74→}
    75→
    76→impl std::fmt::Display for EnforcementTier {
    77→    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
    78→        match self {
    79→            EnforcementTier::Green => write!(f, "GREEN"),
    80→            EnforcementTier::Amber => write!(f, "AMBER"),
    81→            EnforcementTier::Red => write!(f, "RED"),
    82→        }
    83→    }
    84→}

    94→    Read,
    95→    /// Write / insert / update data.
    96→    Write,

    98→    Execute,
    99→    /// Delete data or a resource.
   100→    Delete,

   104→    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
   105→        match self {
   106→            OperationType::Read => write!(f, "READ"),
   107→            OperationType::Write => write!(f, "WRITE"),
   108→            OperationType::Execute => write!(f, "EXECUTE"),
   109→            OperationType::Delete => write!(f, "DELETE"),
   110→        }
   111→    }
   112→}

   118→/// An incoming IPC request from an agent.
   119→///
   120→/// Every request carries a ULID for end-to-end traceability and a
   121→/// Unix-epoch timestamp for audit ordering.
   122→#[derive(Debug, Clone, Serialize)]
   123→pub struct IpcRequest {
   124→    /// The agent's identity name (must match a key in the manifest map).
   125→    pub agent_id: String,
   126→
   127→    /// What kind of operation the agent wants to perform.
   128→    pub operation_type: OperationType,
   129→
   130→    /// The target resource: table name, endpoint URL, or command identifier.
   131→    pub target_resource: String,
   132→
   133→    /// Optional request payload (JSON body, SQL query, etc.).
   134→    pub payload: Option<String>,
   135→
   136→    /// Unique request identifier (ULID) for distributed tracing.
   137→    pub request_ulid: String,
   138→
   139→    /// Unix-epoch timestamp (seconds) when the request was issued.
   140→    pub timestamp: u64,
   141→}
   142→
   143→/// The policy engine's verdict on an IPC request.
   144→#[derive(Debug, Clone, Serialize)]
   145→pub enum IpcResponse {
   146→    /// The request is approved at the given tier.
   147→    Approved {
   148→        /// The enforcement tier the operation was classified into.
   149→        tier: EnforcementTier,
   150→    },
   151→
   152→    /// The request violates the agent's declared SKILL.md policy.
   153→    PolicyViolation {
   154→        /// Detailed information about what was violated and how to fix it.
   155→        violation: PolicyViolationDetail,
   156→    },
   157→
   158→    /// The request triggered an escalation that requires human review.
   159→    Escalated {
   160→        /// Human-readable reason for the escalation.
   161→        reason: String,
   162→    },
   163→}
   164→
   165→/// Rich detail about a policy violation, designed to be both machine-parseable
   166→/// and human-readable in audit logs.
   167→#[derive(Debug, Clone, Serialize)]
   168→pub struct PolicyViolationDetail {
   169→    /// The agent that issued the violating request.
   170→    pub agent_id: String,
   171→
   172→    /// The operation type that was attempted.
   173→    pub operation_type: String,
   174→
   175→    /// The resource the agent tried to access.
   176→    pub target_resource: String,
   177→
   178→    /// The SKILL.md section that was violated (e.g. `IPC Policy`, `Forbidden Actions`).
   179→    pub violated_section: String,
   180→
   181→    /// The specific rule text from the SKILL.md that was violated.
   182→    pub rule_text: String,
   183→
   184→    /// The enforcement tier this violation falls into.
   185→    pub tier: EnforcementTier,
   186→
   187→    /// Human-readable explanation of why this is a violation.
   188→    pub explanation: String,
   189→
   190→    /// Suggested resolution the agent operator can take.
   191→    pub suggested_resolution: String,
   192→}

   194→// ---------------------------------------------------------------------------
   195→// PolicyEngine
   196→// ---------------------------------------------------------------------------
   197→
   198→/// The central policy evaluation engine.
   199→///
   200→/// Holds the fleet's [`CapabilityManifest`] map (keyed by agent name) and
   201→/// evaluates every [`IpcRequest`] against the requesting agent's manifest.
   202→///
   203→/// # Hot reload
   204→///
   205→/// The engine supports **atomic hot reload** via [`hot_reload`](PolicyEngine::hot_reload).
   206→/// In production the engine should be wrapped in an `Arc<RwLock<PolicyEngine>>`:
   207→///
   208→/// ```text
   209→///   Writer (hot_reload)        Reader (evaluate)
   210→///         │                         │
   211→///   ┌─────┴─────┐           ┌──────┴──────┐
   212→///   │ write lock│           │  read lock  │
   213→///   └─────┬─────┘           └──────┬──────┘
   214→///         │                         │
   215→///         ▼                         ▼
   216→///    swap manifests          evaluate request
   217→/// ```
   218→///
   219→/// Writers block readers only for the duration of the `HashMap::clone()`,
   220→/// which is O(n) in the number of agents but completes in microseconds
   221→/// for typical fleet sizes (< 100 agents).
   222→pub struct PolicyEngine {
   223→    /// Agent-name → CapabilityManifest mapping, loaded from parsed SKILL.md files.
   224→    manifests: HashMap<String, CapabilityManifest>,
   225→}
   226→
   227→impl PolicyEngine {
   228→    /// Creates a new policy engine with the given manifest map.
   229→    ///
   230→    /// Call this after [`SkillParser::parse_directory`](eli_skill_parser::SkillParser::parse_directory)
   231→    /// to bootstrap the engine.
   232→    pub fn new(manifests: HashMap<String, CapabilityManifest>) -> Self {
   233→        Self { manifests }
   234→    }
   235→
   236→    /// Evaluates an incoming IPC request against the requesting agent's
   237→    /// SKILL.md manifest and returns an [`IpcResponse`].
   238→    ///
   239→    /// # Evaluation order
   240→    ///
   241→    /// 1. **Agent lookup** — is `request.agent_id` a known agent?
   242→    /// 2. **Forbidden actions** — does `target_resource` match any
   243→    ///    `forbidden_actions` entry exactly?
   244→    /// 3. **IPC policy check** — is the operation+resource combination
   245→    ///    allowed by the agent's declared IPC policy?
   246→    /// 4. **Tier classification** — assign Green / Amber / Red based on
   247→    ///    operation type and risk profile.
   248→    pub fn evaluate(&self, request: &IpcRequest) -> IpcResponse {
   249→        // --- Step 1: Agent lookup ---
   250→        let manifest = match self.manifests.get(&request.agent_id) {
   251→            Some(m) => m,
   252→            None => {
   253→                return IpcResponse::PolicyViolation {
   254→                    violation: generate_violation_detail(
   255→                        &request.agent_id,
   256→                        &request.operation_type.to_string(),
   257→                        &request.target_resource,
   258→                        "Identity",
   259→                        "Agent must be registered with a SKILL.md manifest",
   260→                        EnforcementTier::Red,
   261→                        format!(
   262→                            "No manifest found for agent '{}'. \n                        request.agent_id
   263→                        ),
   264→                        "Add a SKILL.md file for this agent to the manifest directory and reload.".to_string(),
   265→                    ),
   266→                };
   267→            }
   268→        };
   269→
   270→        // --- Step 2: Forbidden actions check ---
   271→        // Exact string match on target_resource against forbidden_actions.
   272→        // We also do a contains-check for API path prefixes.
   273→        for forbidden in &manifest.forbidden_actions {
   274→            let is_exact_match = request.target_resource == *forbidden;
   275→            let is_prefix_match = request.target_resource.starts_with(forbidden)
   276→                && forbidden.ends_with('/');
   277→            if is_exact_match || is_prefix_match {
   278→                return IpcResponse::PolicyViolation {
   279→                    violation: generate_violation_detail(
   280→                        &request.agent_id,
   281→                        &request.operation_type.to_string(),
   282→                        &request.target_resource,
   283→                        "Forbidden Actions",
   284→                        forbidden,
   285→                        EnforcementTier::Red,
   286→                        format!(
   287→                            "Agent '{}' attempted a forbidden action: '{}'. \
   288→                             This action is explicitly listed in the agent's \
   289→                             Forbidden Actions section.",
   290→                            request.agent_id, forbidden
   291→                        ),
   292→                        "Remove this action from the agent's workflow or \
   293→                         request a policy exception from the control plane \
   294→                         administrator."
   295→                            .to_string(),
   296→                    ),
   297→                };
   298→            }
   299→        }
   300→
   301→        // --- Step 3: IPC policy check ---
   302→        let ipc = &manifest.ipc_policy;
   303→
   304→        let tier = match request.operation_type {
   305→            OperationType::Read => {
   306→                // Reads must target an allowed table.
   307→                if !ipc.allowed_tables_read.contains(&request.target_resource)
   308→                    && !ipc
   309→                        .allowed_tables_read
   310→                        .iter()
   311→                        .any(|t| request.target_resource.starts_with(t))
   312→                {
   313→                    return IpcResponse::PolicyViolation {
   314→                        violation: generate_violation_detail(
   315→                            &request.agent_id,
   316→                            &request.operation_type.to_string(),
   317→                            &request.target_resource,
   318→                            "IPC Policy > Allowed Tables (Read)",
   319→                            &format!(
   320→                                "Allowed: {:?}",
   321→                                ipc.allowed_tables_read
   322→                            ),
   323→                            EnforcementTier::Amber,
   324→                            format!(
   325→                                "Agent '{}' attempted to read from table '{}' \
   326→                                 which is not in its allowed read list.",
   327→                                request.agent_id, request.target_resource
   328→                            ),
   329→                            format!(
   330→                                "Add '{}' to the agent's Allowed Tables (Read) \
   331→                                 list in SKILL.md, or restrict the agent's \
   332→                                 workflow to approved tables.",
   333→                                request.target_resource
   334→                            ),
   335→                        ),
   336→                    };
   337→                }
   338→                EnforcementTier::Green
   339→            }
   340→
   341→            OperationType::Write => {
   342→                // Writes must target an allowed write table.
   343→                let is_allowed = ipc.allowed_tables_write.contains(&request.target_resource)
   344→                    || ipc
   345→                        .allowed_tables_write
   346→                        .iter()
   347→                        .any(|t| request.target_resource.starts_with(t));
   348→
   349→                if !is_allowed {
   350→                    // Determine if this is a cross-domain write (Red) or just
   351→                    // an out-of-policy write (Amber). A cross-domain write
   352→                    // is one where the target table name suggests a domain
   353→                    // different from the agent's declared domain.
   354→                    let is_cross_domain = self.is_cross_domain(manifest, &request.target_resource);
   355→
   356→                    let tier = if is_cross_domain {
   357→                        EnforcementTier::Red
   358→                    } else {
   359→                        EnforcementTier::Amber
   360→                    };
   361→
   362→                    return IpcResponse::PolicyViolation {
   363→                        violation: generate_violation_detail(
   364→                            &request.agent_id,
   365→                            &request.operation_type.to_string(),
   366→                            &request.target_resource,
   367→                            "IPC Policy > Allowed Tables (Write)",
   368→                            &format!(
   369→                                "Allowed: {:?}",
   370→                                ipc.allowed_tables_write
   371→                            ),
   372→                            tier,
   373→                            if is_cross_domain {
   374→                                format!(
   375→                                    "Agent '{}' (domain: '{}') attempted to \
   376→                                     write to table '{}', which appears to \
   377→                                     belong to a different domain. \
   378→                                     Cross-domain writes are prohibited.",
   379→                                    request.agent_id,
   380→                                    manifest.identity.domain,
   381→                                    request.target_resource
   382→                                )
   383→                            } else {
   384→                                format!(
   385→                                    "Agent '{}' attempted to write to table '{}' \
   386→                                     which is not in its allowed write list.",
   387→                                    request.agent_id, request.target_resource
   388→                                )
   389→                            },
   390→                            if is_cross_domain {
   391→                                "Cross-domain writes require explicit \
   392→                                 multi-agent orchestration. Create a \
   393→                                 shared-workflow SKILL.md or request an \
   394→                                 administrator override."
   395→                                    .to_string()
   396→                            } else {
   397→                                format!(
   398→                                    "Add '{}' to the agent's Allowed Tables \
   399→                                     (Write) list in SKILL.md.",
   400→                                    request.target_resource
   401→                                )
   402→                            },
   403→                        ),
   404→                    };
   405→                }
   406→                // Writes within allowed tables are Amber-tier (elevated risk).
   407→                EnforcementTier::Amber
   408→            }
   409→
   410→            OperationType::Execute => {
   411→                // Executes must target an allowed endpoint.
   412→                let is_allowed = ipc.allowed_endpoints.contains(&request.target_resource)
   413→                    || ipc
   414→                        .allowed_endpoints
   415→                        .iter()
   416→                        .any(|ep| request.target_resource.starts_with(ep));
   417→
   418→                if !is_allowed {
   419→                    return IpcResponse::PolicyViolation {
   420→                        violation: generate_violation_detail(
   421→                            &request.agent_id,
   422→                            &request.operation_type.to_string(),
   423→                            &request.target_resource,
   424→                            "IPC Policy > Allowed Endpoints",
   425→                            &format!(
   426→                                "Allowed: {:?}",
   427→                                ipc.allowed_endpoints
   428→                            ),
   429→                            EnforcementTier::Amber,
   430→                            format!(
   431→                                "Agent '{}' attempted to call endpoint '{}' \
   432→                                 which is not in its allowed endpoints list.",
   433→                                request.agent_id, request.target_resource
   434→                            ),
   435→                            format!(
   436→                                "Add '{}' to the agent's Allowed Endpoints \
   437→                                 list in SKILL.md.",
   438→                                request.target_resource
   439→                            ),
   440→                        ),
   441→                    };
   442→                }
   443→
   444→                // Check for external paid API calls (heuristics: non-internal URLs).
   445→                if self.is_external_paid_api(&request.target_resource) {
   446→                    return IpcResponse::Escalated {
   447→                        reason: format!(
   448→                            "Agent '{}' is attempting to call an external \
   449→                             API endpoint '{}'. External API calls may incur \
   450→                             costs and require human approval.",
   451→                            request.agent_id, request.target_resource
   452→                        ),
   453→                    };
   454→                }
   455→
   456→                EnforcementTier::Amber
   457→            }
   458→
   459→            OperationType::Delete => {
   460→                // Deletes are always Red-tier and require escalation.
   461→                return IpcResponse::PolicyViolation {
   462→                    violation: generate_violation_detail(
   463→                        &request.agent_id,
   464→                        &request.operation_type.to_string(),
   465→                        &request.target_resource,
   466→                        "IPC Policy (Global)",
   467→                        "DELETE operations are prohibited for all agents \
   468→                         unless explicitly orchestrated by the control plane.",
   469→                        EnforcementTier::Red,
   470→                        format!(
   471→                            "Agent '{}' attempted to DELETE resource '{}'. \
   472→                             All delete operations are blocked at the \
   473→                             policy-engine level and require human \
   474→                             escalation.",
   475→                            request.agent_id, request.target_resource
   476→                        ),
   477→                        "If this deletion is legitimate, create a control-plane \
   478→                         orchestration task that performs the deletion under \
   479→                         human supervision. Do NOT route deletions through \
   480→                         individual agents."
   481→                            .to_string(),
   482→                    ),
   483→                };
   484→            }
   485→        };
   486→
   487→        // --- Step 4: Return approved response with classified tier ---
   488→        IpcResponse::Approved { tier }
   489→    }
   490→
   491→    /// Atomically replaces the entire manifest map.
   492→    ///
   493→    /// This is designed to be called from a writer thread/task while
   494→    /// readers hold `RwLock::read()` guards for `evaluate()`. The swap
   495→    /// is a single `HashMap` assignment, so readers will either see the
   496→    /// old or the new map—never a partial state.
   497→    ///
   498→    /// # Usage pattern
   499→    ///
   500→    /// ```ignore
   501→    /// let engine = Arc::new(RwLock::new(PolicyEngine::new(initial_manifests)));
   502→    ///
   503→    /// // Reader task (hot path):
   504→    /// let guard = engine.read().unwrap();
   505→    /// let response = guard.evaluate(&request);
   506→    ///
   507→    /// // Writer task (rare, e.g. on SKILL.md file change):
   508→    /// let mut guard = engine.write().unwrap();
   509→    /// guard.hot_reload(new_manifests);
   510→    /// ```
   511→    pub fn hot_reload(&mut self, manifests: HashMap<String, CapabilityManifest>) {
   512→        self.manifests = manifests;
   513→    }
   514→
   515→    /// Returns a reference to the current manifest map (for diagnostics).
   516→    pub fn manifests(&self) -> &HashMap<String, CapabilityManifest> {
   517→        &self.manifests
   518→    }
   519→
   520→    /// Returns the number of registered agents.
   521→    pub fn agent_count(&self) -> usize {
   522→        self.manifests.len()
   523→    }

   529→    /// Heuristic: determines whether a write target appears to cross
   530→    /// domain boundaries.
   531→    ///
   532→    /// A write is considered cross-domain if the agent's declared domain
   533→    /// does not appear as a prefix or substring in the target table name.
   534→    /// This is intentionally conservative—it's better to over-escalate
   535→    /// than to allow a silent cross-domain write.
   536→    fn is_cross_domain(&self, manifest: &CapabilityManifest, target: &str) -> bool {
   537→        let domain = manifest.identity.domain.to_lowercase();
   538→        let target_lower = target.to_lowercase();
   539→
   540→        // If the domain appears in the target, it's likely same-domain.
   541→        if target_lower.contains(&domain) {
   542→            return false;
   543→        }
   544→
   545→        // Shared tables (audit_log, system_config, etc.) are not cross-domain.
   546→        let shared_prefixes = ["audit_", "system_", "config_", "meta_"];
   547→        shared_prefixes
   548→            .iter()
   549→            .any(|prefix| target_lower.starts_with(prefix))
   550→    }
   551→
   552→    /// Heuristic: detects likely external paid API calls.
   553→    ///
   554→    /// We flag any endpoint that doesn't start with `/api/v1/` (our
   555→    /// internal convention) or that contains known external domains.
   556→    fn is_external_paid_api(&self, endpoint: &str) -> bool {
   557→        // Internal endpoints start with our standard prefix.
   558→        if endpoint.starts_with("/api/v1/") || endpoint.starts_with("/internal/") {
   559→            return false;
   560→        }
   561→
   562→        // Anything with a full URL scheme is external.
   563→        if endpoint.starts_with("http://") || endpoint.starts_with("https://") {
   564→            return true;
   565→        }
   566→
   567→        // Non-standard internal paths could be external.
   568→        !endpoint.starts_with('/')
   569→    }
   570→}

   601→}
